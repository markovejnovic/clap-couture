1use alloc::borrow::Cow;
2use std::ffi::{OsStr, OsString};

4/// `stem` with anything but letters, digits, `-` and `_` replaced, to name a file with.
5pub(crate) fn file_safe(stem: &OsStr) -> Cow<'_, OsStr> {
6    let is_safe = |char: char| char.is_alphanumeric() || char == '-' || char == '_';
7    let text = stem.to_string_lossy();
8    // Lossy text that borrowed was valid Unicode, so `stem` itself can be handed back.
9    if matches!(text, Cow::Borrowed(_)) && text.chars().all(is_safe) {
10        return Cow::Borrowed(stem);
11    }
12    let safe: String = text.chars().map(|char| if is_safe(char) { char } else { '-' }).collect();
13    Cow::Owned(OsString::from(safe))
14}
