10/// The ASCII codesets, as the `C` and `POSIX` locales report them. [`Encoding::for_label`] reads
11/// them as windows-1252, which takes any byte, so they name no encoding to fall back to.
12#[cfg(unix)]
13const ASCII_CODESETS: &[&str] = &["ansi_x3.4-1968", "us-ascii"];
14
15/// Codesets as macOS and the BSDs spell them, where that isn't also a label
16/// [`Encoding::for_label`] knows, with the label it knows. glibc's spellings already are.
17#[cfg(unix)]
18const CODESET_LABELS: &[(&str, &str)] = &[
19    ("big5hkscs", "big5-hkscs"),
20    ("cp949", "windows-949"),
21    ("euccn", "gb2312"),
22    ("eucjp", "euc-jp"),
23    ("euckr", "euc-kr"),
24];
25
26/// Text in no encoding that could be read, decoded as best it could be.
27#[derive(Debug)]
28pub(crate) struct NotUnicode(pub(crate) String);
29
30/// Representation of the system's locale.
31#[derive(Debug)]
32pub(crate) struct Locale {
33    /// The ANSI code page, which programs that aren't Unicode use.
34    #[cfg(windows)]
35    code_page: u32,
36    /// The codeset, as the C library names it, like `UTF-8`, `EUC-JP` or `eucJP`; empty for a
37    /// locale that names none.
38    #[cfg(unix)]
39    codeset: String,
40}
41
42impl Locale {
43    pub(crate) fn decode(bytes: &[u8]) -> Result<String, NotUnicode> {
44        let legacy = Self::read().and_then(|locale| locale.encoding());
45        decode_with(bytes, legacy).ok_or_else(|| NotUnicode(UTF_8.decode(bytes).0.into_owned()))
46    }
47
48    /// The encoding the locale uses for text that isn't Unicode. `None` for a locale that names
49    /// none, like `C`.
50    pub(crate) fn encoding(&self) -> Option<&'static Encoding> {
51        #[cfg(windows)]
52        {
53            codepage::to_encoding_no_replacement(u16::try_from(self.code_page).ok()?)
54        }
55
56        #[cfg(unix)]
57        {
58            let codeset = self.codeset.to_ascii_lowercase();
59            if ASCII_CODESETS.contains(&codeset.as_str()) {
60                return None;
61            }
62            let label = CODESET_LABELS
63                .iter()
64                .find(|(spelling, _)| *spelling == codeset)
65                .map_or(codeset.as_str(), |(_, label)| label);
66            Encoding::for_label(label.as_bytes()).filter(|encoding| *encoding != REPLACEMENT)
67        }
68
69        #[cfg(not(any(unix, windows)))]
70        {
71            None
72        }
73    }
74
75    /// Read the system's locale. `None` for a locale the C library doesn't have.
76    #[cfg_attr(
77        windows,
78        expect(
79            clippy::unnecessary_wraps,
80            reason = "Windows always has a code page; elsewhere the locale can be missing"
81        )
82    )]
83    pub(crate) fn read() -> Option<Self> {
84        #[cfg(windows)]
85        {
86            // SAFETY: takes nothing, reads a process setting.
87            let code_page = unsafe { windows_sys::Win32::Globalization::GetACP() };
88            Some(Self { code_page })
89        }
90
91        // A locale object of our own, used on this thread only, rather than `setlocale`, which
92        // would change the locale for the whole process.
93        #[cfg(unix)]
94        {
95            // SAFETY: an empty name asks for the locale the environment names; a null base
96            // starts from nothing.
97            let locale = unsafe {
98                libc::newlocale(libc::LC_CTYPE_MASK, c"".as_ptr(), core::ptr::null_mut())
99            };
100            if locale.is_null() {
101                return None;
102            }
103
104            // SAFETY: `locale` is a live locale object; this changes this thread's locale only.
105            let previous = unsafe { libc::uselocale(locale) };
106            // SAFETY: `CODESET` is a valid item.
107            let codeset = unsafe { libc::nl_langinfo(libc::CODESET) };
108            let codeset = (!codeset.is_null()).then(|| {
109                // SAFETY: a C string that lives while `locale` does, copied before it is freed.
110                unsafe { CStr::from_ptr(codeset) }.to_string_lossy().into_owned()
111            });
112
113            // SAFETY: `previous` is the locale this thread had.
114            unsafe {
115                libc::uselocale(previous);
116            }
117            // SAFETY: `locale` is no longer in use.
118            unsafe {
119                libc::freelocale(locale);
120            }
121
122            Some(Self { codeset: codeset.unwrap_or_default() })
123        }
124
125        #[cfg(not(any(unix, windows)))]
126        {
127            None
128        }
129    }
130}

132/// The text of a saved file: in the encoding its byte order mark names, as notepad's "Unicode"
133/// saves UTF-16, or else UTF-8, or else `legacy`, the encoding older editors save in. The mark
134/// is dropped. `None` when none of them fits.
135fn decode_with(bytes: &[u8], legacy: Option<&'static Encoding>) -> Option<String> {
136    if let Some((encoding, mark)) = Encoding::for_bom(bytes) {
137        let text = bytes.get(mark..).unwrap_or_default();
138        return encoding.decode_without_bom_handling_and_without_replacement(text).map(Into::into);
139    }
140    [Some(UTF_8), legacy.filter(|legacy| *legacy != UTF_8)]
141        .into_iter()
142        .flatten()
143        .find_map(|encoding| encoding.decode_without_bom_handling_and_without_replacement(bytes))
144        .map(Into::into)
145}

161    #[case::bsd_euc_cn("eucCN", Some(encoding_rs::GBK))]
162    #[case::bsd_big5_hkscs("Big5HKSCS", Some(encoding_rs::BIG5))]
163    #[case::bsd_cp949("CP949", Some(encoding_rs::EUC_KR))]
164    #[case::shift_jis("SJIS", Some(encoding_rs::SHIFT_JIS))]
