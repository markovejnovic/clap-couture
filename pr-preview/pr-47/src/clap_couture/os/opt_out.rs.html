1//! The environment variables that turn prompts off.
2
3use std::ffi::OsStr;
4
5/// A variable the user set to turn prompts off, as on a pseudo-terminal nobody answers.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub(crate) enum PromptOptOut {
8    /// `CI`, set to anything but `0` or `false`.
9    Ci,
10    /// `NO_PROMPT`, set to anything.
11    NoPrompt,
12}
13
14impl PromptOptOut {
15    /// The opt-out that `no_prompt` and `ci`, as `NO_PROMPT` and `CI` hold them, make; `None`
16    /// when neither does. An empty value counts as unset.
17    fn parse(no_prompt: Option<&OsStr>, ci: Option<&OsStr>) -> Option<Self> {
18        if no_prompt.is_some_and(|value| !value.is_empty()) {
19            return Some(Self::NoPrompt);
20        }
21        ci.filter(|value| !["", "0", "false"].map(OsStr::new).contains(value)).map(|_| Self::Ci)
22    }
23
24    /// Read `$NO_PROMPT` and `$CI`; `None` when neither turns prompts off.
25    pub(crate) fn read() -> Option<Self> {
26        let no_prompt = std::env::var_os("NO_PROMPT");
27        let ci = std::env::var_os("CI");
28        Self::parse(no_prompt.as_deref(), ci.as_deref())
29    }
30}
