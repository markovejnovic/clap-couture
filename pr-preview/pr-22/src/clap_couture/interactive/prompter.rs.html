17/// Asks through a [`Backend`], themed with a command's styles and worded with its [`Messages`].
18///
19/// Get one from [`CoutureParser::prompter`](crate::CoutureParser::prompter).
20#[derive(Debug)]
21pub struct Prompter<B = DefaultBackend> {
22    backend: B,
23    messages: Messages,
24    /// The command's name, which the editor's temp file is named after.
25    #[cfg(feature = "interactive-editor")]
26    name: String,
27    styles: Styles,
28}
29
30impl<B> Prompter<B>
31where
32    B: Backend,
33{
34    /// Ask a yes/no question, with `default` the answer Enter accepts.
35    ///
36    /// # Errors
37    /// - [`PromptError::Cancelled`] when the user backs out.
38    pub fn confirm(&self, question: &str, default: bool) -> Result<bool, PromptError> {
39        self.backend.confirm(&ConfirmPrompt { default, question, styles: self.styles() })
40    }
41
42    /// Ask in the user's editor, in a temp file with `extension`, like `md`, that starts with
43    /// `initial`. An empty answer reports [`Messages::editor_empty`] and cancels.
44    ///
45    /// # Errors
46    /// - [`PromptError::Cancelled`] when the user backs out or leaves the file empty.
47    /// - [`PromptError::Signalled`] when the editor dies of a signal.
48    /// - [`PromptError::Io`] when the editor cannot be found or started, or exits unsuccessfully.
49    #[cfg(feature = "interactive-editor")]
50    pub fn editor(
51        &self,
52        question: &str,
53        extension: &str,
54        initial: Option<&str>,
55    ) -> Result<String, PromptError> {
56        let answer = self.backend.editor(&EditorPrompt {
57            error: None,
58            extension,
59            file_stem: &self.name,
60            initial,
61            instructions: &self.messages.editor_instructions.get(),
62            not_utf8: &self.messages.editor_not_utf8.get(),
63            optional: false,
64            question,
65            returned_early: &self.messages.editor_returned_early.get(),
66            styles: self.styles(),
67            waiting: &self.messages.editor_waiting.get(),
68        })?;
69        answer.ok_or_else(|| {
70            match self.backend.report(&self.messages.editor_empty.get(), self.styles()) {
71                Ok(()) => PromptError::Cancelled,
72                Err(err) => err,
73            }
74        })
75    }
76
77    /// Ask to pick at least one of `T`'s visible values, returned in declared order.
78    ///
79    /// # Errors
80    /// - [`PromptError::Cancelled`] when the user backs out.
81    pub fn multi_select<T>(&self, question: &str) -> Result<Vec<T>, PromptError>
82    where
83        T: ValueEnum,
84    {
85        let selection_required = self.messages.selection_required.get();
86        let prompt = MultiSelectPrompt {
87            defaults: &[],
88            optional: false,
89            options: &visible_values::<T>(),
90            question,
91            selection_required: &selection_required,
92            styles: self.styles(),
93        };
94        loop {
95            let picked = self.backend.multi_select(&prompt)?;
96            match picked.iter().map(|name| value_named(name)).collect() {
97                Ok(values) => return Ok(values),
98                Err(complaint) => self.backend.report(&complaint, self.styles())?,
99            }
100        }
101    }
102
103    /// `None` when `backend` has no terminal to ask on.
104    pub(crate) fn new(backend: B, messages: Messages, cmd: &Command) -> Option<Self> {
105        backend.is_available().then(|| Self {
106            backend,
107            messages,
108            #[cfg(feature = "interactive-editor")]
109            name: cmd.get_name().to_owned(),
110            styles: cmd.get_styles().clone(),
111        })
112    }
113
114    /// Ask for a secret, typed masked and never echoed.
115    ///
116    /// # Errors
117    /// - [`PromptError::Cancelled`] when the user backs out.
118    pub fn secret(&self, question: &str) -> Result<SecretString, PromptError> {
119        let answer = self.backend.secret(&SecretPrompt {
120            optional: false,
121            question,
122            styles: self.styles(),
123        })?;
124        Ok(answer.unwrap_or_else(|| SecretString::from("")))
125    }
126
127    /// Ask to pick one of `T`'s visible values.
128    ///
129    /// # Errors
130    /// - [`PromptError::Cancelled`] when the user backs out.
131    pub fn select<T>(&self, question: &str) -> Result<T, PromptError>
132    where
133        T: ValueEnum,
134    {
135        let none_option = self.messages.none.get();
136        let prompt = SelectPrompt {
137            default: None,
138            none_option: &none_option,
139            optional: false,
140            options: &visible_values::<T>(),
141            question,
142            styles: self.styles(),
143        };
144        loop {
145            // A required select always picks something; a misbehaving backend's `None` is asked
146            // again like any other rejected answer.
147            let picked = self.backend.select(&prompt)?.unwrap_or_default();
148            match value_named(&picked) {
149                Ok(value) => return Ok(value),
150                Err(complaint) => self.backend.report(&complaint, self.styles())?,
151            }
152        }
153    }
154
155    const fn styles(&self) -> CommandStyles<'_> {
156        CommandStyles(&self.styles)
157    }
158
159    /// Ask for free text.
160    ///
161    /// # Errors
162    /// - [`PromptError::Cancelled`] when the user backs out.
163    pub fn text(&self, question: &str) -> Result<String, PromptError> {
164        let answer = self.backend.text(&TextPrompt {
165            default: None,
166            hint: None,
167            initial: None,
168            optional: false,
169            question,
170            styles: self.styles(),
171        })?;
172        Ok(answer.unwrap_or_default())
173    }
174}
175
176/// The `T` named `name`, or clap's wording for a value that names none.
177fn value_named<T>(name: &str) -> Result<T, String>
178where
179    T: ValueEnum,
180{
181    T::value_variants()
182        .iter()
183        .find(|value| value.to_possible_value().is_some_and(|value| value.matches(name, false)))
184        .cloned()
185        .ok_or_else(|| format!("invalid value '{name}'"))
186}
187
188/// `T`'s possible values that are not hidden, in declared order.
189fn visible_values<T>() -> Vec<PossibleValue>
190where
191    T: ValueEnum,
192{
193    T::value_variants()
194        .iter()
195        .filter_map(ValueEnum::to_possible_value)
196        .filter(|value| !value.is_hide_set())
197        .collect()
198}
