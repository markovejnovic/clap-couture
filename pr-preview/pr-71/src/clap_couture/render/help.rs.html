1//! The `--help` couture installs: clap's own text, with the subcommands listed by category.
2
3use clap::{Arg, Command, builder::StyledStr};
4
5use super::{CommandRenderExt as _, Word};
6use crate::{CommandCategoryMap, Messages};
7
8/// What couture shows below the usage, at `{after-help}`: the categorized subcommands, clap's
9/// arg sections, then the command's own after-help.
10///
11/// clap can't regroup its subcommand listing, so its `{all-args}` is rendered here from a copy of
12/// the command with the subcommands hidden, once for `-h` and once for `--help`. Rendering it
13/// through clap keeps positionals, `help_heading`s and hidden args exactly as clap shows them.
14pub(crate) struct HelpBody {
15    /// For `--help`, when clap shows more for it than for `-h`.
16    long: Option<String>,
17    short: String,
18    /// couture's template, styled as clap styles the built command.
19    template: String,
20}
21
22/// Rewords clap's English `Arguments:` and `Options:` headings with a command's [`Messages`].
23struct Headings<'cmd> {
24    cmd: &'cmd Command,
25    messages: &'cmd Messages,
26}
27
28/// The help templates couture tells apart from one the app set.
29pub(crate) trait HelpTemplates {
30    /// clap's templates for a command without one of its own: `DEFAULT_TEMPLATE`, and
31    /// `DEFAULT_NO_ARGS_TEMPLATE` for one with nothing to list (`clap_builder` 4.6.7,
32    /// `src/output/help_template.rs`).
33    const CLAP_DEFAULTS: [&str; 2] = [
34        "{before-help}{about-with-newline}\n{usage-heading} {usage}\n\n{all-args}{after-help}",
35        "{before-help}{about-with-newline}\n{usage-heading} {usage}{after-help}",
36    ];
37
38    /// couture's template: clap's, with the usage heading from `messages`, and everything below
39    /// the usage at `{after-help}`.
40    fn couture_template(&self, messages: &Messages) -> String;
41
42    /// Check whether `--help` shows more than `-h`, as clap decides it
43    /// (`Command::long_help_exists_` in `clap_builder` 4.6.7): otherwise clap shows `-h`'s help
44    /// for both. Exact on a command built from [`Self::without_command_hide_possible_values`].
45    fn has_long_help(&self) -> bool;
46
47    /// Check whether the app set its own help template. clap has no getter for it, so the help
48    /// is rendered under clap's default templates and compared.
49    fn has_own_template(&self) -> bool;
50
51    /// This command and those below it with no command-level `hide_possible_values`: clap decides
52    /// on long help before it applies a command's to its args (`Command::_build_self` in
53    /// `clap_builder` 4.6.7).
54    #[must_use]
55    fn without_command_hide_possible_values(self) -> Self;
56}
57
58impl HelpBody {
59    /// `cmd` showing this body at `{after-help}`, under couture's template unless the app set its
60    /// own.
61    pub(crate) fn install(self, cmd: Command) -> Command {
62        let cmd = if cmd.has_own_template() { cmd } else { cmd.help_template(self.template) };
63        let cmd = cmd.with_rendered_about();
64        // An empty after-help would still print its blank line.
65        let cmd = if self.short.is_empty() { cmd } else { cmd.after_help(self.short) };
66        match self.long {
67            Some(long) => cmd.after_long_help(long),
68            None => cmd,
69        }
70    }
71
72    /// The body for built `cmd`, its subcommands filed by `categories` and wrapped to `width`, its
73    /// headings worded by `messages`; `deciding` is `cmd` built from
74    /// [`HelpTemplates::without_command_hide_possible_values`].
75    pub(crate) fn new(
76        cmd: &Command,
77        deciding: &Command,
78        categories: CommandCategoryMap,
79        messages: &Messages,
80        width: Option<usize>,
81    ) -> Self {
82        let mut menu = StyledStr::new();
83        // `StyledStr`'s `fmt::Write` is infallible.
84        cmd.write_command_sections(&mut menu, categories, width).ok();
85        let menu = Word(&menu.ansi().to_string()).with_wide_lines_unstyled(width);
86
87        let args_only = cmd
88            .clone()
89            .disable_help_subcommand(true)
90            .mut_subcommands(|subcommand| subcommand.hide(true))
91            .help_template(format!("{{all-args}}{}", Headings::ARGS_END))
92            .with_rendered_arg_help();
93        let headings = Headings { cmd, messages };
94        let fitted =
95            |args: &StyledStr| Word(&headings.reword(args)).with_wide_lines_unstyled(width);
96        let short_args =
97            args_only.render_for_after_help(|mut args_only| fitted(&args_only.render_help()));
98        let long_args =
99            args_only.render_for_after_help(|mut args_only| fitted(&args_only.render_long_help()));
100
101        let after_help = cmd.get_after_help().map(|text| text.ansi().to_string());
102        let after_long_help = cmd.get_after_long_help().map(|text| text.ansi().to_string());
103        let short = Self::paragraphs([&menu, &short_args, after_help.as_deref().unwrap_or("")]);
104        let long = deciding.has_long_help().then(|| {
105            let after = after_long_help.or(after_help);
106            Self::paragraphs([&menu, &long_args, after.as_deref().unwrap_or("")])
107        });
108        Self { long, short, template: cmd.couture_template(messages) }
109    }
110
111    /// `parts` that aren't empty, a blank line apart.
112    fn paragraphs<const N: usize>(parts: [&str; N]) -> String {
113        parts.into_iter().filter(|part| !part.is_empty()).collect::<Vec<_>>().join("\n\n")
114    }
115}

117impl Headings<'_> {
118    /// Ends the `{all-args}` template, so clap's trim of the help leaves the args as clap writes
119    /// them above an after-help: an arg without help keeps its indent.
120    const ARGS_END: &'static str = "\u{E000}";
121
122    /// `rendered`, clap's `{all-args}` and [`Self::ARGS_END`], cut at the marker, with its default
123    /// section headings reworded.
124    fn reword(&self, rendered: &StyledStr) -> String {
125        let header = self.cmd.get_styles().get_header();
126        let styled = |heading: &str| format!("{header}{heading}{header:#}\n");
127        let rendered = rendered.ansi().to_string();
128        let mut text =
129            rendered.rsplit_once(Self::ARGS_END).map_or(&*rendered, |(args, _)| args).to_owned();
130        let defaults = Messages::DEFAULT;
131        for (default, worded) in [
132            (defaults.arguments.get(), self.messages.arguments.get()),
133            (defaults.options.get(), self.messages.options.get()),
134        ] {
135            if default == worded {
136                continue;
137            }
138            let (clap_heading, couture_heading) = (styled(&default), styled(&worded));
139            // A section starts the text or follows a blank line; clap writes the arguments first.
140            let at = if text.starts_with(&clap_heading) {
141                Some(0)
142            } else {
143                text.find(&format!("\n\n{clap_heading}")).map(|blank| blank.saturating_add(2))
144            };
145            if let Some(at) = at {
146                text.replace_range(at..at.saturating_add(clap_heading.len()), &couture_heading);
147            }
148        }
149        text
150    }
151}

153impl HelpTemplates for Command {
154    fn couture_template(&self, messages: &Messages) -> String {
155        let usage = self.get_styles().get_usage();
156        let usage_heading = messages.usage.get();
157        // clap's `{usage-heading}` is always English, so the heading is written out.
158        format!(
159            "{{before-help}}{{about-with-newline}}\n{usage}{usage_heading}{usage:#} \
160             {{usage}}{{after-help}}"
161        )
162    }
163
164    fn has_long_help(&self) -> bool {
165        let shows_more_long = |arg: &Arg| {
166            !arg.is_hide_set()
167                && (arg.get_long_help().is_some()
168                    || arg.is_hide_long_help_set()
169                    || arg.is_hide_short_help_set()
170                    || !arg.is_hide_possible_values_set()
171                        && arg
172                            .get_possible_values()
173                            .iter()
174                            .any(|value| !value.is_hide_set() && value.get_help().is_some()))
175        };
176        self.get_long_about().is_some()
177            || self.get_before_long_help().is_some()
178            || self.get_after_long_help().is_some()
179            || self.get_arguments().any(shows_more_long)
180    }
181
182    fn has_own_template(&self) -> bool {
183        let rendered = self.clone().render_help().ansi().to_string();
184        Self::CLAP_DEFAULTS.iter().all(|template| {
185            self.clone().help_template(*template).render_help().ansi().to_string() != rendered
186        })
187    }
188
189    fn without_command_hide_possible_values(self) -> Self {
190        self.hide_possible_values(false).mut_subcommands(Self::without_command_hide_possible_values)
191    }
