14/// How an editor prompt ended, for the line a backend leaves in the transcript.
15#[cfg(feature = "interactive-editor")]
16pub(super) enum Edited {
17    /// Answered, shown as the answer's first line, with `…` when more follow; empty for an
18    /// optional arg left unset.
19    Answered(String),
20    /// Backed out of, or answered with nothing where an answer is required.
21    Cancelled,
22    /// The editor failed; its error is reported once parsing ends.
23    Failed,
24}
25
26#[cfg(feature = "interactive-editor")]
27impl Edited {
28    /// How `answer`, the editor's outcome for a prompt that is `optional` or not, ended.
29    pub(super) fn of(answer: &Result<Option<String>, PromptError>, optional: bool) -> Self {
30        match answer {
31            Ok(Some(answer)) => Self::Answered(match answer.split_once('\n') {
32                Some((first_line, _)) => format!("{first_line}\u{2026}"),
33                None => answer.clone(),
34            }),
35            Ok(None) if optional => Self::Answered(String::new()),
36            Ok(None) | Err(PromptError::Cancelled | PromptError::Signalled(_)) => Self::Cancelled,
37            Err(_) => Self::Failed,
38        }
39    }
40}

42#[cfg(any(feature = "interactive-dialoguer", feature = "interactive-inquire"))]
43impl CommandStyles<'_> {
44    /// `question` as it shows in the header style of a backend that draws the header's `drawn`
45    /// effects: each reset in a styled question restores that style, which would otherwise end at
46    /// its first styled span.
47    pub(super) fn question_in_header(self, question: &str, drawn: Effects) -> Cow<'_, str> {
48        const RESET: &str = "\u{1b}[0m";
49        if !question.contains(RESET) {
50            return Cow::Borrowed(question);
51        }
52        let header = self.0.get_header();
53        let effects = [Effects::BOLD, Effects::DIMMED, Effects::ITALIC, Effects::UNDERLINE]
54            .into_iter()
55            .filter(|effect| drawn.contains(*effect) && header.get_effects().contains(*effect))
56            .fold(Effects::new(), |effects, effect| effects | effect);
57        let restored = Style::new().fg_color(header.get_fg_color()).effects(effects);
58        Cow::Owned(question.replace(RESET, &format!("{RESET}{restored}")))
59    }
60}

71#[cfg(test)]
72#[cfg(any(feature = "interactive-dialoguer", feature = "interactive-inquire"))]
73mod tests {
74    use clap::builder::{
75        Styles,
76        styling::{AnsiColor, Effects},
77    };
78    use rstest::rstest;
79
80    use crate::interactive::CommandStyles;
81
82    fn styles() -> Styles {
83        Styles::styled()
84            .header(AnsiColor::Magenta.on_default().effects(Effects::BOLD | Effects::UNDERLINE))
85    }
86
87    #[rstest]
88    #[case::plain_is_kept("Your name", Effects::BOLD, "Your name")]
89    #[case::a_reset_restores_the_drawn_header(
90        "Your \u{1b}[1mname\u{1b}[0m, in",
91        Effects::BOLD,
92        "Your \u{1b}[1mname\u{1b}[0m\u{1b}[1m\u{1b}[35m, in"
93    )]
94    #[case::only_the_effects_the_backend_draws(
95        "a\u{1b}[0mb",
96        Effects::BOLD | Effects::ITALIC | Effects::UNDERLINE,
97        "a\u{1b}[0m\u{1b}[1m\u{1b}[4m\u{1b}[35mb"
98    )]
99    fn question_in_header(#[case] question: &str, #[case] drawn: Effects, #[case] expected: &str) {
100        let styles = styles();
101        assert_eq!(CommandStyles(&styles).question_in_header(question, drawn), expected);
102    }
