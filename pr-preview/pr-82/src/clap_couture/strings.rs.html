3pub(crate) fn prefix_lines_into<'text>(
4    out: &mut impl io::Write,
5    prefix: &str,
6    texts: impl IntoIterator<Item = &'text str>,
7) -> io::Result<()> {
8    // No trailing whitespace after the prefix of an empty line.
9    let bare_prefix = prefix.trim_end();
10    for text in texts {
11        // `"".lines()` yields nothing, but an empty text still takes a line.
12        if text.is_empty() {
13            writeln!(out, "{bare_prefix}")?;
14        }
15        for line in text.lines() {
16            if line.is_empty() {
17                writeln!(out, "{bare_prefix}")?;
18            } else {
19                writeln!(out, "{prefix}{line}")?;
20            }
21        }
22    }
23    Ok(())
24}

32    fn prefixed(prefix: &str, texts: &[&str]) -> String {
33        let mut out = Vec::new();
34        let written = prefix_lines_into(&mut out, prefix, texts.iter().copied());
35        assert!(written.is_ok(), "writing to a `Vec` can't fail");
36        String::from_utf8(out).unwrap_or_default()
37    }
