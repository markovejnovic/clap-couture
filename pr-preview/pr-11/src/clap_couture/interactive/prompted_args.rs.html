1//! The static tree of prompted args that `#[derive(Couture)]` emits as `Couture::PROMPTED_ARGS`.
2
3use crate::Text;
4
5/// One `#[couture(prompt)]` or `#[couture(prompt = "...")]`.
6#[derive(Debug)]
7pub struct PromptAttr {
8    /// The clap id of the arg it is on.
9    pub arg_id: &'static str,
10    /// The `prompt = ...` override; `None` asks the arg's help text.
11    pub question: Option<Text>,
12}
13
14/// An arg to prompt for, and the subcommand path to the command that owns it.
15#[derive(Debug)]
16pub struct PromptedArg {
17    /// Its `#[couture(prompt)]`.
18    pub attr: &'static PromptAttr,
19    /// Subcommand names from the root, empty for a root arg.
20    pub subcommand_path: Vec<&'static str>,
21}
22
23/// What hangs off one command level besides its own prompted args.
24#[derive(Debug)]
25#[non_exhaustive]
26pub enum PromptedArgBranch {
27    /// A tree merged into this level: a flattened `Args` type, or the subcommand enum.
28    Flattened(&'static PromptedArgTree),
29    /// The subcommand with this name.
30    Subcommand(&'static str, &'static PromptedArgTree),
31}
32
33/// The prompted args of one command level, and the tree below it.
34#[derive(Debug)]
35pub struct PromptedArgTree {
36    /// Merged trees and subcommands.
37    pub branches: &'static [PromptedArgBranch],
38    /// The args prompted for at this level.
39    pub own_args: &'static [PromptAttr],
40}
41
42impl PromptedArgTree {
43    /// A tree with no prompted args, the default for a hand-written `Couture` impl.
44    pub const EMPTY: Self = Self { branches: &[], own_args: &[] };
45
46    fn collect(
47        &'static self,
48        subcommand_path: &mut Vec<&'static str>,
49        prompted_args: &mut Vec<PromptedArg>,
50    ) {
51        prompted_args.extend(
52            self.own_args
53                .iter()
54                .map(|attr| PromptedArg { attr, subcommand_path: subcommand_path.clone() }),
55        );
56        for branch in self.branches {
57            match branch {
58                PromptedArgBranch::Flattened(tree) => tree.collect(subcommand_path, prompted_args),
59                PromptedArgBranch::Subcommand(subcommand_name, tree) => {
60                    subcommand_path.push(subcommand_name);
61                    tree.collect(subcommand_path, prompted_args);
62                    subcommand_path.pop();
63                }
64            }
65        }
66    }
67
68    /// Every prompted arg in the tree: a level's own, then its branches', in declaration order.
69    #[must_use]
70    pub fn prompted_args(&'static self) -> Vec<PromptedArg> {
71        let mut prompted_args = Vec::new();
72        self.collect(&mut Vec::new(), &mut prompted_args);
73        prompted_args
74    }
75}
