21impl TextRenderer for MdTextRenderer<'_> {
22    fn render_display<W, D>(&self, out: &mut W, text: D) -> fmt::Result
23    where
24        W: fmt::Write,
25        D: Display,
26    {

30    fn render_str<W>(&self, out: &mut W, text: &str) -> fmt::Result
31    where
32        W: fmt::Write,
33    {
34        let mut line = LineWriter::new(out, text, self.styles.get_literal());
35        Parser::new(text).into_offset_iter().try_for_each(|(event, span)| line.write(event, span))
36    }
37}

39/// Writes markdown events as one line of styled text, keeping every character a reader needs.
40struct LineWriter<'out, 'src, W> {
41    /// Whether a block or line break awaits a space before the next text.
42    break_pending: bool,
43    /// Whether text is a code block's.
44    code_block: bool,
45    /// Each open link's or image's destination, `None` when its text already shows it.
46    destinations: Vec<Option<CowStr<'src>>>,
47    emphasis: usize,
48    literal: Style,
49    out: &'out mut W,
50    source: &'src str,
51    strong: usize,
52    written: bool,
53}
54
55impl<'out, 'src, W> LineWriter<'out, 'src, W>
56where
57    W: fmt::Write,
58{
59    fn close(&mut self, tag: TagEnd) -> fmt::Result {
60        match tag {
61            TagEnd::Emphasis => {
62                self.emphasis = self.emphasis.saturating_sub(1);
63                Ok(())
64            }
65            TagEnd::Strong => {
66                self.strong = self.strong.saturating_sub(1);
67                Ok(())
68            }
69            TagEnd::Link | TagEnd::Image => {
70                let destination = self.destinations.pop().flatten();
71                destination.map_or(Ok(()), |url| self.write_lines(&format!(" ({url})"), None))
72            }
73            TagEnd::CodeBlock => {
74                self.code_block = false;
75                self.break_pending = true;
76                Ok(())
77            }
78            TagEnd::Paragraph
79            | TagEnd::Heading(_)
80            | TagEnd::BlockQuote(_)
81            | TagEnd::HtmlBlock
82            | TagEnd::List(_)
83            | TagEnd::Item
84            | TagEnd::FootnoteDefinition
85            | TagEnd::DefinitionList
86            | TagEnd::DefinitionListTitle
87            | TagEnd::DefinitionListDefinition
88            | TagEnd::Table
89            | TagEnd::TableHead
90            | TagEnd::TableRow
91            | TagEnd::TableCell
92            | TagEnd::MetadataBlock(_) => {
93                self.break_pending = true;
94                Ok(())
95            }
96            TagEnd::Strikethrough | TagEnd::Superscript | TagEnd::Subscript => Ok(()),
97        }
98    }
99
100    const fn new(out: &'out mut W, source: &'src str, literal: &Style) -> Self {
101        Self {
102            break_pending: false,
103            code_block: false,
104            destinations: Vec::new(),
105            emphasis: 0,
106            literal: *literal,
107            out,
108            source,
109            strong: 0,
110            written: false,
111        }
112    }
113
114    fn open(&mut self, tag: Tag<'src>, span: Range<usize>) -> fmt::Result {
115        match tag {
116            Tag::Emphasis => {
117                self.emphasis = self.emphasis.saturating_add(1);
118                Ok(())
119            }
120            Tag::Strong => {
121                self.strong = self.strong.saturating_add(1);
122                Ok(())
123            }
124            Tag::Link { link_type, dest_url, title: _, id: _ } => {
125                let shown = matches!(link_type, LinkType::Autolink | LinkType::Email);
126                self.destinations.push((!shown && !dest_url.is_empty()).then_some(dest_url));
127                Ok(())
128            }
129            Tag::Image { link_type: _, dest_url, title: _, id: _ } => {
130                self.destinations.push((!dest_url.is_empty()).then_some(dest_url));
131                Ok(())
132            }
133            // The marker (`-`, `3.`) is the item's first word in the source.
134            Tag::Item => {
135                self.break_pending = true;
136                let marker = self.source.get(span).and_then(|item| item.split_whitespace().next());
137                marker.map_or(Ok(()), |marker| self.write_lines(marker, None))?;
138                self.break_pending = true;
139                Ok(())
140            }
141            Tag::CodeBlock(_) => {
142                self.code_block = true;
143                self.break_pending = true;
144                Ok(())
145            }
146            Tag::Paragraph
147            | Tag::Heading { level: _, id: _, classes: _, attrs: _ }
148            | Tag::BlockQuote(_)
149            | Tag::HtmlBlock
150            | Tag::List(_)
151            | Tag::FootnoteDefinition(_)
152            | Tag::DefinitionList
153            | Tag::DefinitionListTitle
154            | Tag::DefinitionListDefinition
155            | Tag::Table(_)
156            | Tag::TableHead
157            | Tag::TableRow
158            | Tag::TableCell
159            | Tag::MetadataBlock(_) => {
160                self.break_pending = true;
161                Ok(())
162            }
163            Tag::Strikethrough | Tag::Superscript | Tag::Subscript => Ok(()),
164        }
165    }
166
167    /// Writes `event`, which `span` locates in the source.
168    fn write(&mut self, event: Event<'src>, span: Range<usize>) -> fmt::Result {
169        match event {
170            Event::Start(tag) => self.open(tag, span),
171            Event::End(tag) => self.close(tag),
172            Event::Text(text) if self.code_block => self.write_lines(&text, Some(self.literal)),
173            Event::Text(text) | Event::Html(text) | Event::InlineHtml(text) => {
174                self.write_lines(&text, None)
175            }
176            Event::Code(code) => self.write_lines(&code, Some(self.literal)),
177            Event::SoftBreak | Event::HardBreak => {
178                self.break_pending = true;
179                Ok(())
180            }
181            Event::Rule
182            | Event::InlineMath(_)
183            | Event::DisplayMath(_)
184            | Event::FootnoteReference(_)
185            | Event::TaskListMarker(_) => self.write_source(span),
186        }
187    }
188
189    /// Writes `text` over `base` (plain when `None`), with each newline as a break.
190    fn write_lines(&mut self, text: &str, base: Option<Style>) -> fmt::Result {
191        text.split('\n').enumerate().try_for_each(|(index, line)| {
192            self.break_pending |= index > 0;
193            let line = line.trim_end_matches('\r');
194            if line.is_empty() {
195                return Ok(());
196            }
197            if self.break_pending && self.written {
198                self.out.write_char(' ')?;
199            }
200            self.break_pending = false;
201            self.written = true;
202            self.write_styled(line, base)
203        })
204    }
205
206    /// Writes the source `span` covers as is, set apart by breaks.
207    fn write_source(&mut self, span: Range<usize>) -> fmt::Result {
208        let source = self.source;
209        source.get(span).map_or(Ok(()), |text| {
210            self.break_pending = true;
211            self.write_lines(text.trim(), None)?;
212            self.break_pending = true;
213            Ok(())
214        })
215    }
216
217    fn write_styled(&mut self, text: &str, base: Option<Style>) -> fmt::Result {
218        let mut style = base.unwrap_or_default();
219        if self.strong > 0 {
220            style = style.bold();
221        }
222        if self.emphasis > 0 {
223            style = style.italic();
224        }
225        if style == Style::new() {
226            self.out.write_str(text)
227        } else {
228            write!(self.out, "{style}{text}{style:#}")
229        }
230    }
231}

239    fn rendered(text: &str) -> String {
240        let styles = Styles::plain();
241        let mut out = String::new();
242        let written = MdTextRenderer::new(&styles).render_str(&mut out, text);
243        assert!(written.is_ok(), "writing to a `String` can't fail");
244        out
245    }
