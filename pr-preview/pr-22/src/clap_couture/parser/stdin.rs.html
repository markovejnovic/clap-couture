1//! `#[couture(accept_stdin)]`: an arg the user left out takes what is piped into stdin.
2
3use std::{ffi::OsString, io};
4
5use clap::{Command, error::ErrorKind};
6
7use super::{
8    Failure,
9    prompting::{arg_at, blocked, relax, unfilled, unrequire},
10};
11use crate::interactive::{PromptedArg, StdinArg};
12
13/// The stdin arg on the path the user took, if it got no value, and the text piped in for it.
14/// `None` when nothing is piped, or the arg conflicts with one the user passed. Stdin is read
15/// only once the rest of `argv` parses, so `--help` never waits on it.
16///
17/// One line ending is dropped from the end, so `echo eu |` reads as `eu`.
18///
19/// # Errors
20/// - Reading stdin failed.
21/// - What was piped is not UTF-8.
22pub(super) fn piped<'stdin_arg>(
23    cmd: &Command,
24    prompted_args: &[PromptedArg],
25    stdin_args: &'stdin_arg [StdinArg],
26    argv: &[OsString],
27    read_piped_stdin: impl FnOnce() -> io::Result<Option<Vec<u8>>>,
28) -> Result<Option<(&'stdin_arg StdinArg, String)>, Failure> {
29    if stdin_args.is_empty() {
30        return Ok(None);
31    }
32    let relaxed = stdin_args.iter().fold(relax(cmd.clone(), prompted_args), |cmd, stdin_arg| {
33        unrequire(cmd, &stdin_arg.subcommand_path, stdin_arg.arg_id)
34    });
35    let Ok(matches) = relaxed.try_get_matches_from(argv) else { return Ok(None) };
36    // Built, so a flag's action has filled in its implicit default, and an arg can be printed.
37    let mut built = cmd.clone();
38    built.build();
39    let Some(stdin_arg) = stdin_args.iter().find(|stdin_arg| {
40        let (path, id) = (stdin_arg.subcommand_path.as_slice(), stdin_arg.arg_id);
41        unfilled(&built, &matches, path, id).is_some() && !blocked(cmd, &matches, path, id, &[])
42    }) else {
43        return Ok(None);
44    };
45    let Some(piped) = read_piped_stdin().map_err(Failure::Io)?.filter(|piped| !piped.is_empty())
46    else {
47        return Ok(None);
48    };
49    let Ok(mut text) = String::from_utf8(piped) else {
50        let arg = arg_at(&built, &stdin_arg.subcommand_path, stdin_arg.arg_id)
51            .map_or_else(|| stdin_arg.arg_id.to_owned(), ToString::to_string);
52        let message = format!("{arg} got invalid UTF-8 from stdin");
53        return Err(Failure::Clap(
54            clap::Error::raw(ErrorKind::InvalidUtf8, message).format(&mut cmd.clone()),
55        ));
56    };
57    if text.ends_with('\n') {
58        text.pop();
59        if text.ends_with('\r') {
60            text.pop();
61        }
62    }
63    Ok(Some((stdin_arg, text)))
64}
