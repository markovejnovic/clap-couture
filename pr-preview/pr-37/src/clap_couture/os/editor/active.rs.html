21/// An editor that exits sooner than this and leaves the file as it was most likely handed the
22/// file to a running instance, like `code` without `--wait`, rather than being done with it.
23const RETURNED_EARLY: Duration = Duration::from_millis(500);
24
25/// The waiting hint, while the editor is open.
26///
27/// Some editors they kind of suck in that they do not wait until the file passed is open. `VsCode`
28/// et al. Consequently, we have to tell the user to "finish" their command.
29///
30/// When they're done, we need to erase the hint.
31#[derive(Debug)]
32struct Hint {
33    /// Erased once the editor exits cleanly.
34    erased: bool,
35    text: String,
36}
37
38/// An editor open on its file. Dropping it waits for the editor, then removes a temporary file.
39#[must_use = "dropping it waits for the editor to exit"]
40pub(crate) struct Opened {
41    /// Removed on drop, after the editor exits.
42    _dir: Option<TempDir>,
43    child: Option<InteractiveChild>,
44    editor: Editor,
45    hint: Option<Hint>,
46    path: PathBuf,
47    prompt_marker: PromptMarker,
48    started: Instant,
49    written: Vec<u8>,
50}
51
52impl Opened {
53    /// Show the hint and start the editor.
54    fn launch(&mut self) -> Result<(), SpawnError> {
55        // Built before the hint shows, so a missing editor reports without one.
56        let mut command = self.editor.command(&self.path)?;
57        if let Some(hint) = &self.hint {
58            let end = if hint.erased { ' ' } else { '\n' };
59            write!(io::stderr(), "{}{end}", hint.text)?;
60        }
61
62        self.started = Instant::now();
63        self.child = Some(command.spawn_interactive().map_err(|err| {
64            if err.kind() == io::ErrorKind::NotFound {
65                SpawnError::Editor(EditorError::NotFound(self.editor.command.clone()))
66            } else {
67                SpawnError::Io(err)
68            }
69        })?);
70        Ok(())
71    }
72
73    /// Read the file as it is now.
74    pub(crate) fn read(&self) -> io::Result<Vec<u8>> {
75        fs::read(&self.path)
76    }
77
78    /// Open the editor on the file again.
79    pub(crate) fn reopen(&mut self) -> Result<(), SpawnError> {
80        drop(self.child.take());
81        self.launch()
82    }
83
84    pub(super) fn start(
85        editor: &Editor,
86        options: EditorSpawnOptions<'_>,
87    ) -> Result<Self, SpawnError> {
88        let (dir, path) = match options.place {
89            FilePlace::Path(path) => (None, path),
90            FilePlace::TempFile(name) => {
91                let prefix = crate::os::process_name().map(|mut prefix| {
92                    prefix.push("-");
93                    prefix
94                });
95                let mut builder = tempfile::Builder::new();
96                if let Some(prefix) = &prefix {
97                    builder.prefix(prefix);
98                }
99                let dir = builder.tempdir()?;
100                let name = name.file_name().map_or_else(default_file_name, Into::into);
101                let path = dir.path().join(name);
102                (Some(dir), path)
103            }
104        };
105
106        let mut written = Vec::new();
107        options.prompt_marker.write(&mut written, options.contents, options.prompt)?;
108        fs::write(&path, &written)?;
109
110        let hint = options.hint.filter(|hint| !hint.is_empty() && io::stderr().is_terminal());
111
112        let dumb_terminal = Term::read().map_or(cfg!(not(windows)), |term| term.is_dumb());
113        let hint = hint.map(|text| Hint {
114            erased: !dumb_terminal && !editor.is_console(),
115            text: text.to_owned(),
116        });
117
118        let mut opened = Self {
119            child: None,
120            _dir: dir,
121            editor: editor.clone(),
122            hint,
123            path,
124            prompt_marker: options.prompt_marker,
125            started: Instant::now(),
126            written,
127        };
128        opened.launch()?;
129        Ok(opened)
130    }
131
132    /// Wait for the editor to exit, then read the file.
133    pub(crate) fn wait(&mut self) -> Result<Vec<u8>, WaitError> {
134        let Some(child) = self.child.take() else { return Ok(self.read()?) };
135        let exit = child.wait();
136
137        if let Some(hint) = self.hint.as_ref().filter(|hint| hint.erased) {
138            // Erase the hint and its trailing space after a clean exit; otherwise keep it,
139            // ending its line.
140            let mut stderr = io::stderr();
141            let erasing = if matches!(exit, Ok(Exit::Success)) {
142                let columns = term::visible_width(&hint.text).saturating_add(1);
143                let vt = term::has_vt(&stderr);
144                term::erase_line(&mut stderr, columns, vt)
145            } else {
146                writeln!(stderr)
147            };
148
149            drop(erasing);
150        }
151
152        match exit? {
153            Exit::Success => {}
154            Exit::Failed => return Err(WaitError::Failed(self.editor.command.clone())),
155            Exit::Killed(signal) => return Err(WaitError::Killed(signal)),
156        }
157
158        let saved = fs::read(&self.path)?;
159        if self.started.elapsed() < RETURNED_EARLY && saved == self.written {
160            return Err(WaitError::ReturnedEarly);
161        }
162
163        Ok(saved)
164    }
165
166    /// Write `contents` and `prompt` to the file, for the next [`Opened::reopen`].
167    pub(crate) fn write(&mut self, contents: &str, prompt: Option<&str>) -> io::Result<()> {
168        self.written.clear();
169        self.prompt_marker.write(&mut self.written, contents, prompt)?;
170        fs::write(&self.path, &self.written)?;
171        Ok(())
172    }
173}
174
175impl Drop for Opened {
176    fn drop(&mut self) {
177        // The editor may still have the file open: wait for it before the directory goes.
178        drop(self.child.take());
179    }
180}

182/// Why the editor couldn't be opened.
183#[derive(Debug)]
184pub(crate) enum SpawnError {
185    Editor(EditorError),
186    Io(io::Error),
187}
188
189impl From<EditorError> for SpawnError {
190    fn from(err: EditorError) -> Self {
191        Self::Editor(err)
192    }
193}
194
195impl From<io::Error> for SpawnError {
196    fn from(err: io::Error) -> Self {
197        Self::Io(err)
198    }
199}
200
201/// Why the editor's file couldn't be read back.
202#[derive(Debug)]
203pub(crate) enum WaitError {
204    /// The editor exited unsuccessfully. Holds its command.
205    Failed(OsString),
206    Io(io::Error),
207    /// The editor died of the terminal's signal.
208    Killed(Signal),
209    /// The editor returned at once and left the file as it was.
210    ReturnedEarly,
211}
212
213impl From<io::Error> for WaitError {
214    fn from(err: io::Error) -> Self {
215        Self::Io(err)
216    }
217}

236    #[test]
237    fn wait_reads_back_what_the_editor_saved_without_the_prompt() {
238        let _handling = crate::os::signal::tests::lock_handling();
239        let opened = editor(r#"sh -c 'printf "Saved.\n" > "$0"'"#)
240            .spawn(EditorSpawnOptions::builder().prompt(PROMPT).build());
241        let saved = opened.ok().and_then(|mut opened| opened.wait().ok());
242        let answer = saved.and_then(|saved| Locale::decode(&saved).ok());
243        let answer = answer.map(|text| PromptMarker::default().strip(&text, Some(PROMPT)));
244        assert_eq!(answer, Some(Some("Saved.".to_owned())));
