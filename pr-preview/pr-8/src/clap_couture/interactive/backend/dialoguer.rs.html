19/// A dialoguer failure.
20struct DialoguerError(dialoguer::Error);
21
22/// Prompts drawn by dialoguer's colorful theme.
23///
24/// At a secret prompt, Ctrl-C interrupts the process rather than returning
25/// [`PromptError::Cancelled`]: dialoguer reads the secret as a plain line, with only echo off. Esc
26/// is taken as part of the answer, and the interrupted process may leave the terminal's echo off.
27#[derive(Clone, Copy, Debug, Default)]
28pub struct Dialoguer;
29
30impl Backend for Dialoguer {
31    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
32        let theme = ColorfulTheme::from(prompt.styles);
33        Confirm::with_theme(&theme)
34            .with_prompt(prompt.question)
35            .default(prompt.default)
36            .interact_opt()
37            .map_err(DialoguerError)?
38            .ok_or(PromptError::Cancelled)
39    }
40
41    /// The closed editor leaves the question with the answer's first line, as an answered input
42    /// does; a cancelled one leaves nothing, as dialoguer clears a cancelled prompt.
43    #[cfg(feature = "interactive-editor")]
44    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
45        use dialoguer::theme::Theme as _;
46
47        let answer = super::EditorInteraction::start(self, prompt)
48            .and_then(super::EditorInteraction::answer);
49        if let Edited::Answered(line) = Edited::of(&answer, prompt.optional) {
50            let mut transcript = String::new();
51            let theme = ColorfulTheme::from(prompt.styles);
52            if theme.format_input_prompt_selection(&mut transcript, prompt.question, &line).is_ok()
53            {
54                // Best effort: after a hang-up there is no terminal left to draw on.
55                drop(console::Term::stderr().write_line(&transcript));
56            }
57        }
58        answer
59    }
60
61    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
62        let theme = ColorfulTheme::from(prompt.styles);
63        let picked_by_default: Vec<bool> = prompt
64            .options
65            .iter()
66            .map(|option| prompt.defaults.iter().any(|default| default == option.get_name()))
67            .collect();
68        // dialoguer's `MultiSelect` has no way to demand at least one selection, so when a required
69        // prompt comes back empty, it says so and is shown again.
70        loop {
71            let picked = MultiSelect::with_theme(&theme)
72                .with_prompt(prompt.question)
73                .items(prompt.options.iter().map(PossibleValue::get_name))
74                .defaults(&picked_by_default)
75                .interact_opt()
76                .map_err(DialoguerError)?
77                .ok_or(PromptError::Cancelled)?;
78            if prompt.optional || !picked.is_empty() {
79                return Ok(picked
80                    .into_iter()
81                    .filter_map(|index| prompt.options.get(index))
82                    .map(|option| option.get_name().to_owned())
83                    .collect());
84            }
85            self.report(prompt.selection_required, prompt.styles)?;
86        }
87    }
88
89    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
90        let theme = ColorfulTheme::from(styles);
91        console::Term::stderr()
92            .write_line(&theme.error_style.apply_to(error).to_string())
93            .map_err(PromptError::Io)
94    }
95
96    fn secret(&self, prompt: &SecretPrompt<'_>) -> Result<Option<SecretString>, PromptError> {
97        let theme = ColorfulTheme::from(prompt.styles);
98        let answer = Password::with_theme(&theme)
99            .with_prompt(prompt.question)
100            .allow_empty_password(prompt.optional)
101            .interact()
102            .map_err(DialoguerError)?;
103        Ok(into_secret(answer))
104    }
105
106    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
107        let theme = ColorfulTheme::from(prompt.styles);
108        let names: Vec<&str> = prompt.options.iter().map(PossibleValue::get_name).collect();
109        let none = prompt.optional.then_some(prompt.none_option);
110        let items: Vec<&str> = names.iter().copied().chain(none).collect();
111        let select = Select::with_theme(&theme).with_prompt(prompt.question).items(&items);
112        let default =
113            prompt.default.and_then(|default| names.iter().position(|name| *name == default));
114        let picked = match default {
115            Some(index) => select.default(index),
116            None => select,
117        }
118        .interact_opt()
119        .map_err(DialoguerError)?
120        .ok_or(PromptError::Cancelled)?;
121        // Past the options is the none option, only listed for an optional prompt.
122        Ok(names.get(picked).map(|name| (*name).to_owned()))
123    }
124
125    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
126        let theme = ColorfulTheme::from(prompt.styles);
127        let input = Input::<String>::with_theme(&theme)
128            .with_prompt(prompt.hint.map_or_else(
129                || prompt.question.to_owned(),
130                |hint| format!("{} {hint}", prompt.question),
131            ))
132            .allow_empty(prompt.optional);
133        let input = match prompt.initial {
134            Some(initial) => input.with_initial_text(initial),
135            None => input,
136        };
137        let answer = match prompt.default {
138            Some(default) => input.default(default.to_owned()),
139            None => input,
140        }
141        .interact_text()
142        .map_err(DialoguerError)?;
143        Ok(Some(answer).filter(|answer| !answer.is_empty()))
144    }
145}

147impl From<DialoguerError> for PromptError {
148    fn from(DialoguerError(err): DialoguerError) -> Self {
149        let dialoguer::Error::IO(err) = err;
150        Self::from(err)
151    }
152}

154impl From<CommandStyles<'_>> for ColorfulTheme {
155    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
156        Self {
157            defaults_style: console::Style::from(ClapStyle(styles.get_placeholder())),
158            error_style: console::Style::from(ClapStyle(styles.get_error())),
159            prompt_style: console::Style::from(ClapStyle(styles.get_header())),
160            values_style: console::Style::from(ClapStyle(styles.get_literal())),
161            ..Self::default()
162        }
163    }
