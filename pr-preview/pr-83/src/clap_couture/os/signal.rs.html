26/// How this process handles each [`Signal`], changed only with it locked, so that guards on
27/// different threads never interleave. [`Signal::raise_by_default`] takes it for good.
28#[cfg(all(unix, feature = "interactive-editor"))]
29static HANDLING: Mutex<Handling> =
30    Mutex::new(Handling { hangup: None, interrupt: None, quit: None, terminate: None });
31
32/// Set once `SIGTERM` reaches [`ForwardedTermination::forward`]; cleared when the first guard
33/// installs it.
34#[cfg(all(unix, feature = "interactive-editor"))]
35static TERMINATION_RECEIVED: AtomicBool = AtomicBool::new(false);
36
37/// The children [`ForwardedTermination`] guards pass `SIGTERM` on to; 0 in a free slot.
38#[cfg(all(unix, feature = "interactive-editor"))]
39static TERMINATION_RECIPIENTS: [AtomicI32; 8] = [const { AtomicI32::new(0i32) }; 8];
40
41/// Each signal [`IgnoredSignals`] and [`ForwardedTermination`] guards are holding.
42#[cfg(all(unix, feature = "interactive-editor"))]
43struct Handling {
44    hangup: Option<Held>,
45    interrupt: Option<Held>,
46    quit: Option<Held>,
47    /// `SIGTERM`, passed on rather than ignored.
48    terminate: Option<Held>,
49}
50
51#[cfg(all(unix, feature = "interactive-editor"))]
52impl Handling {
53    /// Ignore `signal` for one more guard: the first saves its action and ignores it. `false`
54    /// when it can't be ignored.
55    fn hold(&mut self, signal: Signal) -> bool {
56        Held::add(self.slot(signal), || signal.ignore())
57    }
58
59    /// Pass `SIGTERM` on for one more guard: the first installs the handler. `false` when it
60    /// can't, or the app handles or ignores `SIGTERM` itself.
61    fn hold_termination(&mut self) -> bool {
62        Held::add(&mut self.terminate, ForwardedTermination::install)
63    }
64
65    /// Lock the process's handling, even if a thread panicked holding it: the lock guards no
66    /// invariant a panic can break halfway.
67    fn lock() -> MutexGuard<'static, Self> {
68        HANDLING.lock().unwrap_or_else(PoisonError::into_inner)
69    }
70
71    /// Let go of `signal` for one guard: the last restores the action the first saved.
72    fn release(&mut self, signal: Signal) {
73        Held::remove(self.slot(signal));
74    }
75
76    /// Let go of `SIGTERM` for one guard: the last restores the action the first saved.
77    fn release_termination(&mut self) {
78        Held::remove(&mut self.terminate);
79    }
80
81    /// Whether guards are ignoring `signal`, and how many.
82    const fn slot(&mut self, signal: Signal) -> &mut Option<Held> {
83        match signal {
84            Signal::Hangup => &mut self.hangup,
85            Signal::Interrupt => &mut self.interrupt,
86            Signal::Quit => &mut self.quit,
87        }
88    }
89}

91/// A signal some guards are holding: ignoring it, or passing it on.
92#[cfg(all(unix, feature = "interactive-editor"))]
93struct Held {
94    /// The guards holding it; at least one.
95    holders: NonZeroUsize,
96    /// The handler the first of them installed.
97    installed: libc::sighandler_t,
98    /// The action it had before the first of them.
99    previous: libc::sigaction,
100    /// The signal's number.
101    raw: libc::c_int,
102}
103
104#[cfg(all(unix, feature = "interactive-editor"))]
105impl Held {
106    /// One more guard holds `slot`; the first takes the signal over with `take_over`. `false`
107    /// when it can't.
108    fn add(slot: &mut Option<Self>, take_over: impl FnOnce() -> Option<Self>) -> bool {
109        if let Some(held) = slot {
110            held.holders = held.holders.saturating_add(1);
111            return true;
112        }
113        let Some(held) = take_over() else { return false };
114        *slot = Some(held);
115        true
116    }
117
118    /// One guard lets go of `slot`; the last restores the action the first replaced.
119    fn remove(slot: &mut Option<Self>) {
120        let Some(held) = slot else { return };
121        if let Some(holders) = NonZeroUsize::new(held.holders.get().saturating_sub(1)) {
122            held.holders = holders;
123            return;
124        }
125        if let Some(held) = slot.take() {
126            held.restore();
127        }
128    }
129
130    /// Put `previous` back, unless the signal no longer has the handler installed: a handler some
131    /// other code installed meanwhile is put back in turn. Swapping in one call, rather than
132    /// checking first, leaves the other code's handler replaced only for a moment, not for good.
133    fn restore(&self) {
134        // SAFETY: an all-zero `sigaction` is valid storage, only written to.
135        let mut replaced: libc::sigaction = unsafe { core::mem::zeroed() };
136        // SAFETY: `previous` is an action `sigaction` reported, so valid to reinstall.
137        let status =
138            unsafe { libc::sigaction(self.raw, &raw const self.previous, &raw mut replaced) };
139        if status == 0i32 && replaced.sa_sigaction != self.installed {
140            // SAFETY: `replaced` is the action `sigaction` reported.
141            unsafe {
142                libc::sigaction(self.raw, &raw const replaced, core::ptr::null_mut());
143            }
144        }
145    }
146
147    /// Install `action` for signal `raw`, held by one guard. `None` when it can't be.
148    fn take_over(raw: libc::c_int, action: &libc::sigaction) -> Option<Self> {
149        // SAFETY: an all-zero `sigaction` is valid storage, only written to.
150        let mut previous: libc::sigaction = unsafe { core::mem::zeroed() };
151        // SAFETY: both pointers come from live references for the length of the call.
152        let status = unsafe { libc::sigaction(raw, action, &raw mut previous) };
153
154        (status == 0i32).then_some(Self {
155            holders: NonZeroUsize::MIN,
156            installed: action.sa_sigaction,
157            previous,
158            raw,
159        })
160    }
161}

163/// A known signal.
164#[derive(Clone, Copy, Debug, PartialEq, Eq)]
165#[non_exhaustive]
166pub enum Signal {
167    /// `SIGHUP`: the terminal closed.
168    Hangup,
169    /// `SIGINT`: Ctrl-C.
170    Interrupt,
171    /// `SIGQUIT`: Ctrl-\\, or Ctrl+Break on Windows.
172    Quit,
173}
174
175impl Signal {
176    /// The action this signal has now.
177    #[cfg(unix)]
178    fn action(self) -> Option<libc::sigaction> {
179        // SAFETY: an all-zero `sigaction` is valid storage, only written to.
180        let mut current: libc::sigaction = unsafe { core::mem::zeroed() };
181        // SAFETY: a null new action only reads the current one into `current`.
182        let status = unsafe { libc::sigaction(self.to_raw(), core::ptr::null(), &raw mut current) };
183
184        (status == 0i32).then_some(current)
185    }
186
187    /// The action this signal had before the [`IgnoredSignals`] guards, or has now when none
188    /// holds it.
189    #[cfg(unix)]
190    fn action_before_guards(self) -> Option<libc::sigaction> {
191        #[cfg(feature = "interactive-editor")]
192        {
193            let before_guards = Handling::lock().slot(self).as_ref().map(|held| held.previous);
194            before_guards.or_else(|| self.action())
195        }
196        #[cfg(not(feature = "interactive-editor"))]
197        self.action()
198    }
199
200    /// Die of this signal.
201    ///
202    /// Normally, signals which interrupt processes can abort the process. This allows you to exit
203    /// mimicking as if that signal was fired to the app. An app that handles or ignores the signal
204    /// gets it under its own handling instead, and if that doesn't end the process, it exits with
205    /// the status that reports a death by the signal.
206    pub(crate) fn die(self) -> ! {
207        #[cfg(unix)]
208        if self.handled_by_current_process() {
209            Self::raise_unblocked(self.to_raw());
210        } else {
211            Self::raise_by_default(self.to_raw());
212        }
213
214        std::process::exit(self.exit_code())
215    }
216
217    /// The status that reports a death by this signal.
218    const fn exit_code(self) -> i32 {
219        #[cfg(windows)]
220        match self {
221            Self::Hangup | Self::Interrupt | Self::Quit => STATUS_CONTROL_C_EXIT,
222        }
223        // 128 plus the signal is how a shell reports a death by it: 130 for SIGINT.
224        #[cfg(unix)]
225        {
226            128i32.saturating_add(self.to_raw())
227        }
228        // Without signals or a shell convention for them, a plain failure.
229        #[cfg(not(any(unix, windows)))]
230        match self {
231            Self::Hangup | Self::Interrupt | Self::Quit => 1i32,
232        }
233    }
234
235    /// Flush stdout and stderr, then unblock signal number `signal`, so that raising it acts at
236    /// once rather than leaving it pending.
237    #[cfg(unix)]
238    fn flush_and_unblock(signal: libc::c_int) {
239        use std::io::Write as _;
240
241        drop(std::io::stdout().flush());
242        drop(std::io::stderr().flush());
243
244        // SAFETY: an all-zero `sigset_t` is valid storage, filled in by `sigemptyset`.
245        let mut set: libc::sigset_t = unsafe { core::mem::zeroed() };
246        // A blocked signal would stay pending instead of killing, so it is unblocked first.
247        // SAFETY: `set` is live for the call.
248        unsafe {
249            libc::sigemptyset(&raw mut set);
250        }
251        // SAFETY: as above; `signal` is a real signal.
252        unsafe {
253            libc::sigaddset(&raw mut set, signal);
254        }
255        // SAFETY: `set` is live, and the old mask is not asked for.
256        unsafe {
257            libc::pthread_sigmask(libc::SIG_UNBLOCK, &raw const set, core::ptr::null_mut());
258        }
259    }
260
261    /// The signal `raw` is, if it is one of these.
262    #[cfg(unix)]
263    #[cfg_attr(
264        not(any(test, feature = "interactive-editor")),
265        expect(dead_code, reason = "only a child's death names a signal by number")
266    )]
267    pub(crate) const fn from_raw(raw: libc::c_int) -> Option<Self> {
268        match raw {
269            libc::SIGHUP => Some(Self::Hangup),
270            libc::SIGINT => Some(Self::Interrupt),
271            libc::SIGQUIT => Some(Self::Quit),
272            _ => None,
273        }
274    }
275
276    /// Whether the current process handles or ignores this signal rather than taking its default
277    /// action, looking past the [`IgnoredSignals`] guards.
278    #[cfg(unix)]
279    fn handled_by_current_process(self) -> bool {
280        self.action_before_guards().is_some_and(|action| action.sa_sigaction != libc::SIG_DFL)
281    }
282
283    /// Ignore this signal, held by one guard.
284    #[cfg(all(unix, feature = "interactive-editor"))]
285    fn ignore(self) -> Option<Held> {
286        // SAFETY: an all-zero `sigaction` is valid: `SIG_DFL`, no flags, an empty mask.
287        let mut ignore: libc::sigaction = unsafe { core::mem::zeroed() };
288        ignore.sa_sigaction = libc::SIG_IGN;
289        Held::take_over(self.to_raw(), &ignore)
290    }
291
292    /// Check whether the current process ignores this signal, looking past the [`IgnoredSignals`]
293    /// guards: while one holds it, what counts is the action it had before the first of them.
294    #[cfg(all(unix, feature = "interactive-editor"))]
295    pub(crate) fn ignored_by_current_process(self) -> bool {
296        self.action_before_guards().is_some_and(|action| action.sa_sigaction == libc::SIG_IGN)
297    }
298
299    /// The signal `status`'s process died of, if it is one of these. On Windows, a process ended
300    /// by Ctrl+C or Ctrl+Break exits with `STATUS_CONTROL_C_EXIT`, read as Ctrl-C.
301    #[cfg(feature = "interactive-editor")]
302    pub(crate) fn killed(status: ExitStatus) -> Option<Self> {
303        #[cfg(unix)]
304        {
305            use std::os::unix::process::ExitStatusExt as _;
306
307            status.signal().and_then(Self::from_raw)
308        }
309        #[cfg(windows)]
310        {
311            (status.code() == Some(STATUS_CONTROL_C_EXIT)).then_some(Self::Interrupt)
312        }
313        #[cfg(not(any(unix, windows)))]
314        {
315            drop(status);
316            None
317        }
318    }
319
320    /// Raise signal number `signal` with its default action, which ends the process for the
321    /// signals this crate dies of.
322    #[cfg(unix)]
323    fn raise_by_default(signal: libc::c_int) {
324        // No guard may take the signal over between setting its default action and raising it.
325        #[cfg(feature = "interactive-editor")]
326        let _handling = Handling::lock();
327        Self::flush_and_unblock(signal);
328        // SAFETY: the default action is valid for every signal.
329        unsafe {
330            libc::signal(signal, libc::SIG_DFL);
331        }
332        // SAFETY: raising a real signal; with its default action, it ends the process.
333        unsafe {
334            libc::raise(signal);
335        }
336    }
337
338    /// Raise signal number `signal` with the action it has now.
339    #[cfg(unix)]
340    fn raise_unblocked(signal: libc::c_int) {
341        Self::flush_and_unblock(signal);
342        // SAFETY: raising a real signal; unblocked, its handler runs before `raise` returns.
343        unsafe {
344            libc::raise(signal);
345        }
346    }
347
348    /// The signal's number.
349    #[cfg(unix)]
350    pub(crate) const fn to_raw(self) -> libc::c_int {
351        match self {
352            Self::Hangup => libc::SIGHUP,
353            Self::Interrupt => libc::SIGINT,
354            Self::Quit => libc::SIGQUIT,
355        }
356    }
357}

359/// Signals ignored in this process while alive; the previous handling is restored on drop.
360#[cfg(all(unix, feature = "interactive-editor"))]
361pub(crate) struct IgnoredSignals {
362    /// The signals this guard holds in [`HANDLING`].
363    held: Vec<Signal>,
364}
365
366#[cfg(all(unix, feature = "interactive-editor"))]
367impl IgnoredSignals {
368    /// Ignore `signals`. Best effort: a signal that can't be ignored keeps its handling.
369    pub(crate) fn new(signals: &[Signal]) -> Self {
370        let mut handling = Handling::lock();
371        let held = signals.iter().copied().filter(|signal| handling.hold(*signal)).collect();
372        Self { held }
373    }
374}
375
376#[cfg(all(unix, feature = "interactive-editor"))]
377impl Drop for IgnoredSignals {
378    /// Restore each signal still ignored. One whose handling changed meanwhile, say by another
379    /// thread installing a handler, keeps the new handling.
380    fn drop(&mut self) {
381        let mut handling = Handling::lock();
382        for signal in &self.held {
383            handling.release(*signal);
384        }
385    }
386}

388/// `SIGTERM` passed on to an interactive child while alive, rather than ending this process at
389/// once and orphaning the child; the previous handling is restored on drop. Once done with the
390/// child, the caller dies of it with [`ForwardedTermination::die`].
391#[cfg(all(unix, feature = "interactive-editor"))]
392pub(crate) struct ForwardedTermination {
393    /// The handler is installed for this guard, so dropping lets go of it.
394    held: bool,
395    /// The slot in [`TERMINATION_RECIPIENTS`] naming the child, once it has one.
396    recipient: Option<&'static AtomicI32>,
397}
398
399#[cfg(all(unix, feature = "interactive-editor"))]
400impl ForwardedTermination {
401    /// Die of `SIGTERM`, as this process would have when it came.
402    pub(crate) fn die() -> ! {
403        Signal::raise_by_default(libc::SIGTERM);
404        std::process::exit(128i32.saturating_add(libc::SIGTERM))
405    }
406
407    /// The handler: note that `SIGTERM` came, and pass it on. Async-signal-safe: it only touches
408    /// lock-free atomics and calls `kill`.
409    extern "C" fn forward(_signal: libc::c_int) {
410        TERMINATION_RECEIVED.store(true, Ordering::SeqCst);
411        for recipient in &TERMINATION_RECIPIENTS {
412            let child = recipient.load(Ordering::SeqCst);
413            if child > 0i32 {
414                // SAFETY: `kill` is async-signal-safe; `child` is a child not yet reaped.
415                unsafe {
416                    libc::kill(child, libc::SIGTERM);
417                }
418            }
419        }
420    }
421
422    /// Pass `SIGTERM` on to process `child`, at once if it already came.
423    pub(crate) fn forward_to(&mut self, child: u32) {
424        let Ok(child) = libc::pid_t::try_from(child) else { return };
425        if !self.held {
426            return;
427        }
428        self.recipient = TERMINATION_RECIPIENTS.iter().find(|slot| {
429            slot.compare_exchange(0i32, child, Ordering::SeqCst, Ordering::SeqCst).is_ok()
430        });
431        if TERMINATION_RECEIVED.load(Ordering::SeqCst) {
432            // SAFETY: `child` is a child not yet waited for.
433            unsafe {
434                libc::kill(child, libc::SIGTERM);
435            }
436        }
437    }
438
439    /// Install [`ForwardedTermination::forward`] for `SIGTERM`, held by one guard. `None` when
440    /// it can't be, or `SIGTERM` doesn't have its default action: the app's own handling stays.
441    fn install() -> Option<Held> {
442        // SAFETY: an all-zero `sigaction` is valid storage, only written to.
443        let mut current: libc::sigaction = unsafe { core::mem::zeroed() };
444        // SAFETY: a null new action only reads the current one into `current`.
445        let status = unsafe { libc::sigaction(libc::SIGTERM, core::ptr::null(), &raw mut current) };
446        // Checked before installing, so that a `SIGTERM` never reaches `forward` in place of the
447        // app's own handler.
448        if status != 0i32 || current.sa_sigaction != libc::SIG_DFL {
449            return None;
450        }
451        #[expect(clippy::as_conversions, reason = "`sigaction` takes a handler as an address")]
452        let forward = (Self::forward as extern "C" fn(libc::c_int) as *const ()).addr();
453        // SAFETY: an all-zero `sigaction` is valid: `SIG_DFL`, no flags, an empty mask.
454        let mut action: libc::sigaction = unsafe { core::mem::zeroed() };
455        action.sa_sigaction = forward;
456        // Other threads' system calls carry on rather than fail with `EINTR`.
457        action.sa_flags = libc::SA_RESTART;
458        TERMINATION_RECEIVED.store(false, Ordering::SeqCst);
459        let held = Held::take_over(libc::SIGTERM, &action)?;
460        // Code that doesn't take the lock may have installed a handler since.
461        if held.previous.sa_sigaction == libc::SIG_DFL {
462            Some(held)
463        } else {
464            held.restore();
465            None
466        }
467    }
468
469    /// Pass `SIGTERM` on, to the child [`ForwardedTermination::forward_to`] names. Best effort:
470    /// an app that handles or ignores `SIGTERM` itself keeps its handling.
471    pub(crate) fn new() -> Self {
472        let held = Handling::lock().hold_termination();
473        Self { held, recipient: None }
474    }
475
476    /// Stop passing `SIGTERM` on; whether it came while this guard did. Read once let go of, so
477    /// that one coming later takes the app's handling rather than going unnoticed.
478    pub(crate) fn release(&mut self) -> bool {
479        if let Some(recipient) = self.recipient.take() {
480            recipient.store(0i32, Ordering::SeqCst);
481        }
482        if !core::mem::take(&mut self.held) {
483            return false;
484        }
485        Handling::lock().release_termination();
486        TERMINATION_RECEIVED.load(Ordering::SeqCst)
487    }
488
489    /// Wait for process `child` to exit, without reaping it, then stop passing `SIGTERM` on to
490    /// it: until reaped, its PID can't name another process.
491    pub(crate) fn wait_for_exit(&mut self, child: u32) {
492        let Some(recipient) = self.recipient.take() else { return };
493        loop {
494            // SAFETY: an all-zero `siginfo_t` is valid storage, only written to.
495            let mut info: libc::siginfo_t = unsafe { core::mem::zeroed() };
496            // SAFETY: `info` is live for the call; `WNOWAIT` leaves the child unreaped.
497            let status = unsafe {
498                libc::waitid(libc::P_PID, child, &raw mut info, libc::WEXITED | libc::WNOWAIT)
499            };
500            if status == 0i32 || io::Error::last_os_error().kind() != io::ErrorKind::Interrupted {
501                break;
502            }
503        }
504        recipient.store(0i32, Ordering::SeqCst);
505    }
506}

554/// Ctrl+C and Ctrl+Break ignored in this process while alive; see the Unix version.
555///
556/// A handler of our own, removed on drop, rather than the process-wide ignore flag: clearing the
557/// flag would also clear it for an app that had set it. Handlers aren't inherited, so a child
558/// keeps the default handling. A console handler sees both keys or neither, so asking for
559/// [`Signal::Interrupt`] or [`Signal::Quit`] ignores both; there is no hang-up to ignore.
560#[cfg(all(windows, feature = "interactive-editor"))]
561pub(crate) struct IgnoredSignals {
562    /// The handler was added, so dropping removes it.
563    registered: bool,
564}
565
566#[cfg(all(windows, feature = "interactive-editor"))]
567impl IgnoredSignals {
568    /// Swallow Ctrl+C and Ctrl+Break; leave closing, logoff and shutdown to the next handler.
569    const extern "system" fn ignore(ctrl_type: u32) -> BOOL {
570        if matches!(ctrl_type, CTRL_C_EVENT | CTRL_BREAK_EVENT) { TRUE } else { FALSE }
571    }
572
573    pub(crate) fn new(signals: &[Signal]) -> Self {
574        let wanted =
575            signals.iter().any(|signal| matches!(signal, Signal::Interrupt | Signal::Quit));
576        // SAFETY: `ignore` is a valid handler for the whole program, as a plain function.
577        let registered =
578            wanted && unsafe { SetConsoleCtrlHandler(Some(Self::ignore), TRUE) } != FALSE;
579        Self { registered }
580    }
581}
582
583#[cfg(all(windows, feature = "interactive-editor"))]
584impl Drop for IgnoredSignals {
585    fn drop(&mut self) {
586        if self.registered {
587            // SAFETY: removes only the handler `new` added.
588            unsafe {
589                SetConsoleCtrlHandler(Some(Self::ignore), FALSE);
590            }
591        }
592    }
593}

769        unsafe {
