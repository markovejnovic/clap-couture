1//! The `--help` couture installs: clap's own text, with the subcommands listed by category.
2
3use clap::{Arg, Command, builder::StyledStr};
4
5use super::CommandRenderExt as _;
6use crate::{CommandCategoryMap, Messages};
7
8/// What couture shows below the usage, at `{after-help}`: the categorized subcommands, clap's
9/// arg sections, then the command's own after-help.
10///
11/// clap can't regroup its subcommand listing, so its `{all-args}` is rendered here from a copy of
12/// the command with the subcommands hidden, once for `-h` and once for `--help`. Rendering it
13/// through clap keeps positionals, `help_heading`s and hidden args exactly as clap shows them.
14pub(crate) struct HelpBody {
15    /// For `--help`, when clap shows more for it than for `-h`.
16    long: Option<String>,
17    short: String,
18}
19
20/// Rewords clap's English `Arguments:` and `Options:` headings with a command's [`Messages`].
21struct Headings<'cmd> {
22    cmd: &'cmd Command,
23    messages: &'cmd Messages,
24}
25
26/// The help templates couture tells apart from one the app set.
27pub(crate) trait HelpTemplates {
28    /// clap's templates for a command without one of its own: `DEFAULT_TEMPLATE`, and
29    /// `DEFAULT_NO_ARGS_TEMPLATE` for one with nothing to list (`clap_builder` 4.6.7,
30    /// `src/output/help_template.rs`).
31    const CLAP_DEFAULTS: [&str; 2] = [
32        "{before-help}{about-with-newline}\n{usage-heading} {usage}\n\n{all-args}{after-help}",
33        "{before-help}{about-with-newline}\n{usage-heading} {usage}{after-help}",
34    ];
35
36    /// couture's template: clap's, with the usage under a heading of its own, and everything
37    /// below it at `{after-help}`.
38    fn couture_template(&self, messages: &Messages) -> String;
39
40    /// Check whether `--help` shows more than `-h`, as clap decides it
41    /// (`Command::long_help_exists_` in `clap_builder` 4.6.7): otherwise clap shows `-h`'s help
42    /// for both.
43    fn has_long_help(&self) -> bool;
44
45    /// Check whether the app set its own help template. clap has no getter for it, so the help
46    /// is rendered under clap's default templates and compared.
47    fn has_own_template(&self) -> bool;
48}
49
50impl HelpBody {
51    /// `cmd` showing this body at `{after-help}`, under couture's template unless the app set its
52    /// own.
53    pub(crate) fn install(self, cmd: Command, messages: &Messages) -> Command {
54        let cmd = if cmd.has_own_template() {
55            cmd
56        } else {
57            let template = cmd.couture_template(messages);
58            cmd.help_template(template)
59        };
60        let cmd = cmd.with_rendered_about();
61        // An empty after-help would still print its blank line.
62        let cmd = if self.short.is_empty() { cmd } else { cmd.after_help(self.short) };
63        match self.long {
64            Some(long) => cmd.after_long_help(long),
65            None => cmd,
66        }
67    }
68
69    /// The body for `cmd`, its subcommands filed by `categories`, its headings worded by
70    /// `messages`.
71    pub(crate) fn new(cmd: &Command, categories: CommandCategoryMap, messages: &Messages) -> Self {
72        let mut menu = StyledStr::new();
73        // `StyledStr`'s `fmt::Write` is infallible.
74        cmd.write_command_sections(&mut menu, categories).ok();
75        let menu = menu.ansi().to_string();
76
77        let mut args_only = cmd
78            .clone()
79            .disable_help_subcommand(true)
80            .mut_subcommands(|subcommand| subcommand.hide(true))
81            .help_template("{all-args}")
82            .with_rendered_arg_help();
83        let headings = Headings { cmd, messages };
84        let short_args = headings.reword(&args_only.clone().render_help());
85        let long_args = headings.reword(&args_only.render_long_help());
86
87        let after_help = cmd.get_after_help().map(|text| text.ansi().to_string());
88        let after_long_help = cmd.get_after_long_help().map(|text| text.ansi().to_string());
89        let short = Self::paragraphs([&menu, &short_args, after_help.as_deref().unwrap_or("")]);
90        let long = cmd.has_long_help().then(|| {
91            let after = after_long_help.or(after_help);
92            Self::paragraphs([&menu, &long_args, after.as_deref().unwrap_or("")])
93        });
94        Self { long, short }
95    }
96
97    /// `parts` that aren't empty, a blank line apart.
98    fn paragraphs<const N: usize>(parts: [&str; N]) -> String {
99        parts.into_iter().filter(|part| !part.is_empty()).collect::<Vec<_>>().join("\n\n")
100    }
101}
102
103impl Headings<'_> {
104    /// `rendered`, clap's `{all-args}`, with its default section headings reworded, its trailing
105    /// newline dropped.
106    fn reword(&self, rendered: &StyledStr) -> String {
107        let header = self.cmd.get_styles().get_header();
108        let styled = |heading: &str| format!("{header}{heading}{header:#}\n");
109        let mut text = rendered.ansi().to_string().trim_end().to_owned();
110        let defaults = Messages::DEFAULT;
111        for (default, worded) in [
112            (defaults.arguments.get(), self.messages.arguments.get()),
113            (defaults.options.get(), self.messages.options.get()),
114        ] {
115            if default == worded {
116                continue;
117            }
118            let (clap_heading, couture_heading) = (styled(&default), styled(&worded));
119            // A section starts the text or follows a blank line; clap writes the arguments first.
120            let at = if text.starts_with(&clap_heading) {
121                Some(0)
122            } else {
123                text.find(&format!("\n\n{clap_heading}")).map(|blank| blank.saturating_add(2))
124            };
125            if let Some(at) = at {
126                text.replace_range(at..at.saturating_add(clap_heading.len()), &couture_heading);
127            }
128        }
129        text
130    }
131}

133impl HelpTemplates for Command {
134    fn couture_template(&self, messages: &Messages) -> String {
135        let usage = self.get_styles().get_usage();
136        let usage_heading = messages.usage.get();
137        // clap's `{usage-heading}` is always English, so the heading is written out.
138        format!(
139            "{{before-help}}{{about-with-newline}}\n{usage}{usage_heading}{usage:#}\n  \
140             {{usage}}{{after-help}}"
141        )
142    }
143
144    fn has_long_help(&self) -> bool {
145        let shows_more_long = |arg: &Arg| {
146            !arg.is_hide_set()
147                && (arg.get_long_help().is_some()
148                    || arg.is_hide_long_help_set()
149                    || arg.is_hide_short_help_set()
150                    || !arg.is_hide_possible_values_set()
151                        && arg
152                            .get_possible_values()
153                            .iter()
154                            .any(|value| !value.is_hide_set() && value.get_help().is_some()))
155        };
156        self.get_long_about().is_some()
157            || self.get_before_long_help().is_some()
158            || self.get_after_long_help().is_some()
159            || self.get_arguments().any(shows_more_long)
160    }
161
162    fn has_own_template(&self) -> bool {
163        let rendered = self.clone().render_help().ansi().to_string();
164        Self::CLAP_DEFAULTS.iter().all(|template| {
165            self.clone().help_template(*template).render_help().ansi().to_string() != rendered
166        })
167    }
