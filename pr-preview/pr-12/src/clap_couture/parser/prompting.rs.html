1//! Parse, ask for the prompted args that got no value, then parse again with the answers.
2//!
3//! Every answer goes back into the command as its arg's `default_values`, so clap converts and
4//! validates it exactly like a typed value. Each answer is checked with a trial parse as soon as
5//! it is given: the previous trial passed, so a failure can only come from the new answer, which
6//! is then asked again with clap's message. For an arg asked one value at a time, each value is
7//! an answer of its own, checked on top of the values before it.
8//!
9//! Confirmations come last, so their questions can name the answers. A declined one fails the
10//! parse; without a terminal to ask on, one left unanswered fails it as a missing required arg.
11

29/// A prompted arg's accepted answer.
30struct Answer<'prompted_arg> {
31    prompted_arg: &'prompted_arg PromptedArg,
32    /// One value, or any number for an arg that takes many.
33    values: Vec<String>,
34}
35
36/// One parse: the command, its argv, its prompted args and who to ask.
37struct Session<'run> {
38    argv: &'run [OsString],
39    backend: &'run dyn Backend,
40    cmd: &'run Command,
41    messages: &'run Messages,
42    prompted_args: &'run [PromptedArg],
43}
44
45impl Session<'_> {
46    /// Ask for `prompted_arg` until clap accepts the answer; `None` when an optional single-value
47    /// one is left empty. `current` is the arg's default, split into its values.
48    fn answer(
49        &self,
50        prompted_arg: &PromptedArg,
51        current: &[String],
52        answers: &[Answer<'_>],
53    ) -> Result<Option<Vec<String>>, PromptError> {
54        let Some(arg) = arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
55        else {
56            return Ok(None);
57        };
58        let question = self.question(prompted_arg);
59        let options: Vec<PossibleValue> =
60            arg.get_possible_values().into_iter().filter(|value| !value.is_hide_set()).collect();
61        if !takes_many(arg) {
62            return self.until_accepted(answers, prompted_arg, || {
63                Ok(self.ask(arg, &question, &options, current.first())?.map(|value| vec![value]))
64            });
65        }
66        if options.is_empty() {
67            return self.each_value(arg, &question, prompted_arg, current, answers).map(Some);
68        }
69        self.until_accepted(answers, prompted_arg, || {
70            self.backend
71                .multi_select(&MultiSelectPrompt {
72                    defaults: current,
73                    optional: !arg.is_required_set(),
74                    options: &options,
75                    question: &question,
76                    selection_required: &self.messages.selection_required.get(),
77                    styles: CommandStyles(self.cmd.get_styles()),
78                })
79                .map(Some)
80        })
81    }
82
83    fn ask(
84        &self,
85        arg: &Arg,
86        question: &str,
87        options: &[PossibleValue],
88        current: Option<&String>,
89    ) -> Result<Option<String>, PromptError> {
90        let styles = CommandStyles(self.cmd.get_styles());
91        if let Some(negate) = flag_negation(arg.get_action()) {
92            let set = self.backend.confirm(&ConfirmPrompt { default: false, question, styles })?;
93            return Ok(Some((set != negate).to_string()));
94        }
95        if options.is_empty() {
96            return self.backend.text(&TextPrompt {
97                default: current.map(String::as_str),
98                hint: None,
99                initial: None,
100                optional: !arg.is_required_set(),
101                question,
102                styles,
103            });
104        }
105        self.backend.select(&SelectPrompt {
106            default: current.map(String::as_str),
107            none_option: &self.messages.none.get(),
108            optional: !arg.is_required_set(),
109            options,
110            question,
111            styles,
112        })
113    }
114
115    /// Ask whether to go ahead, with each `{id}` in the question filled in from the args as
116    /// passed and answered so far.
117    fn confirm(
118        &self,
119        prompted_arg: &PromptedArg,
120        answers: &[Answer<'_>],
121    ) -> Result<bool, PromptError> {
122        let question = self.question(prompted_arg);
123        let answered = inject(
124            relax(self.cmd.clone(), self.prompted_args),
125            answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice())),
126        )
127        .try_get_matches_from(self.argv);
128        let question = match answered
129            .as_ref()
130            .ok()
131            .and_then(|answered| matches_at(answered, &prompted_arg.subcommand_path))
132        {
133            Some(level) => interpolate(&question, level),
134            None => question,
135        };
136        self.backend.confirm(&ConfirmPrompt {
137            default: false,
138            question: &question,
139            styles: CommandStyles(self.cmd.get_styles()),
140        })
141    }
142
143    /// Ask for one value at a time until an empty answer, each starting out as the default's
144    /// value at its position. Only a required arg's first value may not be empty.
145    ///
146    /// The default goes in as initial text rather than as the answer Enter accepts, so clearing it
147    /// still finishes the list.
148    fn each_value(
149        &self,
150        arg: &Arg,
151        question: &str,
152        prompted_arg: &PromptedArg,
153        current: &[String],
154        answers: &[Answer<'_>],
155    ) -> Result<Vec<String>, PromptError> {
156        let finish_hint = self.messages.finish_hint.get();
157        let mut values = Vec::new();
158        loop {
159            let optional = !values.is_empty() || !arg.is_required_set();
160            let Some(value) = self.backend.text(&TextPrompt {
161                default: None,
162                hint: optional.then_some(&*finish_hint),
163                initial: current.get(values.len()).map(String::as_str),
164                optional,
165                question,
166                styles: CommandStyles(self.cmd.get_styles()),
167            })?
168            else {
169                return Ok(values);
170            };
171            values.push(value);
172            if let Err(err) = self.trial(answers, prompted_arg, &values) {
173                values.pop();
174                self.backend.report(&summary(&err), CommandStyles(self.cmd.get_styles()))?;
175            }
176        }
177    }
178
179    /// The `= ...` override, else the arg's help text, else its id.
180    fn question(&self, prompted_arg: &PromptedArg) -> String {
181        prompted_arg.attr.question.as_ref().map_or_else(
182            || {
183                arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
184                    .and_then(Arg::get_help)
185                    .map_or_else(|| prompted_arg.attr.arg_id.to_owned(), ToString::to_string)
186            },
187            |question| question.get().into_owned(),
188        )
189    }
190
191    /// Parse with `candidate` as `prompted_arg`'s answer on top of the accepted ones.
192    fn trial(
193        &self,
194        answers: &[Answer<'_>],
195        prompted_arg: &PromptedArg,
196        candidate: &[String],
197    ) -> Result<(), clap::Error> {
198        let accepted = answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice()));
199        let cmd = inject(
200            relax(self.cmd.clone(), self.prompted_args),
201            accepted.chain(iter::once((prompted_arg, candidate))),
202        );
203        cmd.try_get_matches_from(self.argv).map(drop)
204    }
205
206    /// Call `ask` until clap accepts its answer, showing clap's complaint before each retry.
207    fn until_accepted(
208        &self,
209        answers: &[Answer<'_>],
210        prompted_arg: &PromptedArg,
211        mut ask: impl FnMut() -> Result<Option<Vec<String>>, PromptError>,
212    ) -> Result<Option<Vec<String>>, PromptError> {
213        loop {
214            let Some(values) = ask()? else { return Ok(None) };
215            match self.trial(answers, prompted_arg, &values) {
216                Ok(()) => return Ok(Some(values)),
217                Err(err) => {
218                    self.backend.report(&summary(&err), CommandStyles(self.cmd.get_styles()))?;
219                }
220            }
221        }
222    }
223}

225/// Parse `argv` against `cmd`, asking `backend` for the prompted args in `prompted_arg_tree` that
226/// got no value.
227pub(super) fn parse<T>(
228    cmd: Command,
229    prompted_arg_tree: &'static PromptedArgTree,
230    messages: &Messages,
231    argv: &[OsString],
232    backend: &dyn Backend,
233) -> Result<T, Failure>
234where
235    T: FromArgMatches,
236{
237    let prompted_args = prompted_arg_tree.prompted_args();
238    if prompted_args.is_empty() || !backend.is_available() {
239        return parse_unprompted(cmd, &prompted_args, argv);
240    }
241    // Help, usage and errors come from the command the user sees, not the relaxed one.
242    let Ok(matches) = relax(cmd.clone(), &prompted_args).try_get_matches_from(argv) else {
243        return super::parse(cmd, argv);
244    };
245
246    let session = Session { argv, backend, cmd: &cmd, messages, prompted_args: &prompted_args };
247    let mut answers = Vec::new();
248    let mut unanswered = pending(&cmd, &matches, &prompted_args);
249    // Stable, so values and confirmations each keep declaration order.
250    unanswered
251        .sort_by_key(|(prompted_arg, _)| prompted_arg.attr.purpose == PromptPurpose::Confirmation);
252    for (prompted_arg, current) in unanswered {
253        if blocked(&cmd, &matches, prompted_arg, &answers) {
254            continue;
255        }
256        let answer = match prompted_arg.attr.purpose {
257            PromptPurpose::Confirmation => match session.confirm(prompted_arg, &answers) {
258                Ok(false) => return Err(Failure::Declined),
259                confirmed => confirmed.map(|_| Some(Vec::from(["true".to_owned()]))),
260            },
261            PromptPurpose::Value => session.answer(prompted_arg, &current, &answers),
262        };
263        match answer {
264            Ok(Some(values)) => answers.push(Answer { prompted_arg, values }),
265            Ok(None) => {}
266            Err(PromptError::Cancelled) => return Err(Failure::Cancelled),
267            Err(PromptError::Io(err)) => {
268                return Err(Failure::Clap(clap::Error::raw(ErrorKind::Io, err)));
269            }
270            Err(PromptError::NotATerminal) => return parse_unprompted(cmd, &prompted_args, argv),
271        }
272    }
273    let cmd_with_answers =
274        inject(cmd, answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice())));
275    super::parse(cmd_with_answers, argv)
276}

278/// Parse as clap would, without asking. A confirmation left unanswered fails the parse with
279/// clap's error for a missing required arg, so only passing its flag gets past it.
280fn parse_unprompted<T>(
281    mut cmd: Command,
282    prompted_args: &[PromptedArg],
283    argv: &[OsString],
284) -> Result<T, Failure>
285where
286    T: FromArgMatches,
287{
288    let confirmations: Vec<_> = prompted_args
289        .iter()
290        .filter(|prompted_arg| prompted_arg.attr.purpose == PromptPurpose::Confirmation)
291        .collect();
292    if confirmations.is_empty() {
293        return super::parse(cmd, argv);
294    }
295    // Kept unbuilt: requiring an arg on the built command makes clap show help instead.
296    let declared = cmd.clone();
297    let mut matches = cmd.try_get_matches_from_mut(argv).map_err(Failure::Clap)?;
298    let unconfirmed = confirmations.iter().any(|confirmation| {
299        let id = confirmation.attr.arg_id;
300        let passed = matches_at(&matches, &confirmation.subcommand_path)
301            .is_none_or(|level| explicit(level, &Id::from(id)));
302        !passed && !blocked(&cmd, &matches, confirmation, &[])
303    });
304    if unconfirmed {
305        // Parsed again only to fail: clap words the error, and the help stays as declared.
306        let required = confirmations.iter().fold(declared, |cmd, confirmation| {
307            with_arg_at(cmd, &confirmation.subcommand_path, confirmation.attr.arg_id, |arg| {
308                arg.required(true)
309            })
310        });
311        return super::parse(required, argv);
312    }
313    T::from_arg_matches_mut(&mut matches).map_err(|err| Failure::Clap(err.format(&mut cmd)))
314}

316fn arg_at<'cmd>(cmd: &'cmd Command, path: &[&str], id: &str) -> Option<&'cmd Arg> {
317    command_at(cmd, path)?.get_arguments().find(|arg| arg.get_id() == id)

320/// Check whether `prompted_arg` conflicts with an arg the user passed or an earlier answer.
321///
322/// Answers go in as default values, which clap never checks for conflicts, so asking here
323/// would let the parse land in a state clap rejects when typed.
324fn blocked(
325    cmd: &Command,
326    matches: &ArgMatches,
327    prompted_arg: &PromptedArg,
328    answers: &[Answer<'_>],
329) -> bool {
330    let (Some(level), Some(level_matches)) = (
331        command_at(cmd, &prompted_arg.subcommand_path),
332        matches_at(matches, &prompted_arg.subcommand_path),
333    ) else {
334        return false;
335    };
336    let Some(arg) = level.get_arguments().find(|arg| arg.get_id() == prompted_arg.attr.arg_id)
337    else {
338        return false;
339    };
340    let answered = |id: &Id| {
341        answers.iter().any(|answer| {
342            answer.prompted_arg.subcommand_path == prompted_arg.subcommand_path
343                && answer.prompted_arg.attr.arg_id == id
344        })
345    };
346    let conflicts = level.get_arg_conflicts_with(arg);
347    level
348        .get_arguments()
349        .filter(|other| other.get_id() != arg.get_id())
350        .filter(|other| answered(other.get_id()) || explicit(level_matches, other.get_id()))
351        .any(|other| {
352            other.is_exclusive_set()
353                || conflicts.iter().any(|conflict| conflict.get_id() == other.get_id())
354                || level.get_arg_conflicts_with(other).iter().any(|c| c.get_id() == arg.get_id())
355                || group_mates(level, arg).any(|mate| mate == other.get_id())
356        })
357}

359/// `arg`'s declared default values, split at its value delimiter.
360/// Without a delimiter, each default is one value. Defaults are set from strings, so any that
361/// is not UTF-8 is skipped.
362fn default_values(arg: &Arg) -> impl Iterator<Item = &str> {
363    let delimiter = arg.get_value_delimiter();
364    arg.get_default_values()
365        .iter()
366        .filter_map(|value| value.to_str())
367        .flat_map(move |value| value.split(move |char| Some(char) == delimiter))
368}

370fn command_at<'cmd>(cmd: &'cmd Command, path: &[&str]) -> Option<&'cmd Command> {
371    path.iter().try_fold(cmd, |level, name| level.find_subcommand(name))
372}

374/// Check whether the user passed `id`, on the command line or through its env var.
375fn explicit(matches: &ArgMatches, id: &Id) -> bool {
376    matches!(
377        matches.value_source(id.as_str()),
378        Some(ValueSource::CommandLine | ValueSource::EnvVariable)
379    )
380}

382/// The args sharing an exclusive `ArgGroup` with `arg`.
383fn group_mates<'cmd>(level: &'cmd Command, arg: &Arg) -> impl Iterator<Item = &'cmd Id> {
384    level
385        .get_groups()
386        .filter(|group| {
387            !ArgGroup::clone(group).is_multiple() && group.get_args().any(|id| id == arg.get_id())
388        })
389        .flat_map(ArgGroup::get_args)
390}

392/// For a flag, whether a "yes" stores `false`; `None` for an arg that takes a value.
393#[expect(
394    clippy::wildcard_enum_match_arm,
395    reason = "`ArgAction` is #[non_exhaustive]; every action but the two flags takes a value"
396)]
397const fn flag_negation(action: &ArgAction) -> Option<bool> {
398    match action {
399        ArgAction::SetTrue => Some(false),
400        ArgAction::SetFalse => Some(true),
401        _ => None,
402    }
403}

405/// Set each answer as its arg's default values. No values leave the arg with no default, so
406/// an emptied list drops the declared default.
407fn inject<'prompted_arg>(
408    cmd: Command,
409    answers: impl IntoIterator<Item = (&'prompted_arg PromptedArg, &'prompted_arg [String])>,
410) -> Command {
411    answers.into_iter().fold(cmd, |cmd, (prompted_arg, values)| {
412        with_arg_at(cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id, |arg| {
413            arg.required(false).default_values(values.iter().cloned())
414        })
415    })
416}

418/// `question` with each `{id}` replaced by arg `id`'s values in `matches`, joined with `, `, and
419/// `{{` and `}}` by single braces. A `{id}` naming no arg stays as written.
420fn interpolate(question: &str, matches: &ArgMatches) -> String {
421    let mut interpolated = String::with_capacity(question.len());
422    let mut rest = question;
423    while let Some(brace) = rest.find(['{', '}']) {
424        let (text, tail) = rest.split_at(brace);
425        interpolated.push_str(text);
426        let escaped = tail.strip_prefix("{{").map(|after| ('{', after));
427        if let Some((brace, after)) =
428            escaped.or_else(|| tail.strip_prefix("}}").map(|after| ('}', after)))
429        {
430            interpolated.push(brace);
431            rest = after;
432            continue;
433        }
434        let placeholder = tail.strip_prefix('{').and_then(|inner| inner.split_once('}'));
435        if let Some((values, after)) = placeholder
436            .and_then(|(id, after)| matches.try_get_raw(id).ok().map(|values| (values, after)))
437        {
438            let values: Vec<_> =
439                values.into_iter().flatten().map(|v| v.to_string_lossy()).collect();
440            interpolated.push_str(&values.join(", "));
441            rest = after;
442            continue;
443        }
444        // A lone brace, or a `{id}` naming no arg: keep the brace as written.
445        let (brace, after) = tail.split_at(1);
446        interpolated.push_str(brace);
447        rest = after;
448    }
449    interpolated.push_str(rest);
450    interpolated
451}

453fn matches_at<'matches>(
454    matches: &'matches ArgMatches,
455    path: &[&str],
456) -> Option<&'matches ArgMatches> {
457    path.iter().try_fold(matches, |level, name| {
458        level.subcommand().filter(|(taken, _)| taken == name).map(|(_, sub)| sub)
459    })
460}

462/// The prompted args on the path the user took that got no value or only their default, with
463/// the default's values, split at the arg's value delimiter as clap splits them. One whose
464/// `default_value_if` fired is left to the arg that fired it: the conditional default beats an
465/// injected answer, so asking would drop the answer.
466fn pending<'prompted_arg>(
467    cmd: &Command,
468    matches: &ArgMatches,
469    prompted_args: &'prompted_arg [PromptedArg],
470) -> Vec<(&'prompted_arg PromptedArg, Vec<String>)> {
471    // Built, so a flag's action has filled in its implicit default (`false` for `SetTrue`).
472    let mut built = cmd.clone();
473    built.build();
474    prompted_args
475        .iter()
476        .filter_map(|prompted_arg| {
477            let arg = arg_at(&built, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)?;
478            let level = matches_at(matches, &prompted_arg.subcommand_path)?;
479            let id = prompted_arg.attr.arg_id;
480            match level.value_source(id) {
481                None if default_values(arg).next().is_none() => Some((prompted_arg, Vec::new())),
482                Some(ValueSource::DefaultValue) => {
483                    let current = raw_values(level, id);
484                    current
485                        .iter()
486                        .map(String::as_str)
487                        .eq(default_values(arg))
488                        .then_some((prompted_arg, current))
489                }
490                None | Some(_) => None,
491            }
492        })
493        .collect()
494}
495
496fn raw_values(matches: &ArgMatches, id: &str) -> Vec<String> {
497    matches
498        .try_get_raw(id)
499        .ok()
500        .flatten()
501        .into_iter()
502        .flatten()
503        .map(|value| value.to_string_lossy().into_owned())
504        .collect()
505}

507/// Check whether `arg` takes many values, as a `Vec<T>` field does.
508fn takes_many(arg: &Arg) -> bool {
509    matches!(arg.get_action(), ArgAction::Append)
510        || arg.get_num_args().is_some_and(|range| range.max_values() > 1)
511}

513/// Make every prompted arg optional, so a missing one does not stop the first parse.
514fn relax(cmd: Command, prompted_args: &[PromptedArg]) -> Command {
515    prompted_args.iter().fold(cmd, |cmd, prompted_arg| {
516        with_arg_at(cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id, |arg| {
517            arg.required(false)
518        })
519    })
520}

522/// The first line of clap's message, without its `error: ` prefix.
523fn summary(err: &clap::Error) -> String {
524    let rendered = err.render().to_string();
525    let line = rendered.lines().next().unwrap_or_default();
526    line.strip_prefix("error: ").unwrap_or(line).to_owned()
527}

529/// Apply `edit` to the arg `id` of the command at `path`; unchanged when either is missing.
530fn with_arg_at<F>(cmd: Command, path: &[&str], id: &str, edit: F) -> Command
531where
532    F: FnOnce(Arg) -> Arg,
533{
534    match path.split_first() {
535        None if cmd.get_arguments().any(|arg| arg.get_id() == id) => cmd.mut_arg(id, edit),
536        Some((name, rest)) if cmd.find_subcommand(name).is_some() => {
537            cmd.mut_subcommand(name, |sub| with_arg_at(sub, rest, id, edit))
538        }
539        None | Some(_) => cmd,
540    }
