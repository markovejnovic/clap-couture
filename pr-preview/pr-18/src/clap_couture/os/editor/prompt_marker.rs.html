2
3use crate::strings;

5/// What every header line starts with. Markdown headings take `# ` and shell comments `# `, so
6/// a user's own line almost never starts with `#|`.
7const HEADER: &str = "#|";
8
9#[derive(Debug, Clone, PartialEq, Eq)]
10pub(crate) struct PromptMarker(String);
11
12impl PromptMarker {
13    pub(crate) fn new(marker: impl Into<String>) -> Self {
14        Self(marker.into())
15    }
16
17    /// The answer in `file`: the header lines and the blank lines around the rest dropped. `None`
18    /// when nothing is left.
19    pub(crate) fn strip(&self, file: &str) -> Option<String> {
20        let is_blank = |line: &&str| line.trim().is_empty();
21        let mut kept: Vec<&str> = file
22            .lines()
23            .filter(|line| !line.starts_with(self.0.as_str()))
24            .skip_while(is_blank)
25            .collect();
26        while kept.last().is_some_and(is_blank) {
27            kept.pop();
28        }
29        (!kept.is_empty()).then(|| kept.join("\n"))
30    }
31
32    /// The file the editor opens: the answer so far, a blank line, then the header.
33    pub(crate) fn write(
34        &self,
35        out: &mut impl io::Write,
36        contents: &str,
37        prompt: Option<&str>,
38    ) -> io::Result<()> {
39        if !contents.is_empty() {
40            writeln!(out, "{contents}")?;
41        }
42        if let Some(prompt) = prompt {
43            writeln!(out)?;
44            // The marker, then a space before the text.
45            strings::prefix_lines_into(out, &format!("{} ", self.0), [prompt])?;
46        }
47        Ok(())
48    }
49}
50
51impl Default for PromptMarker {
52    fn default() -> Self {
53        Self::new(HEADER)
54    }
55}

57#[cfg(test)]
58mod tests {
59    use rstest::rstest;
60
61    use super::*;
62
63    const PROMPT: &str = "Release notes\n\nLeave the file empty to cancel.";
64
65    fn rendered(contents: &str, prompt: Option<&str>) -> String {
66        let mut file = Vec::new();
67        let written = PromptMarker::default().write(&mut file, contents, prompt);
68        assert!(written.is_ok(), "writing to a `Vec` can't fail");
69        String::from_utf8(file).unwrap_or_default()
70    }
71
72    #[test]
73    fn render_puts_the_answer_before_the_prompt() {
74        let file = rendered("Fixed it.", Some(PROMPT));
75        assert_eq!(file, "Fixed it.\n\n#| Release notes\n#|\n#| Leave the file empty to cancel.\n");
76    }
77
78    #[test]
79    fn render_without_a_prompt_is_the_contents() {
80        assert_eq!(rendered("Fixed it.", None), "Fixed it.\n");
81    }
82
83    #[test]
84    fn strip_round_trips_a_rendered_answer() {
85        let answer = "# Heading\n\n  indented: yes  \n\nlast";
86        assert_eq!(
87            PromptMarker::default().strip(&rendered(answer, Some(PROMPT))).as_deref(),
88            Some(answer)
89        );
90    }
91
92    #[rstest]
93    #[case::only_the_header(&rendered("", Some(PROMPT)), None)]
94    #[case::whitespace(" \n\t\n", None)]
95    #[case::blank_lines_around("\n\nhello\n\n", Some("hello"))]
96    #[case::users_hash_lines_stay("# not a header\nbody\n", Some("# not a header\nbody"))]
97    #[case::edited_header_line_goes("body\n#| I changed this\n", Some("body"))]
98    #[case::markdown_and_shell_comments_stay(
99        "## Notes\n#!/bin/sh\n#|x\n",
100        Some("## Notes\n#!/bin/sh")
101    )]
102    fn strip_drops_only_header_and_surrounding_blanks(
103        #[case] file: &str,
104        #[case] expected: Option<&str>,
105    ) {
106        assert_eq!(PromptMarker::default().strip(file).as_deref(), expected);
107    }
108
109    #[test]
110    fn a_custom_marker_renders_and_strips() {
111        let mut file = Vec::new();
112        let marker = PromptMarker::new("//");
113        let written = marker.write(&mut file, "Fixed it.", Some(PROMPT));
114        let file = String::from_utf8(file).unwrap_or_default();
115        assert!(written.is_ok(), "writing to a `Vec` can't fail");
116        assert_eq!(file, "Fixed it.\n\n// Release notes\n//\n// Leave the file empty to cancel.\n");
117        assert_eq!(marker.strip(&file).as_deref(), Some("Fixed it."));
118    }
