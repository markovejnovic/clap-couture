2//!
3//! The `-` comes out of argv before anything else, so every later parse sees the arg left out,
4//! and it is asked for or read like one the user never passed.

16/// An arg's subcommand path from the root, and its clap id.
17type ArgAt = (Vec<&'static str>, &'static str);
18
19/// The args the user passed `-` to, taken out of argv by [`take`].
20pub(super) struct Dashed(Vec<ArgAt>);
21
22impl Dashed {
23    /// Check whether arg `id` of the command at `path` was passed `-`.
24    pub(super) fn contains(&self, path: &[&str], id: &str) -> bool {
25        self.0.iter().any(|(dashed_path, dashed_id)| dashed_path == path && *dashed_id == id)
26    }
27
28    /// `cmd` with no env var on any of these args.
29    pub(super) fn without_env(&self, cmd: Command) -> Command {
30        self.0.iter().fold(cmd, |cmd, (path, id)| with_arg_at(cmd, path, id, |arg| arg.env(None)))
31    }
32}

34/// `argv` without the `-` passed to each `prompt(dash)` arg in `prompted_args` and
35/// `accept_stdin(dash)` arg in `stdin_args`, and the args it was taken from. `argv` that `cmd`
36/// rejects, eg. for `--help`, is left to clap as is.
37///
38/// # Errors
39/// A `-` that can't be taken out without changing another arg, as in `-vn-`.
40pub(super) fn take(
41    cmd: &Command,
42    prompted_args: &[PromptedArg],
43    stdin_args: &[StdinArg],
44    argv: &[OsString],
45) -> Result<(Vec<OsString>, Dashed), Failure> {
46    let relaxed = stdin_args.iter().fold(relax(cmd.clone(), prompted_args), |cmd, stdin_arg| {
47        unrequire(cmd, &stdin_arg.subcommand_path, stdin_arg.attr.arg_id)
48    });
49    let dash_args = prompted_args
50        .iter()
51        .filter(|prompted_arg| prompted_arg.attr.dash)
52        .map(|prompted_arg| (prompted_arg.subcommand_path.clone(), prompted_arg.attr.arg_id))
53        .chain(
54            stdin_args
55                .iter()
56                .filter(|stdin_arg| stdin_arg.attr.dash)
57                .map(|stdin_arg| (stdin_arg.subcommand_path.clone(), stdin_arg.attr.arg_id)),
58        );
59    let mut argv = argv.to_vec();
60    let mut dashed = Vec::new();
61    for (path, id) in dash_args {
62        let Ok(matches) = relaxed.clone().try_get_matches_from(&argv) else { break };
63        let Some(level) = matches_at(&matches, &path) else { continue };
64        if level.value_source(id) != Some(ValueSource::CommandLine)
65            || raw_values(level, id) != ["-"]
66        {
67            continue;
68        }
69        let Some(rest) = without_dash(&relaxed, level, &path, id, &argv) else {
70            return Err(error(
71                cmd,
72                &path,
73                id,
74                "got `-` in a way couture can't take out; pass `-` as a word of its own",
75            ));
76        };
77        argv = rest;
78        dashed.push((path, id));
79    }
80    Ok((argv, Dashed(dashed)))
81}

83/// The error for a `-` that asks for arg `id` of the command at `path` with no terminal to ask on.
84pub(super) fn unaskable(cmd: &Command, path: &[&str], id: &str) -> Failure {
85    error(cmd, path, id, "got `-` to be asked for, but there is no terminal to ask on")
86}

88/// `message` about arg `id` of the command at `path`, as clap formats its own errors.
89fn error(cmd: &Command, path: &[&str], id: &str, message: &str) -> Failure {
90    let mut built = cmd.clone();
91    built.build();
92    let arg = arg_at(&built, path, id).map_or_else(|| id.to_owned(), ToString::to_string);
93    Failure::Clap(
94        clap::Error::raw(ErrorKind::ValueValidation, format!("{arg} {message}"))
95            .format(&mut cmd.clone()),
96    )
97}

99/// `argv` with arg `id`'s `-` cut out, by one of [`cuts`]. `None` when no cut leaves the arg
100/// unpassed and every other arg of its command, `level`, as it was.
101fn without_dash(
102    relaxed: &Command,
103    level: &ArgMatches,
104    path: &[&str],
105    id: &str,
106    argv: &[OsString],
107) -> Option<Vec<OsString>> {
108    let others = |level: &ArgMatches| {
109        let mut others: Vec<_> = level
110            .ids()
111            // A group holds its args' values, so the one holding `id` would hold the `-` too.
112            .filter(|other| other.as_str() != id && arg_at(relaxed, path, other.as_str()).is_some())
113            .map(|other| {
114                let values = raw_values(level, other.as_str());
115                (other.clone(), level.value_source(other.as_str()), values)
116            })
117            .collect();
118        others.sort_by(|left, right| left.0.cmp(&right.0));
119        others
120    };
121    let before = others(level);
122    // The first word is the binary's name.
123    (1..argv.len()).flat_map(|index| cuts(argv, index)).find(|rest| {
124        let Ok(matches) = relaxed.clone().try_get_matches_from(rest) else { return false };
125        matches_at(&matches, path).is_some_and(|after| {
126            after.value_source(id) != Some(ValueSource::CommandLine) && others(after) == before
127        })
128    })
129}

131/// The ways to cut a `-` ending word `index` out of `argv`: the word alone (`-`, `--flag=-`,
132/// `-f-`), with the flag before it (`--flag -`), or with the last flag of a short cluster
133/// (`-vf -`, `-vf-`, `-vf=-`). Empty when the word doesn't end in `-`.
134fn cuts(argv: &[OsString], index: usize) -> Vec<Vec<OsString>> {
135    let Some(word) = argv.get(index).filter(|word| word.as_encoded_bytes().ends_with(b"-")) else {
136        return Vec::new();
137    };
138    let splice = |first: usize, replacement: Option<OsString>| {
139        let mut rest = argv.to_vec();
140        rest.splice(first..=index, replacement);
141        rest
142    };
143    let previous = index.checked_sub(1).filter(|previous| *previous > 0);
144    let mut cuts = vec![splice(index, None)];
145    if let Some(previous) = previous {
146        cuts.push(splice(previous, None));
147        if word == "-"
148            && let Some(head) = argv.get(previous).and_then(|flags| cluster_head(flags, ""))
149        {
150            cuts.push(splice(previous, Some(head)));
151        }
152    }
153    cuts.extend(
154        ["-", "=-"]
155            .into_iter()
156            .filter_map(|tail| cluster_head(word, tail))
157            .map(|head| splice(index, Some(head))),
158    );
159    cuts
160}

162/// `word`, a cluster of short flags followed by `tail`, without its last flag and `tail`: `-vf-`
163/// is `-v` for a `tail` of `-`. `None` for any other word, or a cluster of one flag.
164fn cluster_head(word: &OsStr, tail: &str) -> Option<OsString> {
165    let flags = word.to_str()?.strip_prefix('-')?.strip_suffix(tail)?;
166    if flags.starts_with('-') {
167        return None;
168    }
169    let (last, _) = flags.char_indices().last()?;
170    flags.get(..last).filter(|head| !head.is_empty()).map(|head| OsString::from(format!("-{head}")))
171}
