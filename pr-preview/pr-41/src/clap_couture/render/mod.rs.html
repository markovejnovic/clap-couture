23/// Styles a run of help text (a command's about or a category description).
24pub(crate) trait TextRenderer {
25    fn render_display<W, D>(&self, out: &mut W, text: D) -> fmt::Result
26    where
27        W: fmt::Write,
28        D: Display;
29
30    fn render_str<W>(&self, out: &mut W, text: &str) -> fmt::Result
31    where
32        W: fmt::Write;
33}
34
35pub(crate) trait CommandRenderExt {
36    /// The width clap wraps this command's help to; `None` when it doesn't wrap.
37    fn help_width(&self) -> Option<usize>;
38
39    /// Writes this command's visible subcommands to `out`.
40    fn write_command_sections<W>(&self, out: &mut W, categories: CommandCategoryMap) -> fmt::Result
41    where
42        W: fmt::Write;
43}
44
45impl CommandRenderExt for Command {
46    fn help_width(&self) -> Option<usize> {
47        // clap has no getter for it: it follows `term_width`, `max_term_width`, the terminal and
48        // `COLUMNS`, and wraps only with its `wrap_help` feature. So clap wraps a probe: a line
49        // breaks after the last word ending within the width, and the words of the `a` line end
50        // at odd columns while those of the `b` line end at even ones.
51        const PROBE_WORDS: usize = 500;
52        let odd = vec!["a"; PROBE_WORDS].join(" ");
53        let even = format!("b{}", vec!["b"; PROBE_WORDS].join(" "));
54        let unwrapped = odd.len();
55        let help = self
56            .clone()
57            .help_template("{after-help}")
58            .after_help(format!("{odd}\n{even}"))
59            .render_help()
60            .to_string();
61        let first_line =
62            |letter: char| help.lines().find(|line| line.starts_with(letter)).map_or(0, str::len);
63        let width = first_line('a').max(first_line('b'));
64        (width < unwrapped).then_some(width)
65    }
66
67    fn write_command_sections<W>(&self, out: &mut W, categories: CommandCategoryMap) -> fmt::Result
68    where
69        W: fmt::Write,
70    {
71        let menu = MenuView::new(self, categories, Renderer::new(self.get_styles()));
72        write!(out, "{menu}")
73    }
74}

76/// A command's visible subcommands, partitioned by category.
77struct Menu<'cmd> {
78    category_map: CommandCategoryMap,
79    parent: &'cmd Command,
80}
81
82impl<'cmd> Menu<'cmd> {
83    /// The distinct categories with at least one visible member, in first-appearance order.
84    fn categories(&self) -> impl Iterator<Item = &'static Category> {
85        self.category_map
86            .distinct()
87            .filter(|category| self.members(Some(category)).next().is_some())
88    }
89
90    fn has_uncategorized(&self) -> bool {
91        self.members(None).next().is_some()
92    }
93
94    /// The visible subcommands filed under `category` (`None` for the uncategorized ones).
95    fn members(&self, category: Option<&Category>) -> impl Iterator<Item = &'cmd Command> {
96        let label = category.map(|cat| cat.label);
97        self.subcommands()
98            .filter(move |cmd| self.category_map.find(cmd.get_name()).map(|cat| cat.label) == label)
99    }
100
101    /// The parent's subcommands, minus the hidden ones.
102    fn subcommands(&self) -> impl Iterator<Item = &'cmd Command> {
103        self.parent.get_subcommands().filter(|cmd| !cmd.is_hide_set())
104    }
105}

107/// A category heading, optionally followed by its description.
108struct HeadingView<'view, R> {
109    category: &'static Category,
110    view: &'view MenuView<'view, R>,
111}
112
113impl<R> Display for HeadingView<'_, R>
114where
115    R: TextRenderer,
116{
117    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
118        let header = self.view.styles.get_header();
119        let heading = self.category.heading();
120        write!(f, "{header}{heading}{header:#}")?;
121        self.category.description.as_ref().map_or(Ok(()), |description| {
122            let col = self.view.layout.description_col;
123            let pad = col.saturating_sub(heading.width());
124            write!(f, "{:pad$}", "")?;
125            let mut rendered = String::new();
126            self.view.renderer.render_str(&mut rendered, &description.get())?;
127            // clap counts the blank line above a heading as a cell of the heading's line.
128            let width = self.view.layout.width.map(|width| width.saturating_sub(1));
129            write!(f, "{}", ColumnText { col, text: &rendered, width })
130        })
131    }
132}

134/// Column positions shared by every section of a menu.
135struct MenuLayout {
136    /// Where a category description starts.
137    description_col: usize,
138    /// The widest command name; abouts are aligned two cells past it.
139    name_width: usize,
140    /// The width lines wrap to; `None` when they don't.
141    width: Option<usize>,
142}
143
144impl MenuLayout {
145    /// Where a command's about starts.
146    const fn about_col(&self) -> usize {
147        self.name_width.saturating_add(4)
148    }
149
150    fn measure<'name>(
151        names: impl Iterator<Item = &'name str>,
152        headings: impl Iterator<Item = Cow<'static, str>>,
153        width: Option<usize>,
154    ) -> Self {
155        let name_width = names.map(UnicodeWidthStr::width).max().unwrap_or(0);
156        let heading_width = headings.map(|h| h.width()).max().unwrap_or(0);
157
158        // A description at or next to the about column reads as another row, so push it two
159        // cells past.
160        let about_col = name_width.saturating_add(4);
161        let description_col = heading_width.saturating_add(2);
162        let description_col = if description_col.abs_diff(about_col) < 2 {
163            about_col.saturating_add(2)
164        } else {
165            description_col
166        };
167
168        Self { description_col, name_width, width }
169    }
170}

172/// A command's visible subcommands, grouped under their category headings.
173///
174/// ```text
175///   logs    Tail live logs
176///   status  Show recent activity
177///
178/// Deploy:     Ship code to production
179///   deploy  Ship the current project
180///
181/// Account:
182///   login   Sign in to your account
183/// ```
184struct MenuView<'cmd, R> {
185    layout: MenuLayout,
186    menu: Menu<'cmd>,
187    renderer: R,
188    styles: &'cmd Styles,
189}
190
191impl<'cmd, R> MenuView<'cmd, R> {
192    fn new(parent: &'cmd Command, categories: CommandCategoryMap, renderer: R) -> Self {
193        let menu = Menu { category_map: categories, parent };
194        let layout = MenuLayout::measure(
195            menu.subcommands().map(Command::get_name),
196            menu.categories().map(Category::heading),
197            parent.help_width(),
198        );
199        Self { layout, menu, renderer, styles: parent.get_styles() }
200    }
201}
202
203impl<R> Display for MenuView<'_, R>
204where
205    R: TextRenderer,
206{
207    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
208        let uncategorized =
209            self.menu.has_uncategorized().then_some(SectionView { category: None, view: self });
210        let categorized = self
211            .menu
212            .categories()
213            .map(|category| SectionView { category: Some(category), view: self });
214        write_joined(f, uncategorized.into_iter().chain(categorized), "\n\n")
215    }
216}

218/// One subcommand, with its about aligned past the widest name in the menu.
219struct RowView<'view, R> {
220    cmd: &'view Command,
221    view: &'view MenuView<'view, R>,
222}
223
224impl<R> Display for RowView<'_, R>
225where
226    R: TextRenderer,
227{
228    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
229        let literal = self.view.styles.get_literal();
230        let name = self.cmd.get_name();
231        let pad = self.view.layout.name_width.saturating_sub(name.width());
232        write!(f, "  {literal}{name}{literal:#}{:pad$}", "")?;
233        self.cmd.get_about().map_or(Ok(()), |about| {
234            f.write_str("  ")?;
235            let mut rendered = String::new();
236            self.view.renderer.render_display(&mut rendered, about)?;
237            let (col, width) = (self.view.layout.about_col(), self.view.layout.width);
238            write!(f, "{}", ColumnText { col, text: &rendered, width })
239        })
240    }
241}

243/// Rendered text starting at `col`, broken between words to fit `width`, each line after the
244/// first indented to `col`.
245struct ColumnText<'text> {
246    col: usize,
247    text: &'text str,
248    width: Option<usize>,
249}
250
251impl Display for ColumnText<'_> {
252    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
253        let Some(width) = self.width else {
254            return f.write_str(self.text);
255        };
256        let indent = self.col;
257        let mut column = indent;
258        // The escapes in effect, reopened past the indent of each new line.
259        let mut open = String::new();
260        for (index, word) in self.text.split(' ').map(Word).enumerate() {
261            if index > 0 {
262                if column.saturating_add(1).saturating_add(word.width()) > width {
263                    if !open.is_empty() {
264                        f.write_str(Word::RESET)?;
265                    }
266                    write!(f, "\n{:indent$}{open}", "")?;
267                    column = indent;
268                } else {
269                    f.write_str(" ")?;
270                    column = column.saturating_add(1);
271                }
272            }
273            f.write_str(word.0)?;
274            column = column.saturating_add(word.width());
275            for escape in word.escapes() {
276                if escape == Word::RESET {
277                    open.clear();
278                } else {
279                    open.push_str(escape);
280                }
281            }
282        }
283        Ok(())
284    }
285}

287/// A run of rendered text between spaces, its ANSI SGR escapes taking no cells.
288struct Word<'text>(&'text str);
289
290impl<'text> Word<'text> {
291    /// The escape that ends every style.
292    const RESET: &'static str = "\u{1b}[0m";
293
294    /// The escapes in this word, in order.
295    fn escapes(&self) -> impl Iterator<Item = &'text str> {
296        let text = self.0;
297        text.match_indices('\u{1b}').map(move |(start, _)| {
298            let escape = text.get(start..).unwrap_or_default();
299            escape.find('m').and_then(|end| escape.get(..=end)).unwrap_or(escape)
300        })
301    }
302
303    /// The cells this word takes on screen.
304    fn width(&self) -> usize {
305        let visible = |(index, piece): (usize, &'text str)| {
306            if index == 0 { piece } else { piece.split_once('m').map_or("", |(_, text)| text) }
307        };
308        self.0.split('\u{1b}').enumerate().map(visible).map(UnicodeWidthStr::width).sum()
309    }
310}

312/// A category's heading and rows, or just the rows for uncategorized commands.
313struct SectionView<'view, R> {
314    category: Option<&'static Category>,
315    view: &'view MenuView<'view, R>,
316}
317
318impl<R> Display for SectionView<'_, R>
319where
320    R: TextRenderer,
321{
322    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
323        self.category.map_or(Ok(()), |category| {
324            let heading = HeadingView { category, view: self.view };
325            writeln!(f, "{heading}")
326        })?;
327        let rows =
328            self.view.menu.members(self.category).map(|cmd| RowView { cmd, view: self.view });
329        write_joined(f, rows, "\n")
330    }
331}

333/// Writes each item of `items` to `out`, separated by `sep`.
334fn write_joined<W, I>(out: &mut W, items: I, sep: &str) -> fmt::Result
335where
336    W: fmt::Write,
337    I: IntoIterator,
338    I::Item: Display,
339{
340    let separators = iter::once("").chain(iter::repeat(sep));
341    items.into_iter().zip(separators).try_for_each(|(item, sep)| write!(out, "{sep}{item}"))
342}
