12/// The first enabled of cliclack, dialoguer and inquire, or
13/// [`NonInteractive`](super::NonInteractive) when none is.
14#[cfg(feature = "interactive-cliclack")]
15type Chosen = super::Cliclack;
16/// See the version with `interactive-cliclack` on.
17#[cfg(all(feature = "interactive-dialoguer", not(feature = "interactive-cliclack")))]
18type Chosen = super::Dialoguer;
19/// See the version with `interactive-cliclack` on.
20#[cfg(all(
21    feature = "interactive-inquire",
22    not(any(feature = "interactive-cliclack", feature = "interactive-dialoguer"))

24type Chosen = super::Inquire;
25/// See the version with `interactive-cliclack` on.
26#[cfg(not(any(
27    feature = "interactive-cliclack",
28    feature = "interactive-dialoguer",
29    feature = "interactive-inquire"
30)))]
31type Chosen = super::NonInteractive;

33/// Asks through the first enabled of cliclack, dialoguer and inquire, and never asks when none
34/// is, as [`NonInteractive`](super::NonInteractive) does.
35///
36/// It is one type under every feature set, so enabling another backend anywhere in the dependency
37/// tree changes the UI it shows, not code that names it. Name a backend, like
38/// [`Cliclack`](super::Cliclack), to pin the UI.
39#[derive(Clone, Copy, Debug, Default)]
40pub struct DefaultBackend(Chosen);
41
42impl Backend for DefaultBackend {
43    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
44        self.0.confirm(prompt)
45    }
46
47    #[cfg(feature = "interactive-editor")]
48    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
49        self.0.editor(prompt)
50    }
51
52    fn is_available(&self) -> bool {
53        self.0.is_available()
54    }
55
56    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
57        self.0.multi_select(prompt)
58    }
59
60    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
61        self.0.report(error, styles)
62    }
63
64    fn secret(&self, prompt: &SecretPrompt<'_>) -> Result<Option<SecretString>, PromptError> {
65        self.0.secret(prompt)
66    }
67
68    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
69        self.0.select(prompt)
70    }
71
72    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
73        self.0.text(prompt)
74    }
75}
