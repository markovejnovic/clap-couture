1//! [`Backend`] drawn by [dialoguer](https://docs.rs/dialoguer).

3use clap::builder::PossibleValue;
4use dialoguer::{Confirm, Input, Select, theme::ColorfulTheme};
5
6use super::{Backend, CommandStyles, builtin::ClapStyle};
7use crate::interactive::{ConfirmPrompt, PromptError, SelectPrompt, TextPrompt};
8
9/// A dialoguer failure.
10struct DialoguerError(dialoguer::Error);
11
12/// Prompts drawn by dialoguer's colorful theme.
13#[derive(Clone, Copy, Debug, Default)]
14pub struct Dialoguer;
15
16impl Backend for Dialoguer {
17    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
18        let theme = ColorfulTheme::from(prompt.styles);
19        Confirm::with_theme(&theme)
20            .with_prompt(prompt.question)
21            .default(prompt.default)
22            .interact_opt()
23            .map_err(DialoguerError)?
24            .ok_or(PromptError::Cancelled)
25    }
26
27    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
28        let theme = ColorfulTheme::from(styles);
29        console::Term::stderr()
30            .write_line(&theme.error_style.apply_to(error).to_string())
31            .map_err(PromptError::Io)
32    }
33
34    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
35        let theme = ColorfulTheme::from(prompt.styles);
36        let names: Vec<&str> = prompt.options.iter().map(PossibleValue::get_name).collect();
37        let none = prompt.optional.then_some(prompt.none_option);
38        let items: Vec<&str> = names.iter().copied().chain(none).collect();
39        let select = Select::with_theme(&theme).with_prompt(prompt.question).items(&items);
40        let default =
41            prompt.default.and_then(|default| names.iter().position(|name| *name == default));
42        let picked = match default {
43            Some(index) => select.default(index),
44            None => select,
45        }
46        .interact_opt()
47        .map_err(DialoguerError)?
48        .ok_or(PromptError::Cancelled)?;
49        // Past the options is the none option, only listed for an optional prompt.
50        Ok(names.get(picked).map(|name| (*name).to_owned()))
51    }
52
53    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
54        let theme = ColorfulTheme::from(prompt.styles);
55        let input = Input::<String>::with_theme(&theme)
56            .with_prompt(prompt.question)
57            .allow_empty(prompt.optional);
58        let answer = match prompt.default {
59            Some(default) => input.default(default.to_owned()),
60            None => input,
61        }
62        .interact_text()
63        .map_err(DialoguerError)?;
64        Ok(Some(answer).filter(|answer| !answer.is_empty()))
65    }
66}
67
68impl From<DialoguerError> for PromptError {
69    fn from(DialoguerError(err): DialoguerError) -> Self {
70        let dialoguer::Error::IO(err) = err;
71        Self::from(err)
72    }
73}
74
75impl From<CommandStyles<'_>> for ColorfulTheme {
76    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
77        Self {
78            defaults_style: console::Style::from(ClapStyle(styles.get_placeholder())),
79            error_style: console::Style::from(ClapStyle(styles.get_error())),
80            prompt_style: console::Style::from(ClapStyle(styles.get_header())),
81            values_style: console::Style::from(ClapStyle(styles.get_literal())),
82            ..Self::default()
83        }
84    }
