28/// Styles a run of help text (a command's about or a category description).
29pub(crate) trait TextRenderer {
30    fn render_display<W, D>(&self, out: &mut W, text: D) -> fmt::Result
31    where
32        W: fmt::Write,
33        D: Display;
34
35    fn render_str<W>(&self, out: &mut W, text: &str) -> fmt::Result
36    where
37        W: fmt::Write;
38
39    /// `text` as this renderer shows it, as text clap writes as is.
40    fn restyle(&self, text: &StyledStr) -> StyledStr;
41}

43pub(crate) trait CommandRenderExt {
44    /// The width clap wraps this command's help to; `None` when it doesn't wrap.
45    ///
46    /// clap's build hands a command's `term_width` and `max_term_width` to each subcommand, so
47    /// the built root's width is every built command's.
48    fn help_width(&self) -> Option<usize>;
49
50    /// `text` rendered on one line as a prompt's question: styled when this command's color is on
51    /// for stderr, else plain.
52    #[cfg(feature = "interactive")]
53    fn render_question(&self, text: &str) -> String;
54
55    /// This command with its about and long about rendered.
56    #[must_use]
57    fn with_rendered_about(self) -> Self;
58
59    /// This command with each arg's help and long help rendered.
60    #[must_use]
61    fn with_rendered_arg_help(self) -> Self;
62
63    /// Writes this command's visible subcommands to `out`, wrapped to `width`.
64    fn write_command_sections<W>(
65        &self,
66        out: &mut W,
67        categories: CommandCategoryMap,
68        width: Option<usize>,
69    ) -> fmt::Result
70    where
71        W: fmt::Write;
72}

74impl CommandRenderExt for Command {
75    fn help_width(&self) -> Option<usize> {
76        // clap has no getter for it: it follows `term_width`, `max_term_width`, the terminal and
77        // `COLUMNS`, and wraps only with its `wrap_help` feature. So clap wraps a probe: a line
78        // breaks after the last word ending within the width, and the words of the `a` line end
79        // at odd columns while those of the `b` line end at even ones.
80        const PROBE_WORDS: usize = 500;
81        let odd = vec!["a"; PROBE_WORDS].join(" ");
82        let even = format!("b{}", vec!["b"; PROBE_WORDS].join(" "));
83        let unwrapped = odd.len();
84        let help = self
85            .clone()
86            .help_template("{after-help}")
87            .after_help(format!("{odd}\n{even}"))
88            .render_help()
89            .to_string();
90        let first_line =
91            |letter: char| help.lines().find(|line| line.starts_with(letter)).map_or(0, str::len);
92        let width = first_line('a').max(first_line('b'));
93        (width < unwrapped).then_some(width)
94    }
95
96    #[cfg(feature = "interactive")]
97    fn render_question(&self, text: &str) -> String {
98        let colored = match self.get_color() {
99            ColorChoice::Always => true,
100            ColorChoice::Never => false,
101            ColorChoice::Auto => {
102                anstream::AutoStream::choice(&std::io::stderr()) != anstream::ColorChoice::Never
103            }
104        };
105        let mut question = StyledStr::new();
106        // `StyledStr`'s `fmt::Write` is infallible.
107        Renderer::new(self.get_styles()).render_str(&mut question, text).ok();
108        if colored { question.ansi().to_string() } else { question.to_string() }
109    }
110
111    fn with_rendered_about(self) -> Self {
112        let renderer = Renderer::new(self.get_styles());
113        let about = self.get_about().map(|text| renderer.restyle(text));
114        let long_about = self.get_long_about().map(|text| renderer.restyle(text));
115        self.about(Resettable::from(about)).long_about(Resettable::from(long_about))
116    }
117
118    fn with_rendered_arg_help(self) -> Self {
119        let styles = self.get_styles().clone();
120        let renderer = Renderer::new(&styles);
121        self.mut_args(|arg| {
122            let help = arg.get_help().map(|text| renderer.restyle(text));
123            let long_help = arg.get_long_help().map(|text| renderer.restyle(text));
124            arg.help(Resettable::from(help)).long_help(Resettable::from(long_help))
125        })
126    }
127
128    fn write_command_sections<W>(
129        &self,
130        out: &mut W,
131        categories: CommandCategoryMap,
132        width: Option<usize>,
133    ) -> fmt::Result
134    where
135        W: fmt::Write,
136    {
137        let menu = MenuView::new(self, categories, Renderer::new(self.get_styles()), width);
138        write!(out, "{menu}")
139    }
140}

142/// A command's visible subcommands, partitioned by category.
143struct Menu<'cmd> {
144    category_map: CommandCategoryMap,
145    parent: &'cmd Command,
146}
147
148impl<'cmd> Menu<'cmd> {
149    /// The distinct categories with at least one visible member, in first-appearance order.
150    fn categories(&self) -> impl Iterator<Item = &'static Category> {
151        self.category_map
152            .distinct()
153            .filter(|category| self.members(Some(category)).next().is_some())
154    }
155
156    fn has_uncategorized(&self) -> bool {
157        self.members(None).next().is_some()
158    }
159
160    /// The `help` subcommand clap generated for the parent; `None` when it didn't, or when clap
161    /// lists no subcommands, as when every other one is hidden.
162    fn help_subcommand(&self) -> Option<&'cmd Command> {
163        let listed = self.subcommands().any(|cmd| cmd.get_name() != "help");
164        self.subcommands().find(|cmd| listed && self.is_help_subcommand(cmd))
165    }
166
167    /// Check whether `cmd` is the `help` subcommand clap generated for the parent.
168    fn is_help_subcommand(&self, cmd: &Command) -> bool {
169        !self.parent.is_disable_help_subcommand_set() && cmd.get_name() == "help"
170    }
171
172    /// The visible subcommands filed under `category` (`None` for the uncategorized ones), minus
173    /// clap's `help`.
174    fn members(&self, category: Option<&Category>) -> impl Iterator<Item = &'cmd Command> {
175        let label = category.map(|cat| cat.label);
176        self.subcommands().filter(move |cmd| {
177            !self.is_help_subcommand(cmd)
178                && self.category_map.find(cmd.get_name()).map(|cat| cat.label) == label
179        })
180    }
181
182    /// The parent's subcommands, minus the hidden ones, in clap's display order.
183    fn subcommands(&self) -> impl Iterator<Item = &'cmd Command> {
184        let mut visible: Vec<_> =
185            self.parent.get_subcommands().filter(|cmd| !cmd.is_hide_set()).collect();
186        // `mut_subcommand` moves the subcommand it edits to the end, as guarding a secret does.
187        visible.sort_by_key(|cmd| cmd.get_display_order());
188        visible.into_iter()
189    }
190}

192/// A category heading, optionally followed by its description.
193struct HeadingView<'view, R> {
194    category: &'static Category,
195    view: &'view MenuView<'view, R>,
196}
197
198impl<R> Display for HeadingView<'_, R>
199where
200    R: TextRenderer,
201{
202    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
203        let header = self.view.styles.get_header();
204        let heading = self.category.heading();
205        write!(f, "{header}{heading}{header:#}")?;
206        self.category.description.as_ref().map_or(Ok(()), |description| {
207            let col = self.view.layout.description_col;
208            let pad = col.saturating_sub(heading.width());
209            write!(f, "{:pad$}", "")?;
210            let mut rendered = String::new();
211            self.view.renderer.render_str(&mut rendered, &description.get())?;
212            // clap counts the blank line above a heading as a cell of the heading's line.
213            let width = self.view.layout.width.map(|width| width.saturating_sub(1));
214            write!(f, "{}", ColumnText { col, text: &rendered, width })
215        })
216    }
217}

219/// Column positions shared by every section of a menu.
220struct MenuLayout {
221    /// Where a category description starts.
222    description_col: usize,
223    /// The widest command name; abouts are aligned two cells past it.
224    name_width: usize,
225    /// The width lines wrap to; `None` when they don't.
226    width: Option<usize>,
227}
228
229impl MenuLayout {
230    /// Where a command's about starts.
231    const fn about_col(&self) -> usize {
232        self.name_width.saturating_add(4)
233    }
234
235    fn measure<'name>(
236        names: impl Iterator<Item = &'name str>,
237        headings: impl Iterator<Item = Cow<'static, str>>,
238        width: Option<usize>,
239    ) -> Self {
240        let name_width = names.map(UnicodeWidthStr::width).max().unwrap_or(0);
241        let heading_width = headings.map(|h| h.width()).max().unwrap_or(0);
242
243        // A description at or next to the about column reads as another row, so push it two
244        // cells past.
245        let about_col = name_width.saturating_add(4);
246        let description_col = heading_width.saturating_add(2);
247        let description_col = if description_col.abs_diff(about_col) < 2 {
248            about_col.saturating_add(2)
249        } else {
250            description_col
251        };
252
253        Self { description_col, name_width, width }
254    }
255}

257/// A command's visible subcommands, grouped under their category headings.
258///
259/// ```text
260///   logs    Tail live logs
261///   status  Show recent activity
262///
263/// Deploy:     Ship code to production
264///   deploy  Ship the current project
265///
266/// Account:
267///   login   Sign in to your account
268///
269///   help    Print this message or the help of the given subcommand(s)
270/// ```
271struct MenuView<'cmd, R> {
272    layout: MenuLayout,
273    menu: Menu<'cmd>,
274    renderer: R,
275    styles: &'cmd Styles,
276}
277
278impl<'cmd, R> MenuView<'cmd, R> {
279    fn new(
280        parent: &'cmd Command,
281        categories: CommandCategoryMap,
282        renderer: R,
283        width: Option<usize>,
284    ) -> Self {
285        let menu = Menu { category_map: categories, parent };
286        let layout = MenuLayout::measure(
287            menu.subcommands().map(Command::get_name),
288            menu.categories().map(Category::heading),
289            width,
290        );
291        Self { layout, menu, renderer, styles: parent.get_styles() }
292    }
293}
294
295impl<R> Display for MenuView<'_, R>
296where
297    R: TextRenderer,
298{
299    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
300        let uncategorized =
301            self.menu.has_uncategorized().then_some(SectionView { category: None, view: self });
302        let categorized = self
303            .menu
304            .categories()
305            .map(|category| SectionView { category: Some(category), view: self });
306        let has_categories = self.menu.categories().next().is_some();
307        write_joined(f, uncategorized.into_iter().chain(categorized), "\n\n")?;
308        self.menu.help_subcommand().map_or(Ok(()), |cmd| {
309            // clap's `help` ends the uncategorized rows, or follows the categories a section apart.
310            let sep = match (has_categories, self.menu.has_uncategorized()) {
311                (true, _) => "\n\n",
312                (false, true) => "\n",
313                (false, false) => "",
314            };
315            write!(f, "{sep}{}", RowView { cmd, view: self })
316        })
317    }
318}

320/// One subcommand, with its about aligned past the widest name in the menu.
321struct RowView<'view, R> {
322    cmd: &'view Command,
323    view: &'view MenuView<'view, R>,
324}
325
326impl<R> Display for RowView<'_, R>
327where
328    R: TextRenderer,
329{
330    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
331        let literal = self.view.styles.get_literal();
332        let name = self.cmd.get_name();
333        let pad = self.view.layout.name_width.saturating_sub(name.width());
334        write!(f, "  {literal}{name}{literal:#}{:pad$}", "")?;
335        self.cmd.get_about().map_or(Ok(()), |about| {
336            f.write_str("  ")?;
337            let mut rendered = String::new();
338            self.view.renderer.render_display(&mut rendered, about)?;
339            let (col, width) = (self.view.layout.about_col(), self.view.layout.width);
340            write!(f, "{}", ColumnText { col, text: &rendered, width })
341        })
342    }
343}

345/// Rendered text starting at `col`, broken between words to fit `width`, each line after the
346/// first indented to `col`.
347struct ColumnText<'text> {
348    col: usize,
349    text: &'text str,
350    width: Option<usize>,
351}
352
353impl Display for ColumnText<'_> {
354    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
355        let Some(width) = self.width else {
356            return f.write_str(self.text);
357        };
358        let indent = self.col;
359        let mut column = indent;
360        // The escapes in effect, reopened past the indent of each new line.
361        let mut open = String::new();
362        for (index, word) in self.text.split(' ').map(Word).enumerate() {
363            if index > 0 {
364                if column.saturating_add(1).saturating_add(word.width()) > width {
365                    if !open.is_empty() {
366                        f.write_str(Word::RESET)?;
367                    }
368                    write!(f, "\n{:indent$}{open}", "")?;
369                    column = indent;
370                } else {
371                    f.write_str(" ")?;
372                    column = column.saturating_add(1);
373                }
374            }
375            f.write_str(word.0)?;
376            column = column.saturating_add(word.width());
377            for escape in word.escapes() {
378                if escape == Word::RESET {
379                    open.clear();
380                } else {
381                    open.push_str(escape);
382                }
383            }
384        }
385        Ok(())
386    }
387}

389/// A run of rendered text between spaces, its ANSI SGR escapes taking no cells.
390struct Word<'text>(&'text str);
391
392impl<'text> Word<'text> {
393    /// The escape that ends every style.
394    const RESET: &'static str = "\u{1b}[0m";
395
396    /// The escapes in this word, in order.
397    fn escapes(&self) -> impl Iterator<Item = &'text str> {
398        let text = self.0;
399        text.match_indices('\u{1b}').map(move |(start, _)| {
400            let escape = text.get(start..).unwrap_or_default();
401            escape.find('m').and_then(|end| escape.get(..=end)).unwrap_or(escape)
402        })
403    }
404
405    /// The cells this word takes on screen.
406    fn width(&self) -> usize {
407        let visible = |(index, piece): (usize, &'text str)| {
408            if index == 0 { piece } else { piece.split_once('m').map_or("", |(_, text)| text) }
409        };
410        self.0.split('\u{1b}').enumerate().map(visible).map(UnicodeWidthStr::width).sum()
411    }
412}

414/// A category's heading and rows, or just the rows for uncategorized commands.
415struct SectionView<'view, R> {
416    category: Option<&'static Category>,
417    view: &'view MenuView<'view, R>,
418}
419
420impl<R> Display for SectionView<'_, R>
421where
422    R: TextRenderer,
423{
424    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
425        self.category.map_or(Ok(()), |category| {
426            let heading = HeadingView { category, view: self.view };
427            writeln!(f, "{heading}")
428        })?;
429        let rows =
430            self.view.menu.members(self.category).map(|cmd| RowView { cmd, view: self.view });
431        write_joined(f, rows, "\n")
432    }
433}

435/// Writes each item of `items` to `out`, separated by `sep`.
436fn write_joined<W, I>(out: &mut W, items: I, sep: &str) -> fmt::Result
437where
438    W: fmt::Write,
439    I: IntoIterator,
440    I::Item: Display,
441{
442    let separators = iter::once("").chain(iter::repeat(sep));
443    items.into_iter().zip(separators).try_for_each(|(item, sep)| write!(out, "{sep}{item}"))
