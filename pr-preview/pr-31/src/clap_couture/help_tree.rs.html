1//! The categories and messages of each command's help, from the root down.
2
3use clap::Command;
4
5use crate::{CommandCategoryMap, Messages, render::HelpBody};
6
7/// What hangs off one command level of a [`HelpTree`].
8#[non_exhaustive]
9pub enum HelpBranch {
10    /// A tree merged into this level: a flattened type, or the subcommand enum.
11    Flattened(&'static HelpTree),
12    /// The subcommand with this name.
13    Subcommand(&'static str, &'static HelpTree),
14}
15
16/// One command's categories and messages, and the trees of the commands below it.
17pub struct HelpTree {
18    /// Merged trees and subcommands.
19    pub branches: &'static [HelpBranch],
20    /// The categories its subcommands are filed under.
21    pub categories: CommandCategoryMap,
22    /// The messages its type sets, for its help and the help below it; `None` to keep its
23    /// parent's.
24    pub messages: Option<Messages>,
25}
26
27impl HelpTree {
28    /// A command with no categories, no messages of its own, and nothing below it.
29    pub const EMPTY: Self =
30        Self { branches: &[], categories: CommandCategoryMap::EMPTY, messages: None };
31
32    /// `cmd` showing couture's help filed by `categories` and worded by `messages`, and each
33    /// subcommand below it showing its own, from this tree.
34    ///
35    /// `built` is `cmd` built by clap, so each help lists what clap gives a subcommand: global
36    /// args, styles and settings from its parents. Each menu wraps to `width`.
37    pub(crate) fn install(
38        &self,
39        cmd: Command,
40        built: &Command,
41        categories: CommandCategoryMap,
42        messages: &Messages,
43        width: Option<usize>,
44    ) -> Command {
45        let cmd = cmd.mut_subcommands(|subcommand| {
46            let Some(built_subcommand) = built.find_subcommand(subcommand.get_name()) else {
47                return subcommand;
48            };
49            let (tree, messages) = self
50                .subcommand(subcommand.get_name(), *messages)
51                .unwrap_or((&Self::EMPTY, *messages));
52            tree.install(subcommand, built_subcommand, tree.categories, &messages, width)
53        });
54        HelpBody::new(built, categories, messages, width).install(cmd, messages)
55    }
56
57    /// The tree of the subcommand `name`, and the messages it inherits from `inherited` down
58    /// through the trees merged on the way; `None` when no branch names it.
59    fn subcommand(&self, name: &str, inherited: Messages) -> Option<(&Self, Messages)> {
60        self.branches.iter().find_map(|branch| match branch {
61            HelpBranch::Flattened(tree) => {
62                tree.subcommand(name, tree.messages.unwrap_or(inherited))
63            }
64            HelpBranch::Subcommand(subcommand_name, tree) => {
65                (*subcommand_name == name).then(|| (*tree, tree.messages.unwrap_or(inherited)))
66            }
67        })
68    }
69}
