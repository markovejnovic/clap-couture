21/// The name of a program called as `called`, like `git` for `/usr/bin/git`, as clap names it.
22pub(crate) fn program_name(called: &OsStr) -> Option<&OsStr> {
23    Path::new(called).file_name()
24}
25
26/// The running program's name, like `git`: as it was called, else as its file is named.
27#[cfg(feature = "interactive-editor")]
28pub(crate) fn process_name() -> Option<OsString> {
29    let called = std::env::args_os().next().filter(|called| !called.is_empty());
30    let path = called.map(Into::into).or_else(|| std::env::current_exe().ok())?;
31    let name = program_name(path.as_os_str())?;
32    Path::new(name).file_stem().map(ToOwned::to_owned)
33}
