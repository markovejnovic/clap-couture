1//! Parse, ask for the prompted args that got no value, then parse again with the answers.
2//!
3//! Every answer goes back into the command as its arg's `default_values`, so clap converts and
4//! validates it exactly like a typed value. Each answer is checked with a trial parse as soon as
5//! it is given: the previous trial passed, so a failure can only come from the new answer, which
6//! is then asked again with clap's message. For an arg asked one value at a time, each value is
7//! an answer of its own, checked on top of the values before it.
8//!
9//! Confirmations come last, so their questions can name the answers. A declined one fails the
10//! parse; without a terminal to ask on, one left unanswered fails it as a missing required arg.
11

34/// A prompted arg's accepted answer.
35struct Answer<'prompted_arg> {
36    prompted_arg: &'prompted_arg PromptedArg,
37    /// One value, or any number for an arg that takes many.
38    values: Vec<String>,
39}

41/// A prompted arg about to be asked in the editor, and what is known about it.
42#[cfg(feature = "interactive-editor")]
43struct Asked<'ask> {
44    arg: &'ask Arg,
45    /// Its default's value, pre-filled.
46    current: Option<&'ask str>,
47    /// Leaving it unset passes clap, so an empty answer is allowed.
48    optional: bool,
49    prompted_arg: &'ask PromptedArg,
50    question: &'ask str,
51}

53/// An answer clap turned down, and why; an editor prompt reopens with it.
54#[cfg(feature = "interactive-editor")]
55struct Rejected {
56    error: String,
57    value: String,

60/// One parse: the command, its argv, its prompted args and who to ask.
61struct Session<'run> {
62    argv: &'run [OsString],
63    backend: &'run dyn Backend,
64    cmd: &'run Command,
65    messages: &'run Messages,
66    prompted_args: &'run [PromptedArg],
67}
68
69impl Session<'_> {
70    /// Ask for `prompted_arg` until clap accepts the answer; `None` when an optional single-value
71    /// one is left empty. `current` is the arg's default, split into its values. `later` are the
72    /// prompted args still to be asked after it.
73    fn answer(
74        &self,
75        prompted_arg: &PromptedArg,
76        current: &[String],
77        answers: &[Answer<'_>],
78        later: &[&PromptedArg],
79    ) -> Result<Option<Vec<String>>, PromptError> {
80        let Some(arg) = arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
81        else {
82            return Ok(None);
83        };
84        let optional = !self.required(answers, later);
85        let question = self.question(prompted_arg);
86        // A flag has nothing to edit, so one that slipped past the derive asks as a flag.
87        #[cfg(feature = "interactive-editor")]
88        if let Some(editor) = prompted_arg
89            .attr
90            .editor
91            .as_ref()
92            .filter(|_| arg.get_action().takes_values() && !takes_many(arg))
93        {
94            let asked = Asked {
95                arg,
96                current: current.first().map(String::as_str),
97                optional,
98                prompted_arg,
99                question: &question,
100            };
101            return self.in_editor(&asked, editor, answers);
102        }
103        let options: Vec<PossibleValue> =
104            arg.get_possible_values().into_iter().filter(|value| !value.is_hide_set()).collect();
105        if !takes_many(arg) {
106            return self.until_accepted(answers, prompted_arg, || {
107                let value = self.ask(arg, &question, &options, current.first(), optional)?;
108                Ok(value.map(|value| vec![value]))
109            });
110        }
111        if options.is_empty() {
112            return self.each_value(&question, prompted_arg, current, answers, optional).map(Some);
113        }
114        self.until_accepted(answers, prompted_arg, || {
115            self.backend
116                .multi_select(&MultiSelectPrompt {
117                    defaults: current,
118                    optional,
119                    options: &options,
120                    question: &question,
121                    selection_required: &self.messages.selection_required.get(),
122                    styles: CommandStyles(self.cmd.get_styles()),
123                })
124                .map(Some)
125        })
126    }
127
128    fn ask(
129        &self,
130        arg: &Arg,
131        question: &str,
132        options: &[PossibleValue],
133        current: Option<&String>,
134        optional: bool,
135    ) -> Result<Option<String>, PromptError> {
136        let styles = CommandStyles(self.cmd.get_styles());
137        if let Some(negate) = flag_negation(arg.get_action()) {
138            let set = self.backend.confirm(&ConfirmPrompt { default: false, question, styles })?;
139            return Ok(Some((set != negate).to_string()));
140        }
141        if options.is_empty() {
142            return self.backend.text(&TextPrompt {
143                default: current.map(String::as_str),
144                hint: None,
145                initial: None,
146                optional,
147                question,
148                styles,
149            });
150        }
151        self.backend.select(&SelectPrompt {
152            default: current.map(String::as_str),
153            none_option: &self.messages.none.get(),
154            optional,
155            options,
156            question,
157            styles,
158        })
159    }
160
161    /// Ask in the user's editor, starting from the rejected answer if there is one. An empty
162    /// answer to a required arg aborts, as git aborts on an empty commit message, and so does one
163    /// to an arg with a default, which an empty answer would quietly bring back.
164    #[cfg(feature = "interactive-editor")]
165    fn ask_in_editor(
166        &self,
167        asked: &Asked<'_>,
168        editor: &EditorAttr,
169        rejected: Option<&Rejected>,
170    ) -> Result<Option<String>, PromptError> {
171        let Asked { arg, current, optional: _, prompted_arg, question } = *asked;
172        let styles = CommandStyles(self.cmd.get_styles());
173        let optional = asked.optional && arg.get_default_values().is_empty();
174        let instructions = if optional {
175            self.messages.editor_instructions_optional.get()
176        } else {
177            self.messages.editor_instructions.get()
178        };
179        let answer = self.backend.editor(&EditorPrompt {
180            default: rejected.map_or(current, |rejected| Some(rejected.value.as_str())),
181            error: rejected.map(|rejected| rejected.error.as_str()),
182            extension: editor.extension,
183            file_stem: &iter::once(self.cmd.get_name())
184                .chain(prompted_arg.subcommand_path.iter().copied())
185                .chain(iter::once(prompted_arg.attr.arg_id))
186                .collect::<Vec<_>>()
187                .join("-"),
188            instructions: &instructions,
189            not_utf8: &self.messages.editor_not_utf8.get(),
190            optional,
191            question,
192            returned_early: &self.messages.editor_returned_early.get(),
193            styles,
194            waiting: &self.messages.editor_waiting.get(),
195        })?;
196        if answer.is_none() && !optional {
197            self.backend.report(&self.messages.editor_empty.get(), styles)?;
198            return Err(PromptError::Cancelled);
199        }
200        Ok(answer)
201    }
202
203    /// Ask whether to go ahead, with each `{id}` in the question filled in from the args as
204    /// passed and answered so far.
205    fn confirm(
206        &self,
207        prompted_arg: &PromptedArg,
208        answers: &[Answer<'_>],
209    ) -> Result<bool, PromptError> {
210        let question = self.question(prompted_arg);
211        let answered = inject(
212            relax(self.cmd.clone(), self.prompted_args),
213            answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice())),
214        )
215        .try_get_matches_from(self.argv);
216        let question = match answered
217            .as_ref()
218            .ok()
219            .and_then(|answered| matches_at(answered, &prompted_arg.subcommand_path))
220        {
221            Some(level) => interpolate(&question, level),
222            None => question,
223        };
224        self.backend.confirm(&ConfirmPrompt {
225            default: false,
226            question: &question,
227            styles: CommandStyles(self.cmd.get_styles()),
228        })
229    }
230
231    /// Ask for one value at a time until an empty answer, each starting out as the default's
232    /// value at its position. Only a required arg's first value may not be empty.
233    ///
234    /// The default goes in as initial text rather than as the answer Enter accepts, so clearing it
235    /// still finishes the list.
236    fn each_value(
237        &self,
238        question: &str,
239        prompted_arg: &PromptedArg,
240        current: &[String],
241        answers: &[Answer<'_>],
242        optional: bool,
243    ) -> Result<Vec<String>, PromptError> {
244        let finish_hint = self.messages.finish_hint.get();
245        let mut values = Vec::new();
246        loop {
247            let optional = !values.is_empty() || optional;
248            let Some(value) = self.backend.text(&TextPrompt {
249                default: None,
250                hint: optional.then_some(&*finish_hint),
251                initial: current.get(values.len()).map(String::as_str),
252                optional,
253                question,
254                styles: CommandStyles(self.cmd.get_styles()),
255            })?
256            else {
257                return Ok(values);
258            };
259            values.push(value);
260            if let Err(err) = self.trial(answers, prompted_arg, &values) {
261                self.backend
262                    .report(&summary(&err, &values), CommandStyles(self.cmd.get_styles()))?;
263                values.pop();
264            }
265        }
266    }

268    /// Ask in the editor until clap accepts the answer, reopening with the rejected one.
269    #[cfg(feature = "interactive-editor")]
270    fn in_editor(
271        &self,
272        asked: &Asked<'_>,
273        editor: &EditorAttr,
274        answers: &[Answer<'_>],
275    ) -> Result<Option<Vec<String>>, PromptError> {
276        let mut rejected = None;
277        loop {
278            let Some(value) = self.ask_in_editor(asked, editor, rejected.as_ref())? else {
279                return Ok(None);
280            };
281            let values = vec![value];
282            match self.trial(answers, asked.prompted_arg, &values) {
283                Ok(()) => return Ok(Some(values)),
284                Err(err) => {
285                    let error = summary(&err, &values);
286                    self.backend.report(&error, CommandStyles(self.cmd.get_styles()))?;
287                    rejected = values.into_iter().next().map(|value| Rejected { error, value });
288                }
289            }
290        }
291    }
292
293    /// Check whether leaving the arg being asked unset fails the parse: it is `required`, or
294    /// required through `required_unless_present` or a required group. The accepted answers go
295    /// in, and only the args still to be asked, `later`, stay relaxed, so a group another of them
296    /// can still satisfy doesn't make this one required.
297    fn required(&self, answers: &[Answer<'_>], later: &[&PromptedArg]) -> bool {
298        let accepted = answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice()));
299        let cmd = inject(relax(self.cmd.clone(), later.iter().copied()), accepted);
300        cmd.try_get_matches_from(self.argv)
301            .is_err_and(|err| err.kind() == ErrorKind::MissingRequiredArgument)
302    }
303
304    /// The `= ...` override, else the arg's help text, else its id.
305    fn question(&self, prompted_arg: &PromptedArg) -> String {
306        prompted_arg.attr.question.as_ref().map_or_else(
307            || {
308                arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
309                    .and_then(Arg::get_help)
310                    .map_or_else(|| prompted_arg.attr.arg_id.to_owned(), ToString::to_string)
311            },
312            |question| question.get().into_owned(),
313        )
314    }
315
316    /// Parse with `candidate` as `prompted_arg`'s answer on top of the accepted ones.
317    fn trial(
318        &self,
319        answers: &[Answer<'_>],
320        prompted_arg: &PromptedArg,
321        candidate: &[String],
322    ) -> Result<(), clap::Error> {
323        let accepted = answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice()));
324        let cmd = inject(
325            relax(self.cmd.clone(), self.prompted_args.iter()),
326            accepted.chain(iter::once((prompted_arg, candidate))),
327        );
328        cmd.try_get_matches_from(self.argv).map(drop)
329    }
330
331    /// Call `ask` until clap accepts its answer, showing clap's complaint before each retry.
332    fn until_accepted(
333        &self,
334        answers: &[Answer<'_>],
335        prompted_arg: &PromptedArg,
336        mut ask: impl FnMut() -> Result<Option<Vec<String>>, PromptError>,
337    ) -> Result<Option<Vec<String>>, PromptError> {
338        loop {
339            let Some(values) = ask()? else { return Ok(None) };
340            match self.trial(answers, prompted_arg, &values) {
341                Ok(()) => return Ok(Some(values)),
342                Err(err) => {
343                    let error = summary(&err, &values);
344                    self.backend.report(&error, CommandStyles(self.cmd.get_styles()))?;
345                }
346            }
347        }
348    }
349}

351/// Parse `argv` against `cmd`, asking `backend` for the prompted args in `prompted_arg_tree` that
352/// got no value.
353pub(super) fn parse<T>(
354    cmd: Command,
355    prompted_arg_tree: &'static PromptedArgTree,
356    messages: &Messages,
357    argv: &[OsString],
358    backend: &dyn Backend,
359) -> Result<T, Failure>
360where
361    T: FromArgMatches,
362{
363    let prompted_args = prompted_arg_tree.prompted_args();
364    if prompted_args.is_empty() || !backend.is_available() {
365        return parse_unprompted(cmd, &prompted_args, argv);
366    }
367    // Help, usage and errors come from the command the user sees, not the relaxed one.
368    let Ok(matches) = relax(cmd.clone(), prompted_args.iter()).try_get_matches_from(argv) else {
369        return super::parse(cmd, argv);
370    };
371
372    let session = Session { argv, backend, cmd: &cmd, messages, prompted_args: &prompted_args };
373    let mut answers = Vec::new();
374    let mut unanswered = pending(&cmd, &matches, &prompted_args);
375    // Stable, so values and confirmations each keep declaration order.
376    unanswered
377        .sort_by_key(|(prompted_arg, _)| prompted_arg.attr.purpose == PromptPurpose::Confirmation);
378    for (index, (prompted_arg, current)) in unanswered.iter().enumerate() {
379        if blocked(&cmd, &matches, prompted_arg, &answers) {
380            continue;
381        }
382        let later: Vec<&PromptedArg> =
383            unanswered.iter().skip(index.saturating_add(1)).map(|(later, _)| *later).collect();
384        let answer = match prompted_arg.attr.purpose {
385            PromptPurpose::Confirmation => match session.confirm(prompted_arg, &answers) {
386                Ok(false) => return Err(Failure::Declined),
387                confirmed => confirmed.map(|_| Some(Vec::from(["true".to_owned()]))),
388            },
389            PromptPurpose::Value => session.answer(prompted_arg, current, &answers, &later),
390        };
391        match answer {
392            Ok(Some(values)) => answers.push(Answer { prompted_arg, values }),
393            Ok(None) => {}
394            // Backing out counts as Ctrl-C.
395            Err(PromptError::Cancelled) => return Err(Failure::Cancelled(Signal::Interrupt)),
396            Err(PromptError::Signalled(signal)) => return Err(Failure::Cancelled(signal)),
397            Err(PromptError::Io(err)) => return Err(Failure::Io(err)),
398            Err(PromptError::NotATerminal) => return parse_unprompted(cmd, &prompted_args, argv),
399        }
400    }
401    let cmd_with_answers =
402        inject(cmd, answers.iter().map(|answer| (answer.prompted_arg, answer.values.as_slice())));
403    super::parse(cmd_with_answers, argv)
404}

406/// Parse as clap would, without asking. A confirmation left unanswered fails the parse with
407/// clap's error for a missing required arg, so only passing its flag gets past it.
408fn parse_unprompted<T>(
409    mut cmd: Command,
410    prompted_args: &[PromptedArg],
411    argv: &[OsString],
412) -> Result<T, Failure>
413where
414    T: FromArgMatches,
415{
416    let confirmations: Vec<_> = prompted_args
417        .iter()
418        .filter(|prompted_arg| prompted_arg.attr.purpose == PromptPurpose::Confirmation)
419        .collect();
420    if confirmations.is_empty() {
421        return super::parse(cmd, argv);
422    }
423    // Kept unbuilt: requiring an arg on the built command makes clap show help instead.
424    let declared = cmd.clone();
425    let mut matches = cmd.try_get_matches_from_mut(argv).map_err(Failure::Clap)?;
426    let unconfirmed = confirmations.iter().any(|confirmation| {
427        let id = confirmation.attr.arg_id;
428        let passed = matches_at(&matches, &confirmation.subcommand_path)
429            .is_none_or(|level| explicit(level, &Id::from(id)));
430        !passed && !blocked(&cmd, &matches, confirmation, &[])
431    });
432    if unconfirmed {
433        // Parsed again only to fail: clap words the error, and the help stays as declared.
434        let required = confirmations.iter().fold(declared, |cmd, confirmation| {
435            with_arg_at(cmd, &confirmation.subcommand_path, confirmation.attr.arg_id, |arg| {
436                arg.required(true)
437            })
438        });
439        return super::parse(required, argv);
440    }
441    T::from_arg_matches_mut(&mut matches).map_err(|err| Failure::Clap(err.format(&mut cmd)))
442}

444fn arg_at<'cmd>(cmd: &'cmd Command, path: &[&str], id: &str) -> Option<&'cmd Arg> {
445    command_at(cmd, path)?.get_arguments().find(|arg| arg.get_id() == id)
446}

448/// Check whether `prompted_arg` conflicts with an arg the user passed or an earlier answer.
449///
450/// Answers go in as default values, which clap never checks for conflicts, so asking here
451/// would let the parse land in a state clap rejects when typed.
452fn blocked(
453    cmd: &Command,
454    matches: &ArgMatches,
455    prompted_arg: &PromptedArg,
456    answers: &[Answer<'_>],
457) -> bool {
458    let (Some(level), Some(level_matches)) = (
459        command_at(cmd, &prompted_arg.subcommand_path),
460        matches_at(matches, &prompted_arg.subcommand_path),
461    ) else {
462        return false;
463    };
464    let Some(arg) = level.get_arguments().find(|arg| arg.get_id() == prompted_arg.attr.arg_id)
465    else {
466        return false;
467    };
468    let answered = |id: &Id| {
469        answers.iter().any(|answer| {
470            answer.prompted_arg.subcommand_path == prompted_arg.subcommand_path
471                && answer.prompted_arg.attr.arg_id == id
472        })
473    };
474    let conflicts = level.get_arg_conflicts_with(arg);
475    level
476        .get_arguments()
477        .filter(|other| other.get_id() != arg.get_id())
478        .filter(|other| answered(other.get_id()) || explicit(level_matches, other.get_id()))
479        .any(|other| {
480            other.is_exclusive_set()
481                || conflicts.iter().any(|conflict| conflict.get_id() == other.get_id())
482                || level.get_arg_conflicts_with(other).iter().any(|c| c.get_id() == arg.get_id())
483                || group_mates(level, arg).any(|mate| mate == other.get_id())
484        })
485}

487/// `arg`'s declared default values, split at its value delimiter.
488/// Without a delimiter, each default is one value. Defaults are set from strings, so any that
489/// is not UTF-8 is skipped.
490fn default_values(arg: &Arg) -> impl Iterator<Item = &str> {
491    let delimiter = arg.get_value_delimiter();
492    arg.get_default_values()
493        .iter()
494        .filter_map(|value| value.to_str())
495        .flat_map(move |value| value.split(move |char| Some(char) == delimiter))
496}

498/// The command at `path` passed through `edit`; unchanged when the path is missing.
499fn with_command_at<F>(cmd: Command, path: &[&str], edit: F) -> Command
500where
501    F: FnOnce(Command) -> Command,
502{
503    match path.split_first() {
504        None => edit(cmd),
505        Some((name, rest)) if cmd.find_subcommand(name).is_some() => {
506            cmd.mut_subcommand(name, |sub| with_command_at(sub, rest, edit))
507        }
508        Some(_) => cmd,
509    }
510}

512fn command_at<'cmd>(cmd: &'cmd Command, path: &[&str]) -> Option<&'cmd Command> {
513    path.iter().try_fold(cmd, |level, name| level.find_subcommand(name))
514}

516/// Check whether the user passed `id`, on the command line or through its env var.
517fn explicit(matches: &ArgMatches, id: &Id) -> bool {
518    matches!(
519        matches.value_source(id.as_str()),
520        Some(ValueSource::CommandLine | ValueSource::EnvVariable)
521    )
522}

524/// The args sharing an exclusive `ArgGroup` with `arg`.
525fn group_mates<'cmd>(level: &'cmd Command, arg: &Arg) -> impl Iterator<Item = &'cmd Id> {
526    level
527        .get_groups()
528        .filter(|group| {
529            !ArgGroup::clone(group).is_multiple() && group.get_args().any(|id| id == arg.get_id())
530        })
531        .flat_map(ArgGroup::get_args)
532}

534/// For a flag, whether a "yes" stores `false`; `None` for an arg that takes a value.
535#[expect(
536    clippy::wildcard_enum_match_arm,
537    reason = "`ArgAction` is #[non_exhaustive]; every action but the two flags takes a value"
538)]
539const fn flag_negation(action: &ArgAction) -> Option<bool> {
540    match action {
541        ArgAction::SetTrue => Some(false),
542        ArgAction::SetFalse => Some(true),
543        _ => None,
544    }
545}

547/// Set each answer as its arg's default values. No values leave the arg with no default, so
548/// an emptied list drops the declared default.
549fn inject<'prompted_arg>(
550    cmd: Command,
551    answers: impl IntoIterator<Item = (&'prompted_arg PromptedArg, &'prompted_arg [String])>,
552) -> Command {
553    answers.into_iter().fold(cmd, |cmd, (prompted_arg, values)| {
554        // A default value is no explicit value to clap, so the rules that require the arg would
555        // still miss it.
556        with_arg_at(
557            unrequire(cmd, prompted_arg),
558            &prompted_arg.subcommand_path,
559            prompted_arg.attr.arg_id,
560            |arg| arg.default_values(values.iter().cloned()),
561        )
562    })
563}

565/// `question` with each `{id}` replaced by arg `id`'s values in `matches`, joined with `, `, and
566/// `{{` and `}}` by single braces. A `{id}` naming no arg stays as written.
567fn interpolate(question: &str, matches: &ArgMatches) -> String {
568    let mut interpolated = String::with_capacity(question.len());
569    let mut rest = question;
570    while let Some(brace) = rest.find(['{', '}']) {
571        let (text, tail) = rest.split_at(brace);
572        interpolated.push_str(text);
573        let escaped = tail.strip_prefix("{{").map(|after| ('{', after));
574        if let Some((brace, after)) =
575            escaped.or_else(|| tail.strip_prefix("}}").map(|after| ('}', after)))
576        {
577            interpolated.push(brace);
578            rest = after;
579            continue;
580        }
581        let placeholder = tail.strip_prefix('{').and_then(|inner| inner.split_once('}'));
582        if let Some((values, after)) = placeholder
583            .and_then(|(id, after)| matches.try_get_raw(id).ok().map(|values| (values, after)))
584        {
585            let values: Vec<_> =
586                values.into_iter().flatten().map(|v| v.to_string_lossy()).collect();
587            interpolated.push_str(&values.join(", "));
588            rest = after;
589            continue;
590        }
591        // A lone brace, or a `{id}` naming no arg: keep the brace as written.
592        let (brace, after) = tail.split_at(1);
593        interpolated.push_str(brace);
594        rest = after;
595    }
596    interpolated.push_str(rest);
597    interpolated
598}

600fn matches_at<'matches>(
601    matches: &'matches ArgMatches,
602    path: &[&str],
603) -> Option<&'matches ArgMatches> {
604    path.iter().try_fold(matches, |level, name| {
605        level.subcommand().filter(|(taken, _)| taken == name).map(|(_, sub)| sub)
606    })
607}

609/// The prompted args on the path the user took that got no value or only their default, with
610/// the default's values, split at the arg's value delimiter as clap splits them. One whose
611/// `default_value_if` fired is left to the arg that fired it: the conditional default beats an
612/// injected answer, so asking would drop the answer.
613fn pending<'prompted_arg>(
614    cmd: &Command,
615    matches: &ArgMatches,
616    prompted_args: &'prompted_arg [PromptedArg],
617) -> Vec<(&'prompted_arg PromptedArg, Vec<String>)> {
618    // Built, so a flag's action has filled in its implicit default (`false` for `SetTrue`).
619    let mut built = cmd.clone();
620    built.build();
621    prompted_args
622        .iter()
623        .filter_map(|prompted_arg| {
624            let arg = arg_at(&built, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)?;
625            let level = matches_at(matches, &prompted_arg.subcommand_path)?;
626            let id = prompted_arg.attr.arg_id;
627            match level.value_source(id) {
628                None if default_values(arg).next().is_none() => Some((prompted_arg, Vec::new())),
629                Some(ValueSource::DefaultValue) => {
630                    let current = raw_values(level, id);
631                    current
632                        .iter()
633                        .map(String::as_str)
634                        .eq(default_values(arg))
635                        .then_some((prompted_arg, current))
636                }
637                None | Some(_) => None,
638            }
639        })
640        .collect()
641}
642
643fn raw_values(matches: &ArgMatches, id: &str) -> Vec<String> {
644    matches
645        .try_get_raw(id)
646        .ok()
647        .flatten()
648        .into_iter()
649        .flatten()
650        .map(|value| value.to_string_lossy().into_owned())
651        .collect()
652}

654/// Check whether `arg` takes many values, as a `Vec<T>` field does.
655fn takes_many(arg: &Arg) -> bool {
656    matches!(arg.get_action(), ArgAction::Append)
657        || arg.get_num_args().is_some_and(|range| range.max_values() > 1)
658}

660/// Make each of `prompted_args` optional, so a missing one does not stop a parse.
661fn relax<'prompted_arg>(
662    cmd: Command,
663    prompted_args: impl IntoIterator<Item = &'prompted_arg PromptedArg>,
664) -> Command {
665    prompted_args.into_iter().fold(cmd, unrequire)
666}

668/// The first line of clap's message, without its `error: ` prefix. clap quotes the rejected
669/// `values`, so a multi-line one is shortened to its first line first, keeping the reason after it.
670fn summary(err: &clap::Error, values: &[String]) -> String {
671    let mut rendered = err.render().to_string();
672    for value in values {
673        if let Some((first_line, _)) = value.split_once('\n') {
674            rendered = rendered.replace(value.as_str(), &format!("{first_line}\u{2026}"));
675        }
676    }
677    let line = rendered.lines().next().unwrap_or_default();
678    line.strip_prefix("error: ").unwrap_or(line).to_owned()
679}

681/// Drop the rules that make `prompted_arg` required: its `required`, its
682/// `required_unless_present`, and the `required` of the groups it belongs to. clap can't clear
683/// `required_unless_present_all` or `required_if_eq`, so those stay.
684fn unrequire(cmd: Command, prompted_arg: &PromptedArg) -> Command {
685    let id = prompted_arg.attr.arg_id;
686    with_command_at(cmd, &prompted_arg.subcommand_path, |level| {
687        if !level.get_arguments().any(|arg| arg.get_id() == id) {
688            return level;
689        }
690        let groups: Vec<Id> = level
691            .get_groups()
692            .filter(|group| group.is_required_set() && group.get_args().any(|arg| arg == id))
693            .map(|group| group.get_id().clone())
694            .collect();
695        let level =
696            level.mut_arg(id, |arg| arg.required(false).required_unless_present(Resettable::Reset));
697        groups
698            .iter()
699            .fold(level, |level, group| level.mut_group(group, |group| group.required(false)))
700    })
701}

703/// Apply `edit` to the arg `id` of the command at `path`; unchanged when either is missing.
704fn with_arg_at<F>(cmd: Command, path: &[&str], id: &str, edit: F) -> Command
705where
706    F: FnOnce(Arg) -> Arg,
707{
708    with_command_at(cmd, path, |level| {
709        if level.get_arguments().any(|arg| arg.get_id() == id) {
710            level.mut_arg(id, edit)
711        } else {
712            level
713        }
714    })
