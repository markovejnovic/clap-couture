53/// How a known editor behaves.
54#[derive(Debug)]
55struct EditorBehavior {
56    /// It writes to the console while open.
57    is_console: bool,
58    /// The flag that makes it wait for the file to be closed, if it returns at once without it.
59    wait_flag: Option<&'static str>,
60}

62/// The editor read from `$VISUAL` and `$EDITOR`.
63#[derive(Debug, Clone)]
64pub(crate) struct Editor {
65    /// The raw command held in $TERM.
66    pub(crate) command: OsString,
67}
68
69#[derive(Debug, PartialEq, Eq)]
70pub(crate) enum EditorError {
71    /// $TERM is configured to be dumb.
72    DumbTerminal,
73    /// The editor is `:`, git's no-op editor, which isn't run.
74    NoOp,
75    /// $TERM could not find the terminal. The member contains the $TERM value.
76    NotFound(OsString),
77    /// $TERM had an unclosed quote.
78    UnclosedQuote(OsString),
79}
80
81impl Editor {
82    /// The behavior of a known editor.
83    fn behavior(&self) -> Option<&'static EditorBehavior> {
84        let program = self.program()?;
85        EDITOR_BEHAVIORS.iter().find(|(name, _)| *name == program).map(|(_, behavior)| behavior)
86    }
87
88    /// The editor string split into words.
89    fn cmd_args(&self) -> impl Iterator<Item = Result<OsString, EditorError>> + '_ {
90        command::split(&self.command).map(|word| {
91            word.map(expand_tilde)
92                .map_err(|_unclosed_quote| EditorError::UnclosedQuote(self.command.clone()))
93        })
94    }
95
96    /// The editor's process for `path`: its words, the wait flag a known editor needs, then the
97    /// file. The program is looked up on
98    /// `PATH` first, with `PATHEXT` on Windows.
99    ///
100    /// Certain editors need special wait flags attached to them, and this will add those if
101    /// necessary.
102    pub(crate) fn command(&self, path: &Path) -> Result<Command, EditorError> {
103        // Git handles $TERM=":" as a no-op, so we'll do the same.
104        if self.command == ":" {
105            return Err(EditorError::NoOp);
106        }
107
108        let mut cmd_args = self.cmd_args();
109        let program = cmd_args.next().transpose()?.unwrap_or_default();
110        let found = which::which(&program)
111            .map_err(|_not_found| EditorError::NotFound(self.command.clone()))?;
112
113        let mut command = Command::new(found);
114        #[cfg(unix)]
115        std::os::unix::process::CommandExt::arg0(&mut command, &program);
116        for word in cmd_args {
117            command.arg(word?);
118        }
119        command.args(self.wait_flag().into_iter().flat_map(|flag| flag.split(' ')));
120        command.arg(path);
121
122        Ok(command)
123    }
124
125    /// Check whether an editor writes to the console or not.
126    pub(crate) fn is_console(&self) -> bool {
127        self.behavior().is_some_and(|behavior| behavior.is_console)
128    }
129
130    /// The program's name, lowercased and without directory or extension, like `code`.
131    fn program(&self) -> Option<OsString> {
132        let program = self.cmd_args().next()?.ok()?;
133        Some(Path::new(&program).file_stem()?.to_ascii_lowercase())
134    }
135
136    /// Look the editor from `$VISUAL` unless the terminal is dumb, then `$EDITOR`, then `vi`
137    /// (`notepad` on Windows).
138    pub(crate) fn read() -> Result<Self, EditorError> {
139        // This is git's logic. An unset TERM is dumb, except on Windows.
140        let dumb_terminal = Term::read().map_or(cfg!(not(windows)), |term| term.is_dumb());
141
142        let set = |name: &str| std::env::var_os(name).filter(|value| !value.is_empty());
143
144        let visual = if dumb_terminal { None } else { set("VISUAL") };
145
146        let command = match visual.or_else(|| set("EDITOR")) {
147            Some(command) => command,
148            None if dumb_terminal => return Err(EditorError::DumbTerminal),
149            None => OsString::from(if cfg!(windows) { "notepad" } else { "vi" }),
150        };
151        Ok(Self { command })
152    }
153
154    /// Write the file `options` describe and open the editor on it, without waiting for it.
155    pub(crate) fn spawn(&self, options: EditorSpawnOptions<'_>) -> Result<Opened, SpawnError> {
156        Opened::start(self, options)
157    }
158
159    /// The flag a known editor needs to wait for the file to be closed, unless already given.
160    fn wait_flag(&self) -> Option<&'static str> {
161        let flag = self.behavior()?.wait_flag?;
162        let first_flag = flag.split(' ').next().unwrap_or(flag);
163        let given = self
164            .cmd_args()
165            .skip(1)
166            .any(|word| word.is_ok_and(|word| word == first_flag || word == "--wait"));
167        (!given).then_some(flag)
168    }
169}

171/// How to open the editor, built with [`EditorSpawnOptions::builder`].
172#[derive(Debug)]
173pub(crate) struct EditorSpawnOptions<'options> {
174    contents: &'options str,
175    hint: Option<&'options str>,
176    place: FilePlace,
177    prompt: Option<&'options str>,
178    prompt_marker: PromptMarker,
179}

181impl<'options> EditorSpawnOptions<'options> {
182    /// Options for a file in a temporary directory, empty and without a prompt.
183    pub(crate) fn builder() -> EditorSpawnOptionsBuilder<'options> {
184        EditorSpawnOptionsBuilder(Self::default())
185    }

188impl Default for EditorSpawnOptions<'_> {

210    /// The text the file starts with. `.contents("Fixed the login bug.")` opens:
211    ///
212    /// ```text
213    /// Fixed the login bug.
214    /// ```
215    pub(crate) const fn contents(mut self, contents: &'options str) -> Self {
216        self.0.contents = contents;
217        self
218    }
219
220    /// A line printed to the terminal while the editor is open, erased once it closes; empty
221    /// prints nothing. `.hint("hint: Waiting for your editor to close the file...")` shows:
222    ///
223    /// ```text
224    /// $ deploy
225    /// hint: Waiting for your editor to close the file...
226    /// ```
227    pub(crate) const fn hint(mut self, hint: &'options str) -> Self {
228        self.0.hint = Some(hint);
229        self
230    }

232    /// Put the file at `path`, and leave it there.
233    #[cfg_attr(
234        not(all(test, unix)),
235        expect(dead_code, reason = "prompts only edit temporary files")
236    )]
237    pub(crate) fn path(mut self, path: impl Into<PathBuf>) -> Self {
238        self.0.place = FilePlace::Path(path.into());
239        self
240    }
241
242    /// A question below the contents, removed from what the editor saves. With the contents
243    /// above, `.prompt("Release notes\n\nLeave the file empty to cancel.")` opens:
244    ///
245    /// ```text
246    /// Fixed the login bug.
247    ///
248    /// #| Release notes
249    /// #|
250    /// #| Leave the file empty to cancel.
251    /// ```
252    pub(crate) const fn prompt(mut self, prompt: &'options str) -> Self {
253        self.0.prompt = Some(prompt);
254        self
255    }
256
257    /// What each prompt line starts with, `#|` by default. With the contents and prompt above,
258    /// `.prompt_marker("//")` opens:
259    ///
260    /// ```text
261    /// Fixed the login bug.
262    ///
263    /// // Release notes
264    /// //
265    /// // Leave the file empty to cancel.
266    /// ```
267    #[cfg_attr(not(all(test, unix)), expect(dead_code, reason = "prompts keep the default marker"))]
268    pub(crate) fn prompt_marker(mut self, prompt_marker: PromptMarker) -> Self {
269        self.0.prompt_marker = prompt_marker;
270        self
271    }
272
273    /// Put the file, named `name`, in a temporary directory removed with [`Opened`].
274    pub(crate) fn tmpfile(mut self, name: impl Into<PathBuf>) -> Self {
275        self.0.place = FilePlace::TempFile(name.into());
276        self
277    }

287/// The file name when none is given: the program's, like `git.txt`.
288fn default_file_name() -> PathBuf {
289    let mut name = crate::os::process_name().map_or_else(
290        || OsString::from("edit"),
291        |name| if let Cow::Owned(safe) = crate::fs::file_safe(&name) { safe } else { name },
292    );
293    name.push(".txt");
294    name.into()
295}

297fn expand_tilde(word: OsString) -> OsString {
298    // The split has already dropped any quotes, so a quoted `'~'` expands too.
299    let Ok(rest) = Path::new(&word).strip_prefix("~") else { return word };
300    let Some(home) = std::env::home_dir() else { return word };
301    if rest.as_os_str().is_empty() { home } else { home.join(rest) }.into_os_string()
302}

365        let home = std::env::home_dir().unwrap_or_default();
366        let expected: Vec<OsString> = if expanded.is_empty() {
367            command.split(' ').map(OsString::from).collect()
368        } else {
369            expanded
370                .iter()
371                .map(|word| match word.strip_prefix('~') {
372                    Some("") => home.clone().into_os_string(),
373                    Some(rest) => home.join(rest.trim_start_matches('/')).into_os_string(),

375                })
376                .collect()
377        };
378        let words: Result<Vec<_>, _> = editor(command).cmd_args().collect();
379        assert_eq!(words, Ok(expected));
380    }
