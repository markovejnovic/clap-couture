8/// Prompts drawn by cliclack, in its clack style.
9#[derive(Clone, Copy, Debug, Default)]
10pub struct Cliclack;
11
12impl Backend for Cliclack {
13    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
14        Theme::from(prompt.styles).scoped(|| {
15            ::cliclack::confirm(prompt.question).initial_value(prompt.default).interact()
16        })
17    }
18
19    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
20        Theme::from(prompt.styles).scoped(|| {
21            prompt
22                .options
23                .iter()
24                .fold(::cliclack::multiselect(prompt.question), |select, option| {
25                    let hint = option.get_help().map(ToString::to_string).unwrap_or_default();
26                    select.item(option.get_name().to_owned(), option.get_name(), hint)
27                })
28                .initial_values(prompt.defaults.to_vec())
29                .required(!prompt.optional)
30                .interact()
31        })
32    }
33
34    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
35        Theme::from(styles).scoped(|| ::cliclack::log::error(error))
36    }
37
38    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
39        Theme::from(prompt.styles).scoped(|| {
40            let select = prompt.options.iter().fold(
41                ::cliclack::select(prompt.question),
42                |select, option| {
43                    let hint = option.get_help().map(ToString::to_string).unwrap_or_default();
44                    select.item(Some(option.get_name().to_owned()), option.get_name(), hint)
45                },
46            );
47            let mut select =
48                if prompt.optional { select.item(None, prompt.none_option, "") } else { select };
49            match prompt.default {
50                Some(default) => select.initial_value(Some(default.to_owned())).interact(),
51                None => select.interact(),
52            }
53        })
54    }
55
56    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
57        let answer: String = Theme::from(prompt.styles).scoped(|| {
58            let input = ::cliclack::input(prompt.question).required(!prompt.optional);
59            // TODO(markovejnovic): cliclack's `Input` can't start out holding editable text (no
60            // equivalent of inquire's `with_initial_value`), so `initial` is passed as a default
61            // instead. The cost: a list prompt pre-filled from the arg's defaults can't be
62            // finished early, as Enter on it keeps the default instead of ending the list. Once
63            // cliclack supports starting text, pass `initial` that way and show the hint even
64            // when there's a default.
65            //
66            // When there is a default, the hint is left off: cliclack shows the hint where it
67            // would show "(default)", hiding it, and Enter wouldn't finish the list anyway.
68            let mut input = match (prompt.default.or(prompt.initial), prompt.hint) {
69                (Some(default), _) => input.default_input(default),
70                (None, Some(hint)) => input.placeholder(hint),
71                (None, None) => input,
72            };
73            input.interact()
74        })?;
75        Ok(Some(answer).filter(|answer| !answer.is_empty()))
76    }
77}

79/// The command's styles on cliclack's input line.
80///
81/// By default, [`::cliclack`] uses global theming, which is somewhat inconvenient for the use-case
82/// of following clap's theme. The [`Theme::scoped`] function allows you to run a function with a
83/// temporarily globally set theme.
84struct Theme {
85    input: console::Style,
86    placeholder: console::Style,
87}
88
89impl Theme {
90    /// Run `interact` with this as cliclack's theme.
91    fn scoped<T>(self, interact: impl FnOnce() -> io::Result<T>) -> Result<T, PromptError> {
92        ::cliclack::set_theme(self);
93        let answer = interact();
94        ::cliclack::reset_theme();
95        answer.map_err(PromptError::from)
96    }
97}
98
99impl From<CommandStyles<'_>> for Theme {
100    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
101        Self {
102            input: console::Style::from(ClapStyle(styles.get_literal())),
103            placeholder: console::Style::from(ClapStyle(styles.get_placeholder())),
104        }
105    }
106}
107
108impl ::cliclack::Theme for Theme {
109    fn input_style(&self, _state: &::cliclack::ThemeState) -> console::Style {
110        self.input.clone()
111    }
112
113    fn placeholder_style(&self, _state: &::cliclack::ThemeState) -> console::Style {
114        self.placeholder.clone()
115    }
116}
