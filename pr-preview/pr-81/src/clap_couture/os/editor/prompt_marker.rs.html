7/// What every header line starts with. Markdown headings take `# ` and shell comments `# `, so
8/// a user's own line almost never starts with `#|`.
9const HEADER: &str = "#|";
10
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub(crate) struct PromptMarker(String);
13
14impl PromptMarker {
15    /// Where the header written for `prompt` starts in `lines`: at its last copy of the header's
16    /// first line, else, when the user edited that line, at the marked lines ending the file.
17    fn header_start(&self, lines: &[&str], prompt: &str) -> usize {
18        let prompt = strip_str(prompt).to_string();
19        let first = format!("{} {}", self.0, prompt.lines().next().unwrap_or_default());
20        lines.iter().rposition(|line| line.trim() == first.trim()).unwrap_or_else(|| {
21            let is_answer = |line: &&str| !line.trim().is_empty() && !self.marks(line);
22            lines.iter().rposition(is_answer).map_or(0, |last| last.saturating_add(1))
23        })
24    }
25
26    /// Whether `line` starts with the marker, past any indentation an editor added.
27    fn marks(&self, line: &str) -> bool {
28        line.trim_start().starts_with(self.0.as_str())
29    }
30
31    pub(crate) fn new(marker: impl Into<String>) -> Self {
32        Self(marker.into())
33    }
34
35    /// The answer in `file`, opened with `prompt`: the header lines and the blank lines around the
36    /// rest dropped, each line left ending in a newline as in any text file. `None` when nothing is
37    /// left.
38    pub(crate) fn strip(&self, file: &str, prompt: Option<&str>) -> Option<String> {
39        let is_blank = |line: &&str| line.trim().is_empty();
40        let lines: Vec<&str> = file.lines().collect();
41        let header_start = prompt.map_or(lines.len(), |prompt| self.header_start(&lines, prompt));
42        let (answer, header) = lines.split_at(header_start);
43        // Lines typed into the header, or unmarked there, are the user's too.
44        let typed_in_header = header.iter().filter(|line| !self.marks(line));
45        let mut kept: Vec<&str> =
46            answer.iter().chain(typed_in_header).copied().skip_while(is_blank).collect();
47        while kept.last().is_some_and(is_blank) {
48            kept.pop();
49        }
50        (!kept.is_empty()).then(|| kept.iter().flat_map(|line| [*line, "\n"]).collect())
51    }
52
53    /// The file the editor opens: the answer so far, a blank line, then the header.
54    pub(crate) fn write(
55        &self,
56        out: &mut impl io::Write,
57        contents: &str,
58        prompt: Option<&str>,
59    ) -> io::Result<()> {
60        if !contents.is_empty() {
61            write!(out, "{contents}")?;
62            if !contents.ends_with('\n') {
63                writeln!(out)?;
64            }
65        }
66        if let Some(prompt) = prompt {
67            writeln!(out)?;
68            // The marker, then a space before the text.
69            // A styled question's escapes would show in the file as text.
70            let prompt = strip_str(prompt).to_string();
71            strings::prefix_lines_into(out, &format!("{} ", self.0), [prompt.as_str()])?;
72        }
73        Ok(())
74    }
75}
76
77impl Default for PromptMarker {
78    fn default() -> Self {
79        Self::new(HEADER)
80    }
81}

89    const PROMPT: &str = "Release notes\n\nLeave the file empty to cancel.";
90
91    fn rendered(contents: &str, prompt: Option<&str>) -> String {
92        let mut file = Vec::new();
93        let written = PromptMarker::default().write(&mut file, contents, prompt);
94        assert!(written.is_ok(), "writing to a `Vec` can't fail");
95        String::from_utf8(file).unwrap_or_default()
96    }
97
98    #[test]
99    fn render_puts_the_answer_before_the_prompt() {
100        let file = rendered("Fixed it.", Some(PROMPT));
101        assert_eq!(file, "Fixed it.\n\n#| Release notes\n#|\n#| Leave the file empty to cancel.\n");
102    }
103
104    #[test]
105    fn render_drops_a_styled_prompts_escapes() {
106        let file = rendered("", Some("Release notes, in \u{1b}[1mmarkdown\u{1b}[0m"));
107        assert_eq!(file, "\n#| Release notes, in markdown\n");
108    }
109
110    #[test]
111    fn strip_finds_a_styled_prompts_header() {
112        let prompt = "\u{1b}[1mRelease\u{1b}[0m notes";
113        let file = format!("#| Release notes\nbody\n{}", rendered("", Some(prompt)));
114        assert_eq!(
115            PromptMarker::default().strip(&file, Some(prompt)).as_deref(),
116            Some("#| Release notes\nbody\n")
117        );
118    }
119
120    #[test]
121    fn render_without_a_prompt_is_the_contents() {
122        assert_eq!(rendered("Fixed it.", None), "Fixed it.\n");
