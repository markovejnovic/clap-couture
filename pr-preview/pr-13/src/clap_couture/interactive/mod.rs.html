1//! Ask on the terminal for prompted args the user left out.
2//!
3//! Mark a field with `#[couture(prompt)]` to ask its help text, or `#[couture(prompt = "...")]`
4//! to ask something else. When the user leaves the arg out and stdin and stderr are terminals,
5//! [`CoutureParser`](crate::CoutureParser)'s entry points ask for it, then parse the answer as if
6//! it had been typed. Without a terminal, parsing behaves exactly like clap.
7//!
8//! ```
9//! use clap::Parser;
10//! use clap_couture::{Couture, CoutureParser};
11//!
12//! #[derive(Parser, Couture)]
13//! struct Cli {
14//!     /// Region to deploy to
15//!     #[arg(long)]
16//!     #[couture(prompt)]
17//!     region: String,
18//! }
19//!
20//! let cli = Cli::couture_parse_from(["deploy", "--region", "eu"]);
21//! assert_eq!(cli.region, "eu");
22//! ```
23//!
24//! A prompted arg that has a default is asked with the default pre-filled. An optional arg answered
25//! with nothing stays unset. An arg that takes many values, like a `Vec<T>`, is asked as a
26//! multi-select when it has possible values, and otherwise one value at a time until an empty
27//! answer. An arg passed on the command line or through its env var is never
28//! asked, and neither is one that conflicts with a passed arg or an earlier answer. Answers are
29//! not re-checked against `requires`, `required_if_eq` and similar rules between args.
30//!
31//! `#[couture(confirm)]` on a `bool` flag the user left out asks whether to go ahead, after every
32//! other prompt, with each `{id}` in the question replaced by arg `id`'s value. A yes sets the
33//! flag; a no fails the parse. Without a terminal, the flag must be passed, as if it were required.

55/// Why a prompt produced no answer.
56#[derive(Debug, thiserror::Error)]
57#[non_exhaustive]
58pub enum PromptError {
59    #[error("prompt cancelled")]
60    Cancelled,
61    #[error(transparent)]
62    Io(io::Error),
63    #[error("not a terminal")]
64    NotATerminal,
65}
66
67// `Interrupted` is the user backing out, `NotConnected` a missing terminal.
68impl From<io::Error> for PromptError {
69    #[expect(
70        clippy::wildcard_enum_match_arm,
71        reason = "`io::ErrorKind` is #[non_exhaustive]; every other kind is a plain I/O failure"
72    )]
73    fn from(err: io::Error) -> Self {
74        match err.kind() {
75            io::ErrorKind::Interrupted => Self::Cancelled,
76            io::ErrorKind::NotConnected => Self::NotATerminal,
77            _ => Self::Io(err),
78        }
79    }
80}

82/// A yes/no question.
83#[derive(Debug)]
84#[non_exhaustive]
85pub struct ConfirmPrompt<'prompt> {
86    /// The answer Enter accepts.
87    pub default: bool,
88    /// What to ask.
89    pub question: &'prompt str,
90    /// The command's styles, for backends that theme themselves.
91    pub styles: CommandStyles<'prompt>,
92}

94/// A pick of any number of an arg's possible values.
95#[derive(Debug)]
96#[non_exhaustive]
97pub struct MultiSelectPrompt<'prompt> {
98    /// The names of the options picked to begin with.
99    pub defaults: &'prompt [String],
100    /// Picking none of the options is allowed.
101    pub optional: bool,
102    /// The visible possible values, in declared order.
103    pub options: &'prompt [PossibleValue],
104    /// What to ask.
105    pub question: &'prompt str,
106    /// What to tell the user when they select nothing, unless the prompt is optional.
107    pub selection_required: &'prompt str,
108    /// The command's styles, for backends that theme themselves.
109    pub styles: CommandStyles<'prompt>,
110}

112/// A pick among an arg's possible values.
113#[derive(Debug)]
114#[non_exhaustive]
115pub struct SelectPrompt<'prompt> {
116    /// The name of the option Enter accepts, if any.
117    pub default: Option<&'prompt str>,
118    /// The extra option an optional prompt lists to pick none of the options.

120    /// Picking none of the options is allowed, and leaves the arg unset.
121    pub optional: bool,
122    /// The visible possible values, in declared order.
123    pub options: &'prompt [PossibleValue],
124    /// What to ask.
125    pub question: &'prompt str,
126    /// The command's styles, for backends that theme themselves.
127    pub styles: CommandStyles<'prompt>,

130/// A free-text question.
131#[derive(Debug)]
132#[non_exhaustive]
133pub struct TextPrompt<'prompt> {
134    /// The answer Enter accepts, if any.
135    pub default: Option<&'prompt str>,
136    /// What to show beside the question, if anything: for one more value of a multi-value arg,
137    /// that an empty answer finishes the list.
138    pub hint: Option<&'prompt str>,
139    /// The text the answer starts out as, for the user to keep, edit or clear. Unlike
140    /// [`default`](Self::default), clearing it leaves the answer empty.
141    pub initial: Option<&'prompt str>,
142    /// An empty answer is allowed, and leaves the arg unset.
143    pub optional: bool,
144    /// What to ask.
145    pub question: &'prompt str,
146    /// The command's styles, for backends that theme themselves.
147    pub styles: CommandStyles<'prompt>,
148}
