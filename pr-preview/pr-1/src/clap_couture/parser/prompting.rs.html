1//! Parse, ask for the prompted args that got no value, then parse again with the answers.
2//!
3//! Every answer goes back into the command as its arg's `default_values`, so clap converts and
4//! validates it exactly like a typed value. Each answer is checked with a trial parse as soon as
5//! it is given: the previous trial passed, so a failure can only come from the new answer, which
6//! is then asked again with clap's message. For an arg asked one value at a time, each value is
7//! an answer of its own, checked on top of the values before it.
8//!
9//! Confirmations come last, so their questions can name the answers. A declined one fails the
10//! parse; without a terminal to ask on, one left unanswered fails it as a missing required arg.
11//!
12//! A secret's answer goes into a vault instead, and clap gets a placeholder that the secret's
13//! value parser swaps back. It skips the trial parse: that parser takes any string, so there is
14//! nothing to ask again.
15

41/// A prompted arg's accepted answer.
42struct Answer<'prompted_arg> {
43    /// What clap gets as the arg's defaults: the answer itself, or a secret's placeholder. One
44    /// value, or any number for an arg that takes many.
45    injected_values: Vec<String>,
46    prompted_arg: &'prompted_arg PromptedArg,
47}

49/// A prompted arg about to be asked in the editor, and what is known about it.
50#[cfg(feature = "interactive-editor")]
51struct Asked<'ask> {
52    arg: &'ask Arg,
53    /// Its default's value, pre-filled.
54    current: Option<&'ask str>,
55    /// Leaving it unset passes clap, so an empty answer is allowed.
56    optional: bool,
57    prompted_arg: &'ask PromptedArg,
58    question: &'ask str,
59}

61/// An answer clap turned down, and why; an editor prompt reopens with it.
62#[cfg(feature = "interactive-editor")]
63struct Rejected {
64    error: String,
65    value: String,
66}

68/// One parse: the command, its argv, its prompted args, who to ask and where secrets go.
69struct Session<'run> {
70    argv: &'run [OsString],
71    backend: &'run dyn Backend,
72    cmd: &'run Command,
73    messages: &'run Messages,
74    prompted_args: &'run [PromptedArg],
75    /// Holds the secret answers; `cmd` resolves their placeholders from it.
76    vault: &'run Vault,
77}
78
79impl Session<'_> {
80    /// Ask for `prompted_arg` until clap accepts the answer; `None` when an optional single-value
81    /// one is left empty. `current` is the arg's default, split into its values. `later` are the
82    /// prompted args still to be asked after it. A secret is asked once and comes back as its vault
83    /// placeholder.
84    fn answer(
85        &self,
86        prompted_arg: &PromptedArg,
87        current: &[String],
88        answers: &[Answer<'_>],
89        later: &[&PromptedArg],
90    ) -> Result<Option<Vec<String>>, PromptError> {
91        let Some(arg) = arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
92        else {
93            return Ok(None);
94        };
95        let optional = !self.required(answers, later);
96        let question = self.question(prompted_arg);
97        if prompted_arg.attr.secret.is_some() {
98            // clap has nothing to reject in a string, so a secret skips the trial parse; the vault
99            // keeps it and clap gets a placeholder. `current` goes unused: the derive refuses a
100            // default on a secret, and showing one would echo it.
101            let secret = self.backend.secret(&SecretPrompt {
102                optional,
103                question: &question,
104                styles: CommandStyles(self.cmd.get_styles()),
105            })?;
106            return Ok(secret.map(|secret| vec![self.vault.store(secret)]));
107        }
108        // A flag has nothing to edit, so one that slipped past the derive asks as a flag.
109        #[cfg(feature = "interactive-editor")]
110        if let Some(editor) = prompted_arg
111            .attr
112            .editor
113            .as_ref()
114            .filter(|_| arg.get_action().takes_values() && !takes_many(arg))
115        {
116            let asked = Asked {
117                arg,
118                current: current.first().map(String::as_str),
119                optional,
120                prompted_arg,
121                question: &question,
122            };
123            return self.in_editor(&asked, editor, answers);
124        }
125        let options: Vec<PossibleValue> =
126            arg.get_possible_values().into_iter().filter(|value| !value.is_hide_set()).collect();
127        if !takes_many(arg) {
128            return self.until_accepted(answers, prompted_arg, || {
129                let value = self.ask(arg, &question, &options, current.first(), optional)?;
130                Ok(value.map(|value| vec![value]))
131            });
132        }
133        if options.is_empty() {
134            return self.each_value(&question, prompted_arg, current, answers, optional).map(Some);
135        }
136        self.until_accepted(answers, prompted_arg, || {
137            self.backend
138                .multi_select(&MultiSelectPrompt {
139                    defaults: current,
140                    optional,
141                    options: &options,
142                    question: &question,
143                    selection_required: &self.messages.selection_required.get(),
144                    styles: CommandStyles(self.cmd.get_styles()),
145                })
146                .map(Some)
147        })
148    }
149
150    fn ask(
151        &self,
152        arg: &Arg,
153        question: &str,
154        options: &[PossibleValue],
155        current: Option<&String>,
156        optional: bool,
157    ) -> Result<Option<String>, PromptError> {
158        let styles = CommandStyles(self.cmd.get_styles());
159        if let Some(negate) = flag_negation(arg.get_action()) {
160            let set = self.backend.confirm(&ConfirmPrompt { default: false, question, styles })?;
161            return Ok(Some((set != negate).to_string()));
162        }
163        if options.is_empty() {
164            return self.backend.text(&TextPrompt {
165                default: current.map(String::as_str),
166                hint: None,
167                initial: None,
168                optional,
169                question,
170                styles,
171            });
172        }
173        self.backend.select(&SelectPrompt {
174            default: current.map(String::as_str),
175            none_option: &self.messages.none.get(),
176            optional,
177            options,
178            question,
179            styles,
180        })
181    }
182
183    /// Ask in the user's editor, starting from the rejected answer if there is one. An empty
184    /// answer to a required arg aborts, as git aborts on an empty commit message, and so does one
185    /// to an arg with a default, which an empty answer would quietly bring back.
186    #[cfg(feature = "interactive-editor")]
187    fn ask_in_editor(
188        &self,
189        asked: &Asked<'_>,
190        editor: &EditorAttr,
191        rejected: Option<&Rejected>,
192    ) -> Result<Option<String>, PromptError> {
193        let Asked { arg, current, optional: _, prompted_arg, question } = *asked;
194        let styles = CommandStyles(self.cmd.get_styles());
195        let optional = asked.optional && arg.get_default_values().is_empty();
196        let instructions = if optional {
197            self.messages.editor_instructions_optional.get()
198        } else {
199            self.messages.editor_instructions.get()
200        };
201        let answer = self.backend.editor(&EditorPrompt {
202            default: rejected.map_or(current, |rejected| Some(rejected.value.as_str())),
203            error: rejected.map(|rejected| rejected.error.as_str()),
204            extension: editor.extension,
205            file_stem: &iter::once(self.cmd.get_name())
206                .chain(prompted_arg.subcommand_path.iter().copied())
207                .chain(iter::once(prompted_arg.attr.arg_id))
208                .collect::<Vec<_>>()
209                .join("-"),
210            instructions: &instructions,
211            not_utf8: &self.messages.editor_not_utf8.get(),
212            optional,
213            question,
214            returned_early: &self.messages.editor_returned_early.get(),
215            styles,
216            waiting: &self.messages.editor_waiting.get(),
217        })?;
218        if answer.is_none() && !optional {
219            self.backend.report(&self.messages.editor_empty.get(), styles)?;
220            return Err(PromptError::Cancelled);
221        }
222        Ok(answer)
223    }
224
225    /// Ask whether to go ahead, with each `{id}` in the question filled in from the args as
226    /// passed and answered so far.
227    fn confirm(
228        &self,
229        prompted_arg: &PromptedArg,
230        answers: &[Answer<'_>],
231    ) -> Result<bool, PromptError> {
232        let question = self.question(prompted_arg);
233        let answered = inject(
234            relax(self.cmd.clone(), self.prompted_args),
235            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice())),
236        )
237        .try_get_matches_from(self.argv);
238        let question = match answered
239            .as_ref()
240            .ok()
241            .and_then(|answered| matches_at(answered, &prompted_arg.subcommand_path))
242        {
243            Some(level) => interpolate(&question, level),
244            None => question,
245        };
246        self.backend.confirm(&ConfirmPrompt {
247            default: false,
248            question: &question,
249            styles: CommandStyles(self.cmd.get_styles()),
250        })
251    }
252
253    /// Ask for one value at a time until an empty answer, each starting out as the default's
254    /// value at its position. Only a required arg's first value may not be empty.
255    ///
256    /// The default goes in as initial text rather than as the answer Enter accepts, so clearing it
257    /// still finishes the list.
258    fn each_value(
259        &self,
260        question: &str,
261        prompted_arg: &PromptedArg,
262        current: &[String],
263        answers: &[Answer<'_>],
264        optional: bool,
265    ) -> Result<Vec<String>, PromptError> {
266        let finish_hint = self.messages.finish_hint.get();
267        let mut values = Vec::new();
268        loop {
269            let optional = !values.is_empty() || optional;
270            let Some(value) = self.backend.text(&TextPrompt {
271                default: None,
272                hint: optional.then_some(&*finish_hint),
273                initial: current.get(values.len()).map(String::as_str),
274                optional,
275                question,
276                styles: CommandStyles(self.cmd.get_styles()),
277            })?
278            else {
279                return Ok(values);
280            };
281            values.push(value);
282            if let Err(err) = self.trial(answers, prompted_arg, &values) {
283                self.backend
284                    .report(&summary(&err, &values), CommandStyles(self.cmd.get_styles()))?;
285                values.pop();
286            }
287        }
288    }
289
290    /// Ask in the editor until clap accepts the answer, reopening with the rejected one.
291    #[cfg(feature = "interactive-editor")]
292    fn in_editor(
293        &self,
294        asked: &Asked<'_>,
295        editor: &EditorAttr,
296        answers: &[Answer<'_>],
297    ) -> Result<Option<Vec<String>>, PromptError> {
298        let mut rejected = None;
299        loop {
300            let Some(value) = self.ask_in_editor(asked, editor, rejected.as_ref())? else {
301                return Ok(None);
302            };
303            let values = vec![value];
304            match self.trial(answers, asked.prompted_arg, &values) {
305                Ok(()) => return Ok(Some(values)),
306                Err(err) => {
307                    let error = summary(&err, &values);
308                    self.backend.report(&error, CommandStyles(self.cmd.get_styles()))?;
309                    rejected = values.into_iter().next().map(|value| Rejected { error, value });
310                }
311            }
312        }
313    }
314
315    /// The `= ...` override, else the arg's help text, else its id.
316    fn question(&self, prompted_arg: &PromptedArg) -> String {
317        prompted_arg.attr.question.as_ref().map_or_else(
318            || {
319                arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
320                    .and_then(Arg::get_help)
321                    .map_or_else(|| prompted_arg.attr.arg_id.to_owned(), ToString::to_string)
322            },
323            |question| question.get().into_owned(),
324        )
325    }
326
327    /// Check whether leaving the arg being asked unset fails the parse: it is `required`, or
328    /// required through `required_unless_present` or a required group. The accepted answers go
329    /// in, and only the args still to be asked, `later`, stay relaxed, so a group another of them
330    /// can still satisfy doesn't make this one required.
331    fn required(&self, answers: &[Answer<'_>], later: &[&PromptedArg]) -> bool {
332        let accepted =
333            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice()));
334        let cmd = inject(relax(self.cmd.clone(), later.iter().copied()), accepted);
335        cmd.try_get_matches_from(self.argv)
336            .is_err_and(|err| err.kind() == ErrorKind::MissingRequiredArgument)
337    }
338
339    /// Parse with `candidate` as `prompted_arg`'s answer on top of the accepted ones.
340    fn trial(
341        &self,
342        answers: &[Answer<'_>],
343        prompted_arg: &PromptedArg,
344        candidate: &[String],
345    ) -> Result<(), clap::Error> {
346        let accepted =
347            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice()));
348        let cmd = inject(
349            relax(self.cmd.clone(), self.prompted_args.iter()),
350            accepted.chain(iter::once((prompted_arg, candidate))),
351        );
352        cmd.try_get_matches_from(self.argv).map(drop)
353    }
354
355    /// Call `ask` until clap accepts its answer, showing clap's complaint before each retry.
356    fn until_accepted(
357        &self,
358        answers: &[Answer<'_>],
359        prompted_arg: &PromptedArg,
360        mut ask: impl FnMut() -> Result<Option<Vec<String>>, PromptError>,
361    ) -> Result<Option<Vec<String>>, PromptError> {
362        loop {
363            let Some(values) = ask()? else { return Ok(None) };
364            match self.trial(answers, prompted_arg, &values) {
365                Ok(()) => return Ok(Some(values)),
366                Err(err) => {
367                    let error = summary(&err, &values);
368                    self.backend.report(&error, CommandStyles(self.cmd.get_styles()))?;
369                }
370            }
371        }
372    }
373}

375/// Parse `argv` against `cmd`, asking `backend` for the prompted args in `prompted_arg_tree` that
376/// got no value.
377pub(super) fn parse<T>(
378    cmd: Command,
379    prompted_arg_tree: &'static PromptedArgTree,
380    messages: &Messages,
381    argv: &[OsString],
382    backend: &dyn Backend,
383) -> Result<T, Failure>
384where
385    T: FromArgMatches,
386{
387    let prompted_args = prompted_arg_tree.prompted_args();
388    if prompted_args.is_empty() || !backend.is_available() {
389        return parse_unprompted(cmd, &prompted_args, argv);
390    }
391    // Re-guarded with the vault this parse's answers go into; trial and final parses share it.
392    let vault = Vault::default();
393    let cmd = secrets::guard(cmd, &prompted_args, &vault, &messages.environment.get());
394    // Help, usage and errors come from the command the user sees, not the relaxed one.
395    let Ok(matches) = relax(cmd.clone(), prompted_args.iter()).try_get_matches_from(argv) else {
396        return super::parse(cmd, argv);
397    };
398
399    let session = Session {
400        argv,
401        backend,
402        cmd: &cmd,
403        messages,
404        prompted_args: &prompted_args,
405        vault: &vault,
406    };
407    let mut answers = Vec::new();
408    let mut unanswered = pending(&cmd, &matches, &prompted_args);
409    // Stable, so values and confirmations each keep declaration order.
410    unanswered
411        .sort_by_key(|(prompted_arg, _)| prompted_arg.attr.purpose == PromptPurpose::Confirmation);
412    for (index, (prompted_arg, current)) in unanswered.iter().enumerate() {
413        if blocked(&cmd, &matches, prompted_arg, &answers) {
414            continue;
415        }
416        let later: Vec<&PromptedArg> =
417            unanswered.iter().skip(index.saturating_add(1)).map(|(later, _)| *later).collect();
418        let answer = match prompted_arg.attr.purpose {
419            PromptPurpose::Confirmation => match session.confirm(prompted_arg, &answers) {
420                Ok(false) => return Err(Failure::Declined),
421                confirmed => confirmed.map(|_| Some(Vec::from(["true".to_owned()]))),
422            },
423            PromptPurpose::Value => session.answer(prompted_arg, current, &answers, &later),
424        };
425        match answer {
426            Ok(Some(injected_values)) => answers.push(Answer { injected_values, prompted_arg }),
427            Ok(None) => {}
428            // Backing out counts as Ctrl-C.
429            Err(PromptError::Cancelled) => return Err(Failure::Cancelled(Signal::Interrupt)),
430            Err(PromptError::Signalled(signal)) => return Err(Failure::Cancelled(signal)),
431            Err(PromptError::Io(err)) => return Err(Failure::Io(err)),
432            Err(PromptError::NotATerminal) => return parse_unprompted(cmd, &prompted_args, argv),
433        }
434    }
435    let cmd_with_answers = inject(
436        cmd,
437        answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice())),
438    );
439    super::parse(cmd_with_answers, argv)
440}

442/// Parse as clap would, without asking. A confirmation left unanswered fails the parse with
443/// clap's error for a missing required arg, so only passing its flag gets past it.
444fn parse_unprompted<T>(
445    mut cmd: Command,
446    prompted_args: &[PromptedArg],
447    argv: &[OsString],
448) -> Result<T, Failure>
449where
450    T: FromArgMatches,
451{
452    let confirmations: Vec<_> = prompted_args
453        .iter()
454        .filter(|prompted_arg| prompted_arg.attr.purpose == PromptPurpose::Confirmation)
455        .collect();
456    if confirmations.is_empty() {
457        return super::parse(cmd, argv);
458    }
459    // Kept unbuilt: requiring an arg on the built command makes clap show help instead.
460    let declared = cmd.clone();
461    let mut matches = cmd.try_get_matches_from_mut(argv).map_err(Failure::Clap)?;
462    let unconfirmed = confirmations.iter().any(|confirmation| {
463        let id = confirmation.attr.arg_id;
464        let passed = matches_at(&matches, &confirmation.subcommand_path)
465            .is_none_or(|level| explicit(level, &Id::from(id)));
466        !passed && !blocked(&cmd, &matches, confirmation, &[])
467    });
468    if unconfirmed {
469        // Parsed again only to fail: clap words the error, and the help stays as declared.
470        let required = confirmations.iter().fold(declared, |cmd, confirmation| {
471            with_arg_at(cmd, &confirmation.subcommand_path, confirmation.attr.arg_id, |arg| {
472                arg.required(true)
473            })
474        });
475        return super::parse(required, argv);
476    }
477    T::from_arg_matches_mut(&mut matches).map_err(|err| Failure::Clap(err.format(&mut cmd)))
478}

480fn arg_at<'cmd>(cmd: &'cmd Command, path: &[&str], id: &str) -> Option<&'cmd Arg> {
481    command_at(cmd, path)?.get_arguments().find(|arg| arg.get_id() == id)
482}

484/// Check whether `prompted_arg` conflicts with an arg the user passed or an earlier answer.
485///
486/// Answers go in as default values, which clap never checks for conflicts, so asking here
487/// would let the parse land in a state clap rejects when typed.
488fn blocked(
489    cmd: &Command,
490    matches: &ArgMatches,
491    prompted_arg: &PromptedArg,
492    answers: &[Answer<'_>],
493) -> bool {
494    let (Some(level), Some(level_matches)) = (
495        command_at(cmd, &prompted_arg.subcommand_path),
496        matches_at(matches, &prompted_arg.subcommand_path),
497    ) else {
498        return false;
499    };
500    let Some(arg) = level.get_arguments().find(|arg| arg.get_id() == prompted_arg.attr.arg_id)
501    else {
502        return false;
503    };
504    let answered = |id: &Id| {
505        answers.iter().any(|answer| {
506            answer.prompted_arg.subcommand_path == prompted_arg.subcommand_path
507                && answer.prompted_arg.attr.arg_id == id
508        })
509    };
510    let conflicts = level.get_arg_conflicts_with(arg);
511    level
512        .get_arguments()
513        .filter(|other| other.get_id() != arg.get_id())
514        .filter(|other| answered(other.get_id()) || explicit(level_matches, other.get_id()))
515        .any(|other| {
516            other.is_exclusive_set()
517                || conflicts.iter().any(|conflict| conflict.get_id() == other.get_id())
518                || level.get_arg_conflicts_with(other).iter().any(|c| c.get_id() == arg.get_id())
519                || group_mates(level, arg).any(|mate| mate == other.get_id())
520        })
521}

523/// `arg`'s declared default values, split at its value delimiter.
524/// Without a delimiter, each default is one value. Defaults are set from strings, so any that
525/// is not UTF-8 is skipped.
526fn default_values(arg: &Arg) -> impl Iterator<Item = &str> {
527    let delimiter = arg.get_value_delimiter();
528    arg.get_default_values()
529        .iter()
530        .filter_map(|value| value.to_str())
531        .flat_map(move |value| value.split(move |char| Some(char) == delimiter))
532}

534/// The command at `path` passed through `edit`; unchanged when the path is missing.
535pub(super) fn with_command_at<F>(cmd: Command, path: &[&str], edit: F) -> Command
536where
537    F: FnOnce(Command) -> Command,
538{
539    match path.split_first() {
540        None => edit(cmd),
541        Some((name, rest)) if cmd.find_subcommand(name).is_some() => {
542            cmd.mut_subcommand(name, |sub| with_command_at(sub, rest, edit))
543        }
544        Some(_) => cmd,
545    }
546}

548fn command_at<'cmd>(cmd: &'cmd Command, path: &[&str]) -> Option<&'cmd Command> {
549    path.iter().try_fold(cmd, |level, name| level.find_subcommand(name))
550}

552/// Check whether the user passed `id`, on the command line or through its env var.
553fn explicit(matches: &ArgMatches, id: &Id) -> bool {
554    matches!(
555        matches.value_source(id.as_str()),
556        Some(ValueSource::CommandLine | ValueSource::EnvVariable)
557    )
558}

560/// The args sharing an exclusive `ArgGroup` with `arg`.
561fn group_mates<'cmd>(level: &'cmd Command, arg: &Arg) -> impl Iterator<Item = &'cmd Id> {
562    level
563        .get_groups()
564        .filter(|group| {
565            !ArgGroup::clone(group).is_multiple() && group.get_args().any(|id| id == arg.get_id())
566        })
567        .flat_map(ArgGroup::get_args)
568}

570/// For a flag, whether a "yes" stores `false`; `None` for an arg that takes a value.
571#[expect(
572    clippy::wildcard_enum_match_arm,
573    reason = "`ArgAction` is #[non_exhaustive]; every action but the two flags takes a value"
574)]
575const fn flag_negation(action: &ArgAction) -> Option<bool> {
576    match action {
577        ArgAction::SetTrue => Some(false),
578        ArgAction::SetFalse => Some(true),
579        _ => None,
580    }
581}

583/// Set each answer as its arg's default values. No values leave the arg with no default, so
584/// an emptied list drops the declared default.
585fn inject<'prompted_arg>(
586    cmd: Command,
587    answers: impl IntoIterator<Item = (&'prompted_arg PromptedArg, &'prompted_arg [String])>,
588) -> Command {
589    answers.into_iter().fold(cmd, |cmd, (prompted_arg, values)| {
590        // A default value is no explicit value to clap, so the rules that require the arg would
591        // still miss it.
592        with_arg_at(
593            unrequire(cmd, prompted_arg),
594            &prompted_arg.subcommand_path,
595            prompted_arg.attr.arg_id,
596            |arg| arg.default_values(values.iter().cloned()),
597        )
598    })
599}

601/// `question` with each `{id}` replaced by arg `id`'s values in `matches`, joined with `, `, and
602/// `{{` and `}}` by single braces. A `{id}` naming no arg stays as written.
603fn interpolate(question: &str, matches: &ArgMatches) -> String {
604    let mut interpolated = String::with_capacity(question.len());
605    let mut rest = question;
606    while let Some(brace) = rest.find(['{', '}']) {
607        let (text, tail) = rest.split_at(brace);
608        interpolated.push_str(text);
609        let escaped = tail.strip_prefix("{{").map(|after| ('{', after));
610        if let Some((brace, after)) =
611            escaped.or_else(|| tail.strip_prefix("}}").map(|after| ('}', after)))
612        {
613            interpolated.push(brace);
614            rest = after;
615            continue;
616        }
617        let placeholder = tail.strip_prefix('{').and_then(|inner| inner.split_once('}'));
618        if let Some((values, after)) = placeholder
619            .and_then(|(id, after)| matches.try_get_raw(id).ok().map(|values| (values, after)))
620        {
621            let values: Vec<_> =
622                values.into_iter().flatten().map(|v| v.to_string_lossy()).collect();
623            interpolated.push_str(&values.join(", "));
624            rest = after;
625            continue;
626        }
627        // A lone brace, or a `{id}` naming no arg: keep the brace as written.
628        let (brace, after) = tail.split_at(1);
629        interpolated.push_str(brace);
630        rest = after;
631    }
632    interpolated.push_str(rest);
633    interpolated
634}
635
636fn matches_at<'matches>(
637    matches: &'matches ArgMatches,
638    path: &[&str],
639) -> Option<&'matches ArgMatches> {
640    path.iter().try_fold(matches, |level, name| {
641        level.subcommand().filter(|(taken, _)| taken == name).map(|(_, sub)| sub)
642    })
643}

645/// The prompted args on the path the user took that got no value or only their default, with
646/// the default's values, split at the arg's value delimiter as clap splits them. One whose
647/// `default_value_if` fired is left to the arg that fired it: the conditional default beats an
648/// injected answer, so asking would drop the answer.
649fn pending<'prompted_arg>(
650    cmd: &Command,
651    matches: &ArgMatches,
652    prompted_args: &'prompted_arg [PromptedArg],
653) -> Vec<(&'prompted_arg PromptedArg, Vec<String>)> {
654    // Built, so a flag's action has filled in its implicit default (`false` for `SetTrue`).
655    let mut built = cmd.clone();
656    built.build();
657    prompted_args
658        .iter()
659        .filter_map(|prompted_arg| {
660            let arg = arg_at(&built, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)?;
661            let level = matches_at(matches, &prompted_arg.subcommand_path)?;
662            let id = prompted_arg.attr.arg_id;
663            match level.value_source(id) {
664                None if default_values(arg).next().is_none() => Some((prompted_arg, Vec::new())),
665                Some(ValueSource::DefaultValue) => {
666                    let current = raw_values(level, id);
667                    current
668                        .iter()
669                        .map(String::as_str)
670                        .eq(default_values(arg))
671                        .then_some((prompted_arg, current))
672                }
673                None | Some(_) => None,
674            }
675        })
676        .collect()
677}
678
679fn raw_values(matches: &ArgMatches, id: &str) -> Vec<String> {
680    matches
681        .try_get_raw(id)
682        .ok()
683        .flatten()
684        .into_iter()
685        .flatten()
686        .map(|value| value.to_string_lossy().into_owned())
687        .collect()
688}

690/// Check whether `arg` takes many values, as a `Vec<T>` field does.
691fn takes_many(arg: &Arg) -> bool {
692    matches!(arg.get_action(), ArgAction::Append)
693        || arg.get_num_args().is_some_and(|range| range.max_values() > 1)
694}

696/// Make each of `prompted_args` optional, so a missing one does not stop a parse.
697fn relax<'prompted_arg>(
698    cmd: Command,
699    prompted_args: impl IntoIterator<Item = &'prompted_arg PromptedArg>,
700) -> Command {
701    prompted_args.into_iter().fold(cmd, unrequire)
702}

704/// The first line of clap's message, without its `error: ` prefix. clap quotes the rejected
705/// `values`, so a multi-line one is shortened to its first line first, keeping the reason after it.
706fn summary(err: &clap::Error, values: &[String]) -> String {
707    let mut rendered = err.render().to_string();
708    for value in values {
709        if let Some((first_line, _)) = value.split_once('\n') {
710            rendered = rendered.replace(value.as_str(), &format!("{first_line}\u{2026}"));
711        }
712    }
713    let line = rendered.lines().next().unwrap_or_default();
714    line.strip_prefix("error: ").unwrap_or(line).to_owned()
715}

717/// Drop the rules that make `prompted_arg` required: its `required`, its
718/// `required_unless_present`, and the `required` of the groups it belongs to. clap can't clear
719/// `required_unless_present_all` or `required_if_eq`, so those stay.
720fn unrequire(cmd: Command, prompted_arg: &PromptedArg) -> Command {
721    let id = prompted_arg.attr.arg_id;
722    with_command_at(cmd, &prompted_arg.subcommand_path, |level| {
723        if !level.get_arguments().any(|arg| arg.get_id() == id) {
724            return level;
725        }
726        let groups: Vec<Id> = level
727            .get_groups()
728            .filter(|group| group.is_required_set() && group.get_args().any(|arg| arg == id))
729            .map(|group| group.get_id().clone())
730            .collect();
731        let level =
732            level.mut_arg(id, |arg| arg.required(false).required_unless_present(Resettable::Reset));
733        groups
734            .iter()
735            .fold(level, |level, group| level.mut_group(group, |group| group.required(false)))
736    })
737}

739/// Apply `edit` to the arg `id` of the command at `path`; unchanged when either is missing.
740pub(super) fn with_arg_at<F>(cmd: Command, path: &[&str], id: &str, edit: F) -> Command
741where
742    F: FnOnce(Arg) -> Arg,
743{
744    with_command_at(cmd, path, |level| {
745        if level.get_arguments().any(|arg| arg.get_id() == id) {
746            level.mut_arg(id, edit)
747        } else {
748            level
749        }
750    })
