27impl Backend for Inquire {
28    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
29        Ok(Confirm::new(prompt.question)
30            .with_default(prompt.default)
31            .with_render_config(RenderConfig::from(prompt.styles))
32            .prompt()
33            .map_err(InquireError)?)
34    }
35
36    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
37        let names: Vec<&str> = prompt.options.iter().map(PossibleValue::get_name).collect();
38        let defaults: Vec<usize> = (0..names.len())
39            .filter(|index| {
40                prompt.defaults.iter().any(|default| Some(&**default) == names.get(*index).copied())
41            })
42            .collect();
43        let select = MultiSelect::new(prompt.question, names)
44            .with_default(&defaults)
45            .with_render_config(RenderConfig::from(prompt.styles));
46        let select = if prompt.optional {
47            select
48        } else {
49            select
50                .with_validator(MinLengthValidator::new(1).with_message(prompt.selection_required))
51        };
52        let picked = select.prompt().map_err(InquireError)?;
53        Ok(picked.into_iter().map(str::to_owned).collect())
54    }
55
56    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
57        let names: Vec<&str> = prompt.options.iter().map(PossibleValue::get_name).collect();
58        let default =
59            prompt.default.and_then(|default| names.iter().position(|name| *name == default));
60        let none = prompt.optional.then_some(prompt.none_option);
61        let items: Vec<&str> = names.iter().copied().chain(none).collect();
62        let select = Select::new(prompt.question, items)
63            .with_render_config(RenderConfig::from(prompt.styles));
64        let picked = match default {
65            Some(index) => select.with_starting_cursor(index),
66            None => select,
67        }
68        .raw_prompt()
69        .map_err(InquireError)?;
70        // Past the options is the none option, only listed for an optional prompt.
71        Ok(names.get(picked.index).map(|name| (*name).to_owned()))
72    }
73
74    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
75        let text = Text::new(prompt.question).with_render_config(RenderConfig::from(prompt.styles));
76        let text = match prompt.default {
77            Some(default) => text.with_default(default),
78            None => text,
79        };
80        let text = match prompt.hint {
81            Some(hint) => text.with_help_message(hint),
82            None => text,
83        };
84        let text = match prompt.initial {
85            Some(initial) => text.with_initial_value(initial),
86            None => text,
87        };
88        let text = if prompt.optional {
89            text
90        } else {
91            text.with_validator(ValueRequiredValidator::default())
92        };
93        let answer = text.prompt().map_err(InquireError)?;
94        Ok(Some(answer).filter(|answer| !answer.is_empty()))
95    }
96}

98impl From<NamedColor> for inquire::ui::Color {
99    fn from(NamedColor(ansi): NamedColor) -> Self {
100        match ansi {
101            AnsiColor::Black => Self::Black,
102            AnsiColor::Red => Self::DarkRed,
103            AnsiColor::Green => Self::DarkGreen,
104            AnsiColor::Yellow => Self::DarkYellow,
105            AnsiColor::Blue => Self::DarkBlue,
106            AnsiColor::Magenta => Self::DarkMagenta,
107            AnsiColor::Cyan => Self::DarkCyan,
108            AnsiColor::White => Self::Grey,
109            AnsiColor::BrightBlack => Self::DarkGrey,
110            AnsiColor::BrightRed => Self::LightRed,
111            AnsiColor::BrightGreen => Self::LightGreen,
112            AnsiColor::BrightYellow => Self::LightYellow,
113            AnsiColor::BrightBlue => Self::LightBlue,
114            AnsiColor::BrightMagenta => Self::LightMagenta,
115            AnsiColor::BrightCyan => Self::LightCyan,
116            AnsiColor::BrightWhite => Self::White,
117        }
118    }
119}

121impl From<CommandStyles<'_>> for RenderConfig<'static> {
122    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
123        let base = Self::default_colored();
124        Self {
125            answer: StyleSheet::from(ClapStyle(styles.get_literal())),
126            default_value: StyleSheet::from(ClapStyle(styles.get_placeholder())),
127            error_message: ErrorMessageRenderConfig {
128                message: StyleSheet::from(ClapStyle(styles.get_error())),
129                ..base.error_message
130            },
131            prompt: StyleSheet::from(ClapStyle(styles.get_header())),
132            ..base
133        }
134    }
135}

137impl From<ClapColor> for inquire::ui::Color {
138    fn from(ClapColor(color): ClapColor) -> Self {
139        match color {
140            Color::Ansi(ansi) => Self::from(NamedColor(ansi)),
141            Color::Ansi256(index) => Self::AnsiValue(index.0),
142            Color::Rgb(rgb) => Self::Rgb { r: rgb.0, g: rgb.1, b: rgb.2 },
143        }
144    }
145}

147impl From<ClapStyle<'_>> for StyleSheet {
148    fn from(ClapStyle(style): ClapStyle<'_>) -> Self {
149        let base = style
150            .get_fg_color()
151            .map_or_else(Self::new, |color| Self::new().with_fg(ClapColor(color).into()));
152        [(Effects::BOLD, Attributes::BOLD), (Effects::ITALIC, Attributes::ITALIC)]
153            .into_iter()
154            .filter(|(effect, _)| style.get_effects().contains(*effect))
155            .fold(base, |sheet, (_, attribute)| sheet.with_attr(attribute))
156    }
157}

159impl From<InquireError> for PromptError {
160    #[expect(
161        clippy::wildcard_enum_match_arm,
162        reason = "`InquireError` is #[non_exhaustive]; anything unlisted is an I/O-level failure"
163    )]
164    fn from(InquireError(err): InquireError) -> Self {
165        match err {
166            inquire::InquireError::IO(err) => Self::from(err),
167            inquire::InquireError::NotTTY => Self::NotATerminal,
168            inquire::InquireError::OperationCanceled
169            | inquire::InquireError::OperationInterrupted => Self::Cancelled,
170            other => Self::Io(io::Error::other(other)),
171        }
172    }
