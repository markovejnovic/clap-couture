4use crate::interactive::PromptError;

6/// How an editor prompt ended, for the line a backend leaves in the transcript.
7#[cfg(feature = "interactive-editor")]
8pub(super) enum Edited {
9    /// Answered, shown as the answer's first line, with `…` when more follow; empty for an
10    /// optional arg left unset.
11    Answered(String),
12    /// Backed out of, or answered with nothing where an answer is required.
13    Cancelled,
14    /// The editor failed; its error is reported once parsing ends.
15    Failed,
16}
17
18#[cfg(feature = "interactive-editor")]
19impl Edited {
20    /// How `answer`, the editor's outcome for a prompt that is `optional` or not, ended.
21    pub(super) fn of(answer: &Result<Option<String>, PromptError>, optional: bool) -> Self {
22        match answer {
23            Ok(Some(answer)) => Self::Answered(match answer.split_once('\n') {
24                Some((first_line, _)) => format!("{first_line}\u{2026}"),
25                None => answer.clone(),
26            }),
27            Ok(None) if optional => Self::Answered(String::new()),
28            Ok(None) | Err(PromptError::Cancelled | PromptError::Signalled(_)) => Self::Cancelled,
29            Err(_) => Self::Failed,
30        }
31    }
32}
