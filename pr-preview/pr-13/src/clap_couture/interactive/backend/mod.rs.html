69/// A terminal UI that can ask for one arg's value.
70///
71/// Implement it to bring your own UI, or to answer from a script in tests.
72pub trait Backend {
73    /// Ask a yes/no question.
74    ///
75    /// # Errors
76    /// - [`PromptError::Cancelled`] when the user backs out.
77    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError>;
78
79    /// Ask in the user's editor, returning `None` for an empty answer. Whether an empty answer is
80    /// allowed is decided by the caller, from `prompt.optional`.
81    ///
82    /// The default opens the editor through
83    /// [`EditorInteraction`](crate::interactive::EditorInteraction), asking with
84    /// [`Backend::confirm`] whether the user is done when the editor returns right
85    /// away. Override it to draw something around the editor, and use `EditorInteraction` from the
86    /// override.
87    ///
88    /// # Errors
89    /// - [`PromptError::Signalled`] when the editor dies of Ctrl-C or of the terminal closing;
90    ///   [`PromptError::Cancelled`] on Windows.
91    /// - [`PromptError::Cancelled`] when the done-editing question is cancelled.
92    /// - [`PromptError::Io`] when the editor cannot be found or started, or exits unsuccessfully.
93    #[cfg(feature = "interactive-editor")]
94    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
95        EditorInteraction::start(self, prompt)?.answer()
96    }
97
98    /// Check whether there is a terminal to ask on: stdin and stderr must both be one.
99    fn is_available(&self) -> bool {
100        io::stdin().is_terminal() && io::stderr().is_terminal()
101    }
102
103    /// Ask to pick any number of `prompt.options`, returning the picked options' names in
104    /// declared order. Unless `prompt.optional`, at least one must be picked.
105    ///
106    /// # Errors
107    /// - [`PromptError::Cancelled`] when the user backs out.
108    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError>;
109
110    /// Show clap's complaint about the previous answer, before asking again.
111    ///
112    /// # Errors
113    /// - [`PromptError::Io`] when the message cannot be shown.
114    fn report(&self, error: &str, _styles: CommandStyles<'_>) -> Result<(), PromptError> {
115        writeln!(io::stderr(), "{error}").map_err(PromptError::Io)
116    }
117
118    /// Ask for a secret without echoing it, returning `None` only for an empty answer to an
119    /// optional prompt.
120    ///
121    /// # Errors
122    /// - [`PromptError::Cancelled`] when the user backs out.
123    fn secret(&self, prompt: &SecretPrompt<'_>) -> Result<Option<SecretString>, PromptError>;
124
125    /// Ask to pick one of `prompt.options`, returning the picked option's name, or `None` when an
126    /// optional prompt is answered with none of them.
127    ///
128    /// # Errors
129    /// - [`PromptError::Cancelled`] when the user backs out.
130    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError>;
131
132    /// Ask for free text, returning `None` only for an empty answer to an optional prompt.
133    ///
134    /// # Errors
135    /// - [`PromptError::Cancelled`] when the user backs out.
136    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError>;
137}
138
139impl<B> Backend for &B
140where
141    B: Backend + ?Sized,
142{
143    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
144        (**self).confirm(prompt)
145    }
146
147    #[cfg(feature = "interactive-editor")]
148    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
149        (**self).editor(prompt)
150    }
151
152    fn is_available(&self) -> bool {
153        (**self).is_available()
154    }
155
156    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
157        (**self).multi_select(prompt)
158    }
159
160    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
161        (**self).report(error, styles)
162    }
163
164    fn secret(&self, prompt: &SecretPrompt<'_>) -> Result<Option<SecretString>, PromptError> {
165        (**self).secret(prompt)
166    }
167
168    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
169        (**self).select(prompt)
170    }
171
172    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
173        (**self).text(prompt)
174    }
175}
