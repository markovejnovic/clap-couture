1//! Stdin, read when something is piped into it or the user asked for it with `-`.
2
3use std::io::{self, IsTerminal as _, Read as _};
4
5/// The most stdin a parse reads, 16 MiB: an arg's value, not a stream, so `cat /dev/zero |`
6/// fails the parse rather than exhausting memory.
7pub(crate) const STDIN_LIMIT: usize = 16 << 20;
8
9/// Which stdin [`Stdin::read`] reads.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub(crate) enum ReadWhen {
12    /// Whatever stdin is, as `-` asks: a terminal is read until end of input.
13    Always,
14    /// Only a pipe; a terminal, a file or `/dev/null` is left alone. A socket counts as a pipe:
15    /// some runtimes, like Node's, pipe a child's stdin through one.
16    Piped,
17}
18
19/// The process's stdin, as a parse reads it.
20pub(crate) struct ProcessStdin;
21
22/// Stdin as a parse sees it: the process's own, [`ProcessStdin`], or a stand-in in tests.
23pub(crate) trait Stdin {
24    /// Check whether stdin is a terminal.
25    fn is_terminal(&self) -> bool;
26
27    /// Everything on stdin, but at most one byte past [`STDIN_LIMIT`]; `None` when `when` leaves
28    /// it unread.
29    ///
30    /// # Errors
31    /// Reading stdin failed.
32    fn read(&self, when: ReadWhen) -> io::Result<Option<Vec<u8>>>;
33}
34
35impl Stdin for ProcessStdin {
36    fn is_terminal(&self) -> bool {
37        io::stdin().is_terminal()
38    }
39
40    fn read(&self, when: ReadWhen) -> io::Result<Option<Vec<u8>>> {
41        if when == ReadWhen::Piped && !is_pipe() {
42            return Ok(None);
43        }
44        let mut stdin = Vec::new();
45        let limit = u64::try_from(STDIN_LIMIT).unwrap_or(u64::MAX).saturating_add(1);
46        io::stdin().lock().take(limit).read_to_end(&mut stdin)?;
47        Ok(Some(stdin))
48    }
49}
50
51#[cfg(unix)]
52fn is_pipe() -> bool {
53    use std::{
54        fs::File,
55        os::{fd::AsFd as _, unix::fs::FileTypeExt as _},
56    };
57
58    // A duplicate, so dropping the `File` leaves stdin open.
59    let Ok(stdin) = io::stdin().as_fd().try_clone_to_owned() else { return false };
60    File::from(stdin).metadata().is_ok_and(|metadata| {
61        let file_type = metadata.file_type();
62        file_type.is_fifo() || file_type.is_socket()
63    })
64}
65
66#[cfg(windows)]
67fn is_pipe() -> bool {
68    use std::os::windows::io::AsRawHandle as _;
69
70    use windows_sys::Win32::Storage::FileSystem::{FILE_TYPE_PIPE, GetFileType};
71
72    // SAFETY: the handle is borrowed from stdin for the call. An invalid or missing one makes
73    // `GetFileType` report `FILE_TYPE_UNKNOWN`.
74    unsafe { GetFileType(io::stdin().as_raw_handle()) == FILE_TYPE_PIPE }
75}
76
77/// Elsewhere, there is no telling a pipe apart, so stdin is never read.
78#[cfg(not(any(unix, windows)))]
79const fn is_pipe() -> bool {
80    false
