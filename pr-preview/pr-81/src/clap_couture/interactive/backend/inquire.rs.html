26/// The header effects a `StyleSheet` carries, so the ones inquire draws a question in.
27const DRAWN: Effects = Effects::BOLD.insert(Effects::ITALIC);
28
29struct InquireError(inquire::InquireError);

31/// Prompts drawn by inquire.
32#[derive(Clone, Copy, Debug, Default)]
33pub struct Inquire;

35impl Backend for Inquire {
36    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
37        let question = prompt.styles.question_in_header(prompt.question, DRAWN);
38        Ok(Confirm::new(&question)
39            .with_default(prompt.default)
40            .with_render_config(RenderConfig::from(prompt.styles))
41            .prompt()
42            .map_err(InquireError)?)
43    }
44
45    /// The closed editor leaves the question with the answer's first line, or `<canceled>`, as
46    /// inquire leaves an answered or cancelled prompt.
47    #[cfg(feature = "interactive-editor")]
48    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
49        use std::io::Write as _;
50
51        use clap::builder::styling::Style;
52
53        use super::builtin::Edited;
54
55        let answer = super::EditorInteraction::start(self, prompt)
56            .and_then(super::EditorInteraction::answer);
57        let CommandStyles(styles) = prompt.styles;
58        // inquire's colored defaults: a light green `>` before an answered prompt, and a dark red
59        // `<canceled>` after a cancelled one.
60        let (closing, closing_style) = match Edited::of(&answer, prompt.optional) {
61            Edited::Answered(line) => (line, *styles.get_literal()),
62            Edited::Cancelled => {
63                ("<canceled>".to_owned(), Style::new().fg_color(Some(AnsiColor::Red.into())))
64            }
65            Edited::Failed => return answer,
66        };
67        let prefix = Style::new().fg_color(Some(AnsiColor::BrightGreen.into()));
68        let header = styles.get_header();
69        // Best effort: after a hang-up there is no terminal left to draw on.
70        drop(writeln!(
71            io::stderr(),
72            "{prefix}>{prefix:#} {header}{}{header:#} {closing_style}{closing}{closing_style:#}",
73            // Drawn in the whole header style here, not inquire's.
74            prompt.styles.question_in_header(prompt.question, header.get_effects())
75        ));
76        answer
77    }

79    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
80        let names: Vec<&str> = prompt.options.iter().map(PossibleValue::get_name).collect();
81        let defaults: Vec<usize> = (0..names.len())
82            .filter(|index| {
83                prompt.defaults.iter().any(|default| Some(&**default) == names.get(*index).copied())
84            })
85            .collect();
86        let question = prompt.styles.question_in_header(prompt.question, DRAWN);
87        let select = MultiSelect::new(&question, names)
88            .with_default(&defaults)
89            .with_render_config(RenderConfig::from(prompt.styles));
90        let select = if prompt.optional {
91            select
92        } else {
93            select
94                .with_validator(MinLengthValidator::new(1).with_message(prompt.selection_required))
95        };
96        let picked = select.prompt().map_err(InquireError)?;
97        Ok(picked.into_iter().map(str::to_owned).collect())
98    }
99
100    fn secret(&self, prompt: &SecretPrompt<'_>) -> Result<Option<SecretString>, PromptError> {
101        let question = prompt.styles.question_in_header(prompt.question, DRAWN);
102        let input = Password::new(&question)
103            .without_confirmation()
104            .with_display_mode(PasswordDisplayMode::Masked)
105            .with_render_config(RenderConfig::from(prompt.styles));
106        let input = if prompt.optional {
107            input
108        } else {
109            input.with_validator(ValueRequiredValidator::default())
110        };
111        let answer = input.prompt().map_err(InquireError)?;
112        Ok((!answer.is_empty()).then(|| SecretString::from(answer)))
113    }
114
115    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
116        let names: Vec<&str> = prompt.options.iter().map(PossibleValue::get_name).collect();
117        let default =
118            prompt.default.and_then(|default| names.iter().position(|name| *name == default));
119        let none = prompt.optional.then_some(prompt.none_option);
120        let items: Vec<&str> = names.iter().copied().chain(none).collect();
121        let question = prompt.styles.question_in_header(prompt.question, DRAWN);
122        let select =
123            Select::new(&question, items).with_render_config(RenderConfig::from(prompt.styles));
124        let picked = match default {
125            Some(index) => select.with_starting_cursor(index),
126            None => select,
127        }
128        .raw_prompt()
129        .map_err(InquireError)?;
130        // Past the options is the none option, only listed for an optional prompt.
131        Ok(names.get(picked.index).map(|name| (*name).to_owned()))
132    }
133
134    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
135        let question = prompt.styles.question_in_header(prompt.question, DRAWN);
136        let text = Text::new(&question).with_render_config(RenderConfig::from(prompt.styles));
137        let text = match prompt.default {
138            Some(default) => text.with_default(default),
139            None => text,
140        };
141        let text = match prompt.hint {
142            Some(hint) => text.with_help_message(hint),
143            None => text,
144        };
145        let text = match prompt.initial {
146            Some(initial) => text.with_initial_value(initial),
147            None => text,
148        };
149        let text = if prompt.optional {
150            text
151        } else {
152            text.with_validator(ValueRequiredValidator::default())
153        };
154        let answer = text.prompt().map_err(InquireError)?;
155        Ok(Some(answer).filter(|answer| !answer.is_empty()))
156    }
157}

159impl From<NamedColor> for inquire::ui::Color {
160    fn from(NamedColor(ansi): NamedColor) -> Self {
161        match ansi {
162            AnsiColor::Black => Self::Black,
163            AnsiColor::Red => Self::DarkRed,
164            AnsiColor::Green => Self::DarkGreen,
165            AnsiColor::Yellow => Self::DarkYellow,
166            AnsiColor::Blue => Self::DarkBlue,
167            AnsiColor::Magenta => Self::DarkMagenta,
168            AnsiColor::Cyan => Self::DarkCyan,
169            AnsiColor::White => Self::Grey,
170            AnsiColor::BrightBlack => Self::DarkGrey,
171            AnsiColor::BrightRed => Self::LightRed,
172            AnsiColor::BrightGreen => Self::LightGreen,
173            AnsiColor::BrightYellow => Self::LightYellow,
174            AnsiColor::BrightBlue => Self::LightBlue,
175            AnsiColor::BrightMagenta => Self::LightMagenta,
176            AnsiColor::BrightCyan => Self::LightCyan,
177            AnsiColor::BrightWhite => Self::White,
178        }
179    }
180}

182impl From<CommandStyles<'_>> for RenderConfig<'static> {
183    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
184        let base = Self::default_colored();
185        Self {
186            answer: StyleSheet::from(ClapStyle(styles.get_literal())),
187            default_value: StyleSheet::from(ClapStyle(styles.get_placeholder())),
188            error_message: ErrorMessageRenderConfig {
189                message: StyleSheet::from(ClapStyle(styles.get_error())),
190                ..base.error_message
191            },
192            prompt: StyleSheet::from(ClapStyle(styles.get_header())),
193            ..base
194        }
195    }
196}

198impl From<ClapColor> for inquire::ui::Color {
199    fn from(ClapColor(color): ClapColor) -> Self {
200        match color {
201            Color::Ansi(ansi) => Self::from(NamedColor(ansi)),
202            Color::Ansi256(index) => Self::AnsiValue(index.0),
203            Color::Rgb(rgb) => Self::Rgb { r: rgb.0, g: rgb.1, b: rgb.2 },
204        }
205    }

208impl From<ClapStyle<'_>> for StyleSheet {
209    fn from(ClapStyle(style): ClapStyle<'_>) -> Self {
210        let base = style
211            .get_fg_color()
212            .map_or_else(Self::new, |color| Self::new().with_fg(ClapColor(color).into()));
213        [(Effects::BOLD, Attributes::BOLD), (Effects::ITALIC, Attributes::ITALIC)]
214            .into_iter()
215            .filter(|(effect, _)| style.get_effects().contains(*effect))
216            .fold(base, |sheet, (_, attribute)| sheet.with_attr(attribute))
217    }
218}

220impl From<InquireError> for PromptError {
221    #[expect(
222        clippy::wildcard_enum_match_arm,
223        reason = "`InquireError` is #[non_exhaustive]; anything unlisted is an I/O-level failure"
224    )]
225    fn from(InquireError(err): InquireError) -> Self {
226        match err {
227            inquire::InquireError::IO(err) => Self::from(err),
228            inquire::InquireError::NotTTY => Self::NotATerminal,
229            inquire::InquireError::OperationCanceled
230            | inquire::InquireError::OperationInterrupted => Self::Cancelled,
231            other => Self::Io(io::Error::other(other)),
232        }
233    }
