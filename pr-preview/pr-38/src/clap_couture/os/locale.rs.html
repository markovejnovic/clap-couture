10/// The ASCII codesets, as the `C` and `POSIX` locales report them. [`Encoding::for_label`] reads
11/// them as windows-1252, which takes any byte, so they name no encoding to fall back to.
12#[cfg(unix)]
13const ASCII_CODESETS: &[&str] = &["ansi_x3.4-1968", "us-ascii"];
14
15/// Codesets as macOS and the BSDs spell them, where that isn't also a label
16/// [`Encoding::for_label`] knows, with the label it knows. glibc's spellings already are.
17#[cfg(unix)]
18const CODESET_LABELS: &[(&str, &str)] = &[
19    ("big5hkscs", "big5-hkscs"),
20    ("cp949", "windows-949"),
21    ("euccn", "gb2312"),
22    ("eucjp", "euc-jp"),
23    ("euckr", "euc-kr"),
24];

26/// Text in no encoding that could be read, decoded as best it could be.
27#[derive(Debug)]
28pub(crate) struct NotUnicode {
29    /// How many U+FFFD in `text` mark bytes that couldn't be read, not text that was there.
30    pub(crate) marks: usize,
31    pub(crate) text: String,
32}
33
34impl NotUnicode {
35    /// `bytes` as UTF-8, or as the encoding their byte order mark names, each malformed sequence
36    /// marked with a U+FFFD.
37    fn read(bytes: &[u8]) -> Self {
38        let mut decoder = UTF_8.new_decoder();
39        let mut unread = bytes;
40        let mut readable = Self { marks: 0, text: String::new() };
41        loop {
42            readable.text.reserve(
43                decoder
44                    .max_utf8_buffer_length_without_replacement(unread.len())
45                    .unwrap_or_default(),
46            );
47            let (result, read) =
48                decoder.decode_to_string_without_replacement(unread, &mut readable.text, true);
49            unread = unread.get(read..).unwrap_or_default();
50            match result {
51                DecoderResult::InputEmpty => return readable,
52                DecoderResult::OutputFull => {}
53                DecoderResult::Malformed(..) => {
54                    readable.text.push(char::REPLACEMENT_CHARACTER);
55                    readable.marks = readable.marks.saturating_add(1);
56                }
57            }
58        }
59    }
60}

62/// Representation of the system's locale.
63#[derive(Debug)]
64pub(crate) struct Locale {
65    /// The ANSI code page, which programs that aren't Unicode use.
66    #[cfg(windows)]
67    code_page: u32,
68    /// The codeset, as the C library names it, like `UTF-8`, `EUC-JP` or `eucJP`; empty for a
69    /// locale that names none.
70    #[cfg(unix)]
71    codeset: String,
72}
73
74impl Locale {
75    pub(crate) fn decode(bytes: &[u8]) -> Result<String, NotUnicode> {
76        let legacy = Self::read().and_then(|locale| locale.encoding());
77        decode_with(bytes, legacy).ok_or_else(|| NotUnicode::read(bytes))
78    }
79
80    /// The encoding the locale uses for text that isn't Unicode. `None` for a locale that names
81    /// none, like `C`.
82    pub(crate) fn encoding(&self) -> Option<&'static Encoding> {
83        #[cfg(windows)]
84        {
85            codepage::to_encoding_no_replacement(u16::try_from(self.code_page).ok()?)
86        }
87
88        #[cfg(unix)]
89        {
90            let codeset = self.codeset.to_ascii_lowercase();
91            if ASCII_CODESETS.contains(&codeset.as_str()) {
92                return None;
93            }
94            let label = CODESET_LABELS
95                .iter()
96                .find(|(spelling, _)| *spelling == codeset)
97                .map_or(codeset.as_str(), |(_, label)| label);
98            Encoding::for_label(label.as_bytes()).filter(|encoding| *encoding != REPLACEMENT)
99        }
100
101        #[cfg(not(any(unix, windows)))]
102        {
103            None
104        }
105    }
106
107    /// Read the system's locale. `None` for a locale the C library doesn't have.
108    #[cfg_attr(
109        windows,
110        expect(
111            clippy::unnecessary_wraps,
112            reason = "Windows always has a code page; elsewhere the locale can be missing"
113        )
114    )]
115    pub(crate) fn read() -> Option<Self> {
116        #[cfg(windows)]
117        {
118            // SAFETY: takes nothing, reads a process setting.
119            let code_page = unsafe { windows_sys::Win32::Globalization::GetACP() };
120            Some(Self { code_page })
121        }
122
123        // A locale object of our own, used on this thread only, rather than `setlocale`, which
124        // would change the locale for the whole process.
125        #[cfg(unix)]
126        {
127            // SAFETY: an empty name asks for the locale the environment names; a null base
128            // starts from nothing.
129            let locale = unsafe {
130                libc::newlocale(libc::LC_CTYPE_MASK, c"".as_ptr(), core::ptr::null_mut())
131            };
132            if locale.is_null() {
133                return None;
134            }
135
136            // SAFETY: `locale` is a live locale object; this changes this thread's locale only.
137            let previous = unsafe { libc::uselocale(locale) };
138            // SAFETY: `CODESET` is a valid item.
139            let codeset = unsafe { libc::nl_langinfo(libc::CODESET) };
140            let codeset = (!codeset.is_null()).then(|| {
141                // SAFETY: a C string that lives while `locale` does, copied before it is freed.
142                unsafe { CStr::from_ptr(codeset) }.to_string_lossy().into_owned()
143            });
144
145            // SAFETY: `previous` is the locale this thread had.
146            unsafe {
147                libc::uselocale(previous);
148            }
149            // SAFETY: `locale` is no longer in use.
150            unsafe {
151                libc::freelocale(locale);
152            }
153
154            Some(Self { codeset: codeset.unwrap_or_default() })
155        }
156
157        #[cfg(not(any(unix, windows)))]
158        {
159            None
160        }
161    }
162}

164/// The text of a saved file: in the encoding its byte order mark names, as notepad's "Unicode"
165/// saves UTF-16, or else UTF-8, or else `legacy`, the encoding older editors save in. The mark
166/// is dropped. `None` when none of them fits.
167fn decode_with(bytes: &[u8], legacy: Option<&'static Encoding>) -> Option<String> {
168    if let Some((encoding, mark)) = Encoding::for_bom(bytes) {
169        let text = bytes.get(mark..).unwrap_or_default();
170        return encoding.decode_without_bom_handling_and_without_replacement(text).map(Into::into);
171    }
172    [Some(UTF_8), legacy.filter(|legacy| *legacy != UTF_8)]
173        .into_iter()
174        .flatten()
175        .find_map(|encoding| encoding.decode_without_bom_handling_and_without_replacement(bytes))
176        .map(Into::into)
177}
