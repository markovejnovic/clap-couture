30/// Styles a run of help text (a command's about or a category description).
31pub(crate) trait TextRenderer {
32    fn render_str<W>(&self, out: &mut W, text: &str) -> fmt::Result
33    where
34        W: fmt::Write;
35
36    /// `text` as this renderer shows it, as text clap writes as is.
37    fn restyle(&self, text: &StyledStr) -> StyledStr;
38
39    /// `text` as this renderer shows it in a subcommand listing's row.
40    fn restyle_line(&self, text: &StyledStr) -> StyledStr;
41
42    /// `wrapped`, text clap wrapped and indented, as this renderer shows it.
43    fn restyle_wrapped(&self, wrapped: &str) -> String;
44}

46pub(crate) trait CommandRenderExt {
47    /// The width clap wraps this command's help to; `None` when it doesn't wrap.
48    ///
49    /// clap's build hands a command's `term_width` and `max_term_width` to each subcommand, so
50    /// the built root's width is every built command's.
51    fn help_width(&self) -> Option<usize>;
52
53    /// Check whether clap, wrapping `text` as this command's after-help, leaves it as it is.
54    fn keeps_after_help(&self, text: &str) -> bool;
55
56    /// `render`'s text of this command, each help on the line below its name when clap's wrap of
57    /// an after-help would break the text of clap's usual layout.
58    fn render_for_after_help<F>(&self, render: F) -> String
59    where
60        Self: Sized,
61        F: Fn(Self) -> String;
62
63    /// `text` rendered on one line as a prompt's question: styled when this command's color is on
64    /// for stderr, else plain.
65    #[cfg(feature = "interactive")]
66    fn render_question(&self, text: &str) -> String;
67
68    /// This command with its about and long about rendered.
69    #[must_use]
70    fn with_rendered_about(self) -> Self;
71
72    /// This command with each arg's help and long help rendered.
73    #[must_use]
74    fn with_rendered_arg_help(self) -> Self;
75
76    /// Writes this command's visible subcommands to `out`, wrapped to `width`.
77    fn write_command_sections<W>(
78        &self,
79        out: &mut W,
80        categories: CommandCategoryMap,
81        width: Option<usize>,
82    ) -> fmt::Result
83    where
84        W: fmt::Write;
85}

87impl CommandRenderExt for Command {
88    fn help_width(&self) -> Option<usize> {
89        // clap has no getter for it: it follows `term_width`, `max_term_width`, the terminal and
90        // `COLUMNS`, and wraps only with its `wrap_help` feature. So clap wraps a probe: a line
91        // breaks after the last word ending within the width, and the words of the `a` line end
92        // at odd columns while those of the `b` line end at even ones.
93        const PROBE_WORDS: usize = 500;
94        let odd = vec!["a"; PROBE_WORDS].join(" ");
95        let even = format!("b{}", vec!["b"; PROBE_WORDS].join(" "));
96        let unwrapped = odd.len();
97        let help = self
98            .clone()
99            .help_template("{after-help}")
100            .after_help(format!("{odd}\n{even}"))
101            .render_help()
102            .to_string();
103        let first_line =
104            |letter: char| help.lines().find(|line| line.starts_with(letter)).map_or(0, str::len);
105        let width = first_line('a').max(first_line('b'));
106        (width < unwrapped).then_some(width)
107    }
108
109    fn keeps_after_help(&self, text: &str) -> bool {
110        // clap writes an after-help below a blank line, of which its trim of the help keeps one
111        // newline.
112        let rewrapped = self
113            .clone()
114            .help_template("{after-help}")
115            .after_help(text.to_owned())
116            .render_help()
117            .ansi()
118            .to_string();
119        text.is_empty() || rewrapped == format!("\n{}\n", text.trim_end())
120    }
121
122    fn render_for_after_help<F>(&self, render: F) -> String
123    where
124        F: Fn(Self) -> String,
125    {
126        let text = render(self.clone());
127        if self.keeps_after_help(&text) { text } else { render(self.clone().next_line_help(true)) }
128    }
129
130    #[cfg(feature = "interactive")]
131    fn render_question(&self, text: &str) -> String {
132        let colored = match self.get_color() {
133            ColorChoice::Always => true,
134            ColorChoice::Never => false,
135            ColorChoice::Auto => {
136                anstream::AutoStream::choice(&std::io::stderr()) != anstream::ColorChoice::Never
137            }
138        };
139        let mut question = StyledStr::new();
140        // `StyledStr`'s `fmt::Write` is infallible.
141        Renderer::new(self.get_styles()).render_str(&mut question, text).ok();
142        if colored { question.ansi().to_string() } else { question.to_string() }
143    }
144
145    fn with_rendered_about(self) -> Self {
146        let renderer = Renderer::new(self.get_styles());
147        let about = self.get_about().map(|text| renderer.restyle(text));
148        let long_about = self.get_long_about().map(|text| renderer.restyle(text));
149        self.about(Resettable::from(about)).long_about(Resettable::from(long_about))
150    }
151
152    fn with_rendered_arg_help(self) -> Self {
153        let styles = self.get_styles().clone();
154        let renderer = Renderer::new(&styles);
155        self.mut_args(|arg| {
156            let help = arg.get_help().map(|text| renderer.restyle(text));
157            let long_help = arg.get_long_help().map(|text| renderer.restyle(text));
158            arg.help(Resettable::from(help)).long_help(Resettable::from(long_help))
159        })
160    }
161
162    fn write_command_sections<W>(
163        &self,
164        out: &mut W,
165        categories: CommandCategoryMap,
166        width: Option<usize>,
167    ) -> fmt::Result
168    where
169        W: fmt::Write,
170    {
171        let menu = MenuView::new(self, categories, Renderer::new(self.get_styles()), width);
172        write!(out, "{menu}")
173    }
174}

176/// A command's subcommands as clap lists them, partitioned by category.
177struct Menu<'cmd> {
178    category_map: CommandCategoryMap,
179    parent: &'cmd Command,
180    /// Each listed subcommand's lines, in clap's order.
181    rows: Vec<Row<'cmd>>,
182}
183
184impl<'cmd> Menu<'cmd> {
185    /// Ends the `{subcommands}` template, so clap's trim of the help keeps the last row's padding.
186    const LISTING_END: &'static str = "\u{E000}";
187
188    /// The distinct categories with at least one visible member, in first-appearance order.
189    fn categories(&self) -> impl Iterator<Item = &'static Category> {
190        self.category_map
191            .distinct()
192            .filter(|category| self.members(Some(category)).next().is_some())
193    }
194
195    fn has_uncategorized(&self) -> bool {
196        self.members(None).next().is_some()
197    }
198
199    /// The row of the `help` subcommand clap generated for the parent; `None` when it didn't.
200    fn help_row(&self) -> Option<&Row<'cmd>> {
201        self.rows.iter().find(|row| self.is_help_subcommand(row.cmd))
202    }
203
204    /// Check whether `cmd` is the `help` subcommand clap generated for the parent.
205    fn is_help_subcommand(&self, cmd: &Command) -> bool {
206        !self.parent.is_disable_help_subcommand_set() && cmd.get_name() == "help"
207    }
208
209    /// The rows of the subcommands filed under `category` (`None` for the uncategorized ones),
210    /// minus clap's `help`.
211    fn members(&self, category: Option<&Category>) -> impl Iterator<Item = &Row<'cmd>> {
212        let label = category.map(|cat| cat.label);
213        self.rows.iter().filter(move |row| {
214            !self.is_help_subcommand(row.cmd)
215                && self.category_map.find(row.cmd.get_name()).map(|cat| cat.label) == label
216        })
217    }
218
219    /// `parent`'s subcommands as clap lists them under `Commands:`, their abouts rendered by
220    /// `renderer`, laid out for an after-help wrapped to `width`: none when clap lists none, as
221    /// when only `help` would show.
222    fn new<R>(
223        parent: &'cmd Command,
224        category_map: CommandCategoryMap,
225        renderer: &R,
226        width: Option<usize>,
227    ) -> Self
228    where
229        R: TextRenderer,
230    {
231        let lists_subcommands =
232            parent.get_subcommands().any(|sub| sub.get_name() != "help" && !sub.is_hide_set());
233        let rows = if lists_subcommands {
234            Row::all_of(parent, &Row::listing(parent, renderer, width))
235        } else {
236            Vec::new()
237        };
238        Self { category_map, parent, rows }
239    }
240}

242/// One subcommand's lines in clap's listing: its name and flags, then its about and aliases,
243/// wrapped and aligned as clap does.
244struct Row<'cmd> {
245    cmd: &'cmd Command,
246    lines: String,
247}
248
249impl<'cmd> Row<'cmd> {
250    /// `listing`, clap's listing of `parent`'s visible subcommands, split into each one's row.
251    fn all_of(parent: &'cmd Command, listing: &str) -> Vec<Self> {
252        let mut rows: Vec<Self> = Vec::new();
253        for line in listing.split('\n') {
254            // A row starts at clap's two-space indent; the lines it wraps onto are indented past.
255            let starts_row = Word(line)
256                .visible()
257                .strip_prefix("  ")
258                .is_some_and(|rest| rest.chars().next().is_some_and(|first| first != ' '));
259            let cmd = starts_row.then(|| Self::subcommand_on(parent, line)).flatten();
260            match (cmd, rows.last_mut()) {
261                (Some(cmd), _) => rows.push(Self { cmd, lines: line.to_owned() }),
262                (None, Some(row)) => {
263                    row.lines.push('\n');
264                    row.lines.push_str(line);
265                }
266                (None, None) => {}
267            }
268        }
269        rows
270    }
271
272    /// clap's listing of `parent`'s visible subcommands, their abouts rendered by `renderer`, laid
273    /// out so that clap's wrap of an after-help to `width` keeps it.
274    fn listing<R>(parent: &Command, renderer: &R, width: Option<usize>) -> String
275    where
276        R: TextRenderer,
277    {
278        parent
279            .clone()
280            .mut_subcommands(|sub| {
281                // clap lists the long about when there is no short one.
282                let about = sub.get_about().or_else(|| sub.get_long_about());
283                let about = about.map(|about| renderer.restyle_line(about));
284                sub.about(Resettable::from(about))
285            })
286            .help_template(format!("{{subcommands}}{}", Menu::LISTING_END))
287            .render_for_after_help(|mut listed| {
288                let listing = listed.render_help().ansi().to_string();
289                let listing =
290                    listing.rsplit_once(Menu::LISTING_END).map_or(&*listing, |(rows, _)| rows);
291                Word(&renderer.restyle_wrapped(listing)).with_wide_lines_unstyled(width)
292            })
293    }
294
295    /// The cells clap gives `cmd`'s name and flags.
296    fn name_width(cmd: &Command) -> usize {
297        let short =
298            cmd.get_short_flag().map_or(0, |short| ", -".len().saturating_add(short.len_utf8()));
299        let long = cmd.get_long_flag().map_or(0, |long| ", --".len().saturating_add(long.width()));
300        cmd.get_name().width().saturating_add(short).saturating_add(long)
301    }
302
303    /// The subcommand of `parent` whose row starts with `line`: the longest name it starts with,
304    /// followed by its flags, its about or nothing.
305    fn subcommand_on(parent: &'cmd Command, line: &str) -> Option<&'cmd Command> {
306        let visible = Word(line).visible();
307        let rest = visible.strip_prefix("  ")?;
308        parent
309            .get_subcommands()
310            .filter(|sub| {
311                rest.strip_prefix(sub.get_name())
312                    .is_some_and(|after| after.is_empty() || after.starts_with([',', ' ']))
313            })
314            .max_by_key(|sub| sub.get_name().len())
315    }
316}
317/// A category heading, optionally followed by its description.
318struct HeadingView<'view, R> {
319    category: &'static Category,
320    view: &'view MenuView<'view, R>,
321}
322
323impl<R> Display for HeadingView<'_, R>
324where
325    R: TextRenderer,
326{
327    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
328        let header = self.view.styles.get_header();
329        let heading = self.category.heading();
330        write!(f, "{header}{heading}{header:#}")?;
331        self.category.description.as_ref().map_or(Ok(()), |description| {
332            let mut rendered = String::new();
333            self.view.renderer.render_str(&mut rendered, &description.get())?;
334            // clap counts the blank line above a heading as a cell of the heading's line.
335            let width = self.view.layout.width.map(|width| width.saturating_sub(1));
336            let beside = self.view.layout.description_col;
337            let first_word = rendered.split(' ').next().map_or(0, |word| Word(word).width());
338            let fits_beside = width.is_none_or(|width| beside.saturating_add(first_word) <= width);
339            // As clap puts an about below its name when there is no room for it beside.
340            let col = if fits_beside {
341                write!(f, "{:pad$}", "", pad = beside.saturating_sub(heading.width()))?;
342                beside
343            } else {
344                write!(f, "\n{:col$}", "", col = MenuLayout::NEXT_LINE_COL)?;
345                MenuLayout::NEXT_LINE_COL
346            };
347            write!(f, "{}", ColumnText { col, text: &rendered, width })
348        })
349    }
350}

352/// Column positions shared by every section of a menu.
353struct MenuLayout {
354    /// Where a category description starts.
355    description_col: usize,
356    /// The width lines wrap to; `None` when they don't.
357    width: Option<usize>,
358}
359
360impl MenuLayout {
361    /// Where clap starts a help on the line below its name.
362    const NEXT_LINE_COL: usize = 10;
363
364    /// The layout for rows with names and flags `name_widths` cells wide, under `headings`.
365    fn measure(
366        name_widths: impl Iterator<Item = usize>,
367        headings: impl Iterator<Item = Cow<'static, str>>,
368        width: Option<usize>,
369    ) -> Self {
370        // clap aligns abouts two cells past the widest name, and never narrower than `-x`.
371        let name_width = name_widths.max().unwrap_or(0).max(2);
372        let heading_width = headings.map(|h| h.width()).max().unwrap_or(0);
373
374        // A description at or next to the about column reads as another row, so push it two
375        // cells past.
376        let about_col = name_width.saturating_add(4);
377        let description_col = heading_width.saturating_add(2);
378        let description_col = if description_col.abs_diff(about_col) < 2 {
379            about_col.saturating_add(2)
380        } else {
381            description_col
382        };
383
384        Self { description_col, width }
385    }
386}

388/// A command's visible subcommands, grouped under their category headings.
389///
390/// ```text
391///   logs    Tail live logs
392///   status  Show recent activity
393///
394/// Deploy:     Ship code to production
395///   deploy  Ship the current project
396///
397/// Account:
398///   login   Sign in to your account
399///
400///   help    Print this message or the help of the given subcommand(s)
401/// ```
402struct MenuView<'cmd, R> {
403    layout: MenuLayout,
404    menu: Menu<'cmd>,
405    renderer: R,
406    styles: &'cmd Styles,
407}
408
409impl<'cmd, R> MenuView<'cmd, R>
410where
411    R: TextRenderer,
412{
413    fn new(
414        parent: &'cmd Command,
415        categories: CommandCategoryMap,
416        renderer: R,
417        width: Option<usize>,
418    ) -> Self {
419        let menu = Menu::new(parent, categories, &renderer, width);
420        let layout = MenuLayout::measure(
421            menu.rows.iter().map(|row| Row::name_width(row.cmd)),
422            menu.categories().map(Category::heading),
423            width,
424        );
425        Self { layout, menu, renderer, styles: parent.get_styles() }
426    }
427}
428
429impl<R> Display for MenuView<'_, R>
430where
431    R: TextRenderer,
432{
433    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
434        let uncategorized =
435            self.menu.has_uncategorized().then_some(SectionView { category: None, view: self });
436        let categorized = self
437            .menu
438            .categories()
439            .map(|category| SectionView { category: Some(category), view: self });
440        let has_categories = self.menu.categories().next().is_some();
441        write_joined(f, uncategorized.into_iter().chain(categorized), "\n\n")?;
442        self.menu.help_row().map_or(Ok(()), |row| {
443            // clap's `help` ends the uncategorized rows, or follows the categories a section apart.
444            let sep = match (has_categories, self.menu.has_uncategorized()) {
445                (true, _) => "\n\n",
446                (false, true) => "\n",
447                (false, false) => "",
448            };
449            write!(f, "{sep}{}", row.lines)
450        })
451    }
452}

454/// Rendered text starting at `col`, broken between words to fit `width`, each line after the
455/// first indented to `col`.
456struct ColumnText<'text> {
457    col: usize,
458    text: &'text str,
459    width: Option<usize>,
460}
461
462impl Display for ColumnText<'_> {
463    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
464        let Some(width) = self.width else {
465            return f.write_str(self.text);
466        };
467        let indent = self.col;
468        let mut column = indent;
469        // The escapes in effect, reopened past the indent of each new line.
470        let mut open = String::new();
471        for (index, word) in self.text.split(' ').map(Word).enumerate() {
472            if index > 0 {
473                if column.saturating_add(1).saturating_add(word.width()) > width {
474                    if !open.is_empty() {
475                        f.write_str(Word::RESET)?;
476                    }
477                    write!(f, "\n{:indent$}{open}", "")?;
478                    column = indent;
479                } else {
480                    f.write_str(" ")?;
481                    column = column.saturating_add(1);
482                }
483            }
484            f.write_str(word.0)?;
485            column = column.saturating_add(word.width());
486            for escape in word.escapes() {
487                if escape == Word::RESET {
488                    open.clear();
489                } else {
490                    open.push_str(escape);
491                }
492            }
493        }
494        Ok(())
495    }
496}

498/// A run of rendered text between spaces, its ANSI SGR escapes taking no cells.
499struct Word<'text>(&'text str);
500
501impl<'text> Word<'text> {
502    /// The escape that ends every style.
503    const RESET: &'static str = "\u{1b}[0m";
504
505    /// The escapes in this word, in order.
506    fn escapes(&self) -> impl Iterator<Item = &'text str> {
507        let text = self.0;
508        text.match_indices('\u{1b}').map(move |(start, _)| {
509            let escape = text.get(start..).unwrap_or_default();
510            escape.find('m').and_then(|end| escape.get(..=end)).unwrap_or(escape)
511        })
512    }
513
514    /// The pieces of this text shown on screen, between its escapes.
515    fn shown(&self) -> impl Iterator<Item = &'text str> {
516        self.0.split('\u{1b}').enumerate().map(|(index, piece)| {
517            if index == 0 { piece } else { piece.split_once('m').map_or("", |(_, text)| text) }
518        })
519    }
520
521    /// This text without its escapes.
522    fn visible(&self) -> String {
523        self.shown().collect()
524    }
525
526    /// The cells this word takes on screen.
527    fn width(&self) -> usize {
528        self.shown().map(UnicodeWidthStr::width).sum()
529    }
530
531    /// Lines clap wrapped, each style still open at a line's end closed there and opened again
532    /// past the next line's indent, so the indent is never styled.
533    fn with_styles_reopened(&self) -> String {
534        let mut reopened = String::with_capacity(self.0.len());
535        // The escapes in effect at the end of the line so far.
536        let mut open = String::new();
537        for (index, line) in self.0.split('\n').enumerate() {
538            if index > 0 {
539                if !open.is_empty() {
540                    reopened.push_str(Self::RESET);
541                }
542                reopened.push('\n');
543                let text = line.trim_start_matches(' ');
544                reopened.push_str(line.get(..line.len().saturating_sub(text.len())).unwrap_or(""));
545                reopened.push_str(&open);
546                reopened.push_str(text);
547            } else {
548                reopened.push_str(line);
549            }
550            for escape in Word(line).escapes() {
551                if escape == Self::RESET {
552                    open.clear();
553                } else {
554                    open.push_str(escape);
555                }
556            }
557        }
558        reopened
559    }
560
561    /// Lines clap wrapped, each one wider than `width` without its styles: clap's wrap of an
562    /// after-help breaks a line it can't fit around its styles, but leaves an unstyled word whole.
563    fn with_wide_lines_unstyled(&self, width: Option<usize>) -> String {
564        let fits = |line: &str| {
565            // clap counts the blank line above a line opening with a style as one of its cells.
566            let blank_line_cell = usize::from(line.starts_with('\u{1b}'));
567            let line_width = Word(line).visible().trim_end().width();
568            width.is_none_or(|width| line_width.saturating_add(blank_line_cell) <= width)
569        };
570        if self.0.split('\n').all(fits) {
571            return self.0.to_owned();
572        }
573        // Each line closes its own styles, so dropping one line's leaves the others'.
574        let reopened = self.with_styles_reopened();
575        let lines = reopened.split('\n').map(|line| {
576            if fits(line) { Cow::Borrowed(line) } else { Cow::Owned(Word(line).visible()) }
577        });
578        lines.collect::<Vec<_>>().join("\n")
579    }
580}

582/// A category's heading and rows, or just the rows for uncategorized commands.
583struct SectionView<'view, R> {
584    category: Option<&'static Category>,
585    view: &'view MenuView<'view, R>,
586}
587
588impl<R> Display for SectionView<'_, R>
589where
590    R: TextRenderer,
591{
592    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
593        self.category.map_or(Ok(()), |category| {
594            let heading = HeadingView { category, view: self.view };
595            writeln!(f, "{heading}")
596        })?;
597        let rows = self.view.menu.members(self.category).map(|row| &row.lines);
598        write_joined(f, rows, "\n")
599    }
600}

602/// Writes each item of `items` to `out`, separated by `sep`.
603fn write_joined<W, I>(out: &mut W, items: I, sep: &str) -> fmt::Result
604where
605    W: fmt::Write,
606    I: IntoIterator,
607    I::Item: Display,
