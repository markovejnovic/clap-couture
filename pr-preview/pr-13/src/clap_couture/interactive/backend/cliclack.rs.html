20impl Backend for Cliclack {
21    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
22        Theme::from(prompt.styles).scoped(|| {
23            ::cliclack::confirm(prompt.question).initial_value(prompt.default).interact()
24        })
25    }

27    /// The waiting hint sits in cliclack's gutter, and the closed editor leaves the question
28    /// with the answer's first line, as a submitted input does.
29    #[cfg(feature = "interactive-editor")]
30    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
31        use ::cliclack::{Theme as _, ThemeState};
32
33        let theme = Theme::from(prompt.styles);
34        // cliclack's gutter bar, `│`, as its footer draws it under a submitted prompt.
35        let footer = theme.format_footer(&ThemeState::Submit);
36        let bar = footer.trim_end();
37        let waiting = if prompt.waiting.is_empty() {
38            String::new()
39        } else {
40            format!("{bar}  {}", prompt.waiting)
41        };
42        let hinted = EditorPrompt { waiting: &waiting, ..*prompt };
43        let answer = super::EditorInteraction::start(self, &hinted)
44            .and_then(super::EditorInteraction::answer);
45        let transcript = match Edited::of(&answer, prompt.optional) {
46            Edited::Answered(line) => {
47                let input = if line.is_empty() {
48                    String::new()
49                } else {
50                    let line = theme.input_style(&ThemeState::Submit).apply_to(line);
51                    format!("{bar}  {line}\n")
52                };
53                let header = theme.format_header(&ThemeState::Submit, prompt.question);
54                format!("{header}{input}{footer}")
55            }
56            Edited::Cancelled => {
57                format!("{}{footer}", theme.format_header(&ThemeState::Cancel, prompt.question))
58            }
59            Edited::Failed => return answer,
60        };
61        // Best effort: after a hang-up there is no terminal left to draw on.
62        drop(console::Term::stderr().write_str(&transcript));
63        answer
64    }

66    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
67        Theme::from(prompt.styles).scoped(|| {
68            prompt
69                .options
70                .iter()
71                .fold(::cliclack::multiselect(prompt.question), |select, option| {
72                    let hint = option.get_help().map(ToString::to_string).unwrap_or_default();
73                    select.item(option.get_name().to_owned(), option.get_name(), hint)
74                })
75                .initial_values(prompt.defaults.to_vec())
76                // TODO(markovejnovic): cliclack says "Input required" on an empty selection and
77                // can't be told to say `prompt.selection_required` instead.
78                .required(!prompt.optional)
79                .interact()
80        })
81    }
82
83    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
84        Theme::from(styles).scoped(|| ::cliclack::log::error(error))
85    }

87    fn secret(&self, prompt: &SecretPrompt<'_>) -> Result<Option<SecretString>, PromptError> {
88        let answer = Theme::from(prompt.styles).scoped(|| {
89            let input = ::cliclack::password(prompt.question);
90            let mut input = if prompt.optional { input.allow_empty() } else { input };
91            input.interact()
92        })?;
93        Ok((!answer.is_empty()).then(|| SecretString::from(answer)))
94    }
95
96    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
97        Theme::from(prompt.styles).scoped(|| {
98            let select = prompt.options.iter().fold(
99                ::cliclack::select(prompt.question),
100                |select, option| {
101                    let hint = option.get_help().map(ToString::to_string).unwrap_or_default();
102                    select.item(Some(option.get_name().to_owned()), option.get_name(), hint)
103                },
104            );
105            let mut select =
106                if prompt.optional { select.item(None, prompt.none_option, "") } else { select };
107            match prompt.default {
108                Some(default) => select.initial_value(Some(default.to_owned())).interact(),
109                None => select.interact(),
110            }
111        })
112    }
113
114    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
115        let answer: String = Theme::from(prompt.styles).scoped(|| {
116            let input = ::cliclack::input(prompt.question).required(!prompt.optional);
117            // TODO(markovejnovic): cliclack's `Input` can't start out holding editable text (no
118            // equivalent of inquire's `with_initial_value`), so `initial` is passed as a default
119            // instead. The cost: a list prompt pre-filled from the arg's defaults can't be
120            // finished early, as Enter on it keeps the default instead of ending the list. Once
121            // cliclack supports starting text, pass `initial` that way and show the hint even
122            // when there's a default.
123            //
124            // When there is a default, the hint is left off: cliclack shows the hint where it
125            // would show "(default)", hiding it, and Enter wouldn't finish the list anyway.
126            let mut input = match (prompt.default.or(prompt.initial), prompt.hint) {
127                (Some(default), _) => input.default_input(default),
128                (None, Some(hint)) => input.placeholder(hint),
129                (None, None) => input,
130            };
131            input.interact()
132        })?;
133        Ok(Some(answer).filter(|answer| !answer.is_empty()))
134    }
135}

137/// The command's styles on cliclack's input line.
138///
139/// By default, [`::cliclack`] uses global theming, which is somewhat inconvenient for the use-case
140/// of following clap's theme. The [`Theme::scoped`] function allows you to run a function with a
141/// temporarily globally set theme.
142struct Theme {
143    input: console::Style,
144    placeholder: console::Style,
145}
146
147impl Theme {
148    /// Run `interact` with this as cliclack's theme.
149    fn scoped<T>(self, interact: impl FnOnce() -> io::Result<T>) -> Result<T, PromptError> {
150        ::cliclack::set_theme(self);
151        let answer = interact();
152        ::cliclack::reset_theme();
153        answer.map_err(PromptError::from)
154    }
155}
156
157impl From<CommandStyles<'_>> for Theme {
158    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
159        Self {
160            input: console::Style::from(ClapStyle(styles.get_literal())),
161            placeholder: console::Style::from(ClapStyle(styles.get_placeholder())),
162        }
163    }
164}
165
166impl ::cliclack::Theme for Theme {
167    fn input_style(&self, _state: &::cliclack::ThemeState) -> console::Style {
168        self.input.clone()
169    }
170
171    fn placeholder_style(&self, _state: &::cliclack::ThemeState) -> console::Style {
172        self.placeholder.clone()
173    }
