1//! Parse, ask for the prompted args that got no value, then parse again with the answers.
2//!
3//! Every answer goes back into the command as its arg's `default_values`, so clap converts and
4//! validates it exactly like a typed value. Each answer is checked with a trial parse as soon as
5//! it is given: the previous trial passed, so a failure can only come from the new answer, which
6//! is then asked again with clap's message. For an arg asked one value at a time, each value is
7//! an answer of its own, checked on top of the values before it. clap holds no default to
8//! `num_args`, so the answer is held to it here, as one time the arg is passed.
9//!
10//! Confirmations come last, so their questions can name the answers. A declined one fails the
11//! parse; without a terminal to ask on, one left unanswered fails it as a missing required arg.
12//!
13//! A secret's answer goes into a vault instead, and clap gets a placeholder that the secret's
14//! value parser swaps back. It skips the trial parse: that parser takes any string, so there is
15//! nothing to ask again.
16//!
17//! An arg passed `-` that `dash` makes ask is asked like one left out, but its answer fills the
18//! `-` through its value parser instead, as the command line beats a default.
19//!
20//! Before any of it, the `#[couture(stdin)]` arg on the path the user took takes stdin: all of it
21//! when passed `-`, else what is piped, if it got no value. It is then no longer asked.
22

83pub(super) trait BinNamed {

88impl BinNamed for Command {
89    fn bin_named(self, argv: &[OsString]) -> Self {
90        // As clap's own parse names it: a multicall command or one set without a binary name is
91        // named by clap alone, and a name set already is kept.
92        if self.is_multicall_set() || self.is_no_binary_name_set() || self.get_bin_name().is_some()
93        {
94            return self;
95        }
96        let program_name = argv.first().and_then(|called| crate::os::program_name(called));
97        match program_name.and_then(OsStr::to_str) {
98            Some(bin_name) => self.bin_name(bin_name.to_owned()),
99            None => self,
100        }
101    }

104/// [`Command`], asked whether its args conflict.
105trait Conflicting {
106    /// Check whether clap refuses `together` passed at once. Parses a `--{id}` flag per arg, with
107    /// the conflicts and exclusivity of those in `together`, beside `self`'s groups as declared:
108    /// clap has no getter for a group's own conflicts. `inherited` are globals a group may name.
109    fn refuses_together(&self, together: &[&str], inherited: &[&Arg]) -> bool;
110}
111
112impl Conflicting for Command {
113    fn refuses_together(&self, together: &[&str], inherited: &[&Arg]) -> bool {
114        let mut args: Vec<&Arg> = self.get_arguments().collect();
115        for global in inherited {
116            if !args.iter().any(|arg| arg.get_id() == global.get_id()) {
117                args.push(global);
118            }
119        }
120        let flags = args.iter().map(|arg| {
121            let flag = Arg::new(arg.get_id().clone())
122                .long(arg.get_id().clone())
123                .action(ArgAction::SetTrue)
124                .exclusive(arg.is_exclusive_set());
125            if !together.iter().any(|id| arg.get_id() == id) {
126                return flag;
127            }
128            let conflicts: Vec<Id> = self
129                .get_arg_conflicts_with(arg)
130                .into_iter()
131                .map(Arg::get_id)
132                .filter(|conflict| args.iter().any(|arg| arg.get_id() == *conflict))
133                .cloned()
134                .collect();
135            flag.conflicts_with_all(conflicts)
136        });
137        Self::new("conflicts")
138            .no_binary_name(true)
139            .disable_help_flag(true)
140            .disable_version_flag(true)
141            .args(flags)
142            .groups(self.get_groups().cloned())
143            .try_get_matches_from(together.iter().map(|id| format!("--{id}")))
144            .is_err_and(|err| err.kind() == ErrorKind::ArgumentConflict)
145    }
146}

175impl Session<'_> {
176    /// Ask for `prompted_arg` until clap accepts the answer; `None` when an optional single-value
177    /// one is left empty. `current` is the arg's default, split into its values. `later` are the
178    /// prompted args still to be asked after it. A secret is asked once and comes back as its vault
179    /// placeholder.
180    fn answer(
181        &self,
182        prompted_arg: &PromptedArg,
183        current: &[String],
184        answers: &[Answer<'_>],
185        later: &[&PromptedArg],
186    ) -> Result<Option<Vec<String>>, PromptError> {
187        let Some(arg) = arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
188        else {
189            return Ok(None);
190        };
191        // An arg passed `-` was asked for by the user, so it takes an answer.
192        let dashed = self.dashed.asks(&prompted_arg.subcommand_path, prompted_arg.attr.arg_id);
193        let optional = !dashed && !self.required(answers, later);
194        let question = self.question(prompted_arg);
195        if prompted_arg.attr.secret.is_some() {
196            // clap has nothing to reject in a string, so a secret skips the trial parse; the vault
197            // keeps it and clap gets a placeholder. `current` goes unused: the derive refuses a
198            // default on a secret, and showing one would echo it.
199            let secret = self.backend.secret(&SecretPrompt {
200                optional,
201                question: &question,
202                styles: CommandStyles(self.cmd.get_styles()),
203            })?;
204            let placeholder = secret.map(|secret| vec![self.vault.store(secret)]);
205            // Every other answer fills its `-` in its trial parse, which a secret skips.
206            if let Some(placeholder) = &placeholder {
207                self.dash_args.fill(prompted_arg, placeholder);
208            }
209            return Ok(placeholder);
210        }
211        // A flag has nothing to edit, so one that slipped past the derive asks as a flag.
212        #[cfg(feature = "interactive-editor")]
213        if let Some(editor) = prompted_arg
214            .attr
215            .editor
216            .as_ref()
217            .filter(|_| arg.get_action().takes_values() && !takes_many(arg))
218        {
219            let asked = Asked {
220                arg,
221                current: current.first().map(String::as_str),
222                optional,
223                prompted_arg,
224                question: &question,
225            };
226            return self.in_editor(&asked, editor, answers);
227        }
228        let options: Vec<PossibleValue> =
229            arg.get_possible_values().into_iter().filter(|value| !value.is_hide_set()).collect();
230        let count = ValueCount::of(arg);
231        if !takes_many(arg) {
232            return self.until_accepted(answers, prompted_arg, count, || {
233                let value = self.ask(arg, &question, &options, current.first(), optional)?;
234                Ok(value.map(|value| vec![value]))
235            });
236        }
237        if options.is_empty() {
238            return self
239                .each_value(&question, prompted_arg, current, answers, optional, count)
240                .map(Some);
241        }
242        self.until_accepted(answers, prompted_arg, count, || {
243            self.backend
244                .multi_select(&MultiSelectPrompt {
245                    defaults: current,
246                    optional,
247                    options: &options,
248                    question: &question,
249                    selection_required: &self.messages.selection_required.get(),
250                    styles: CommandStyles(self.cmd.get_styles()),
251                })
252                .map(Some)
253        })
254    }
255
256    fn ask(
257        &self,
258        arg: &Arg,
259        question: &str,
260        options: &[PossibleValue],
261        current: Option<&String>,
262        optional: bool,
263    ) -> Result<Option<String>, PromptError> {
264        let styles = CommandStyles(self.cmd.get_styles());
265        if let Some(negate) = flag_negation(arg.get_action()) {
266            let set = self.backend.confirm(&ConfirmPrompt { default: false, question, styles })?;
267            return Ok(Some((set != negate).to_string()));
268        }
269        if options.is_empty() {
270            return self.backend.text(&TextPrompt {
271                default: current.map(String::as_str),
272                hint: None,
273                initial: None,
274                optional,
275                question,
276                styles,
277            });
278        }
279        self.backend.select(&SelectPrompt {
280            default: current.map(String::as_str),
281            none_option: &self.messages.none.get(),
282            optional,
283            options,
284            question,
285            styles,
286        })
287    }

289    /// Ask in the user's editor, starting from the rejected answer, else the arg's default, else
290    /// the editor's `initial` text. A text arg keeps the newline the answer ends in. An empty
291    /// answer to a required arg aborts, as git aborts on an empty commit message, and so does one
292    /// to an arg with a default, which an empty answer would quietly bring back.
293    #[cfg(feature = "interactive-editor")]
294    fn ask_in_editor(
295        &self,
296        asked: &Asked<'_>,
297        editor: &EditorAttr,
298        rejected: Option<&Rejected>,
299    ) -> Result<Option<String>, PromptError> {
300        use core::any::TypeId;
301
302        let Asked { arg, current, optional: _, prompted_arg, question } = *asked;
303        let styles = CommandStyles(self.cmd.get_styles());
304        let optional = asked.optional && arg.get_default_values().is_empty();
305        let instructions = if optional {
306            self.messages.editor_instructions_optional.get()
307        } else {
308            self.messages.editor_instructions.get()
309        };
310        let initial = editor.initial.as_ref().map(Text::get);
311        let answer = self.backend.editor(&EditorPrompt {
312            initial: rejected
313                .map(|rejected| rejected.value.as_str())
314                .or(current)
315                .or(initial.as_deref()),
316            error: rejected.map(|rejected| rejected.error.as_str()),
317            extension: editor.extension,
318            file_stem: &iter::once(self.cmd.get_name())
319                .chain(prompted_arg.subcommand_path.iter().copied())
320                .chain(iter::once(prompted_arg.attr.arg_id))
321                .collect::<Vec<_>>()
322                .join("-"),
323            instructions: &instructions,
324            not_utf8: &self.messages.editor_not_utf8.get(),
325            optional,
326            question,
327            returned_early: &self.messages.editor_returned_early.get(),
328            styles,
329            unreadable: &self.messages.editor_unreadable.get(),
330            waiting: &self.messages.editor_waiting.get(),
331        })?;
332        if answer.is_none() && !optional {
333            self.backend.report(&self.messages.editor_empty.get(), styles)?;
334            return Err(PromptError::Cancelled);
335        }
336        // The file's last newline is part of a text, not of a number, a flag or a path.
337        let value_type = arg.get_value_parser().type_id();
338        let takes_text = [TypeId::of::<String>(), TypeId::of::<OsString>()]
339            .iter()
340            .any(|text| value_type == *text);
341        Ok(answer.map(|mut answer| {
342            if !takes_text && answer.ends_with('\n') {
343                answer.pop();
344            }
345            answer
346        }))
347    }

349    /// Ask whether to go ahead, with each `{id}` in the question filled in from the args as
350    /// passed and answered so far.
351    fn confirm(
352        &self,
353        prompted_arg: &PromptedArg,
354        answers: &[Answer<'_>],
355    ) -> Result<bool, PromptError> {
356        let question = self.question(prompted_arg);
357        let answered = inject(
358            self.relaxed(self.prompted_args),
359            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice())),
360        )
361        .try_get_matches_from(self.argv);
362        let question = match answered
363            .as_ref()
364            .ok()
365            .and_then(|answered| matches_at(answered, &prompted_arg.subcommand_path))
366        {
367            Some(level) => interpolate(&question, |id| {
368                let path = &prompted_arg.subcommand_path;
369                let values = self
370                    .dashed
371                    .text(path, id)
372                    .map(|text| vec![text])
373                    .or_else(|| level.try_get_raw(id).ok().map(|_| raw_values(level, id)))?;
374                if self.is_secret(path, id) {
375                    return Some(values.iter().map(|_| secrets::MASKED.to_owned()).collect());
376                }
377                Some(values)
378            }),
379            None => question,
380        };
381        self.backend.confirm(&ConfirmPrompt {
382            default: false,
383            question: &question,
384            styles: CommandStyles(self.cmd.get_styles()),
385        })
386    }

388    /// Ask for one value at a time until an empty answer or `count`'s most, each starting out as
389    /// the default's value at its position. Only a required arg's first value may not be empty,
390    /// and an empty answer short of `count`'s least is asked again with clap's message.
391    ///
392    /// The default goes in as initial text rather than as the answer Enter accepts, so clearing it
393    /// still finishes the list.
394    fn each_value(
395        &self,
396        question: &str,
397        prompted_arg: &PromptedArg,
398        current: &[String],
399        answers: &[Answer<'_>],
400        optional: bool,
401        count: Option<ValueCount<'_>>,
402    ) -> Result<Vec<String>, PromptError> {
403        let finish_hint = self.messages.finish_hint.get();
404        let mut values = Vec::new();
405        loop {
406            if count.is_some_and(|count| !count.has_room(values.len())) {
407                return Ok(values);
408            }
409            let short = count.and_then(|count| count.check(&values).err());
410            let optional = !values.is_empty() || optional;
411            let Some(value) = self.backend.text(&TextPrompt {
412                default: None,
413                hint: (optional && short.is_none()).then_some(&*finish_hint),
414                initial: current.get(values.len()).map(String::as_str),
415                optional,
416                question,
417                styles: CommandStyles(self.cmd.get_styles()),
418            })?
419            else {
420                let Some(err) = short else { return Ok(values) };
421                self.backend
422                    .report(&summary(&err, &values), CommandStyles(self.cmd.get_styles()))?;
423                continue;
424            };
425            values.push(value);
426            if let Err(err) = self.trial(answers, prompted_arg, &values) {
427                self.backend
428                    .report(&summary(&err, &values), CommandStyles(self.cmd.get_styles()))?;
429                values.pop();
430            }
431        }
432    }

434    /// Ask in the editor until clap accepts the answer, reopening with the rejected one.
435    #[cfg(feature = "interactive-editor")]
436    fn in_editor(
437        &self,
438        asked: &Asked<'_>,
439        editor: &EditorAttr,
440        answers: &[Answer<'_>],
441    ) -> Result<Option<Vec<String>>, PromptError> {
442        let mut rejected = None;
443        loop {
444            let Some(value) = self.ask_in_editor(asked, editor, rejected.as_ref())? else {
445                return Ok(None);
446            };
447            let values = vec![value];
448            match self.trial(answers, asked.prompted_arg, &values) {
449                Ok(()) => return Ok(Some(values)),
450                Err(err) => {
451                    let error = summary(&err, &values);
452                    self.backend.report(&error, CommandStyles(self.cmd.get_styles()))?;
453                    rejected = values.into_iter().next().map(|value| Rejected { error, value });
454                }
455            }
456        }
457    }

459    /// Check whether arg `id` at `path` is a secret, its own or a global one from a level above.
460    fn is_secret(&self, path: &[&str], id: &str) -> bool {
461        self.secret_args.iter().any(|secret_arg| {
462            secret_arg.attr.arg_id == id && path.starts_with(&secret_arg.subcommand_path)
463        })
464    }

466    /// The `= ...` override, else the arg's help text, else its id, rendered.
467    fn question(&self, prompted_arg: &PromptedArg) -> String {
468        let question = prompted_arg.attr.question.as_ref().map_or_else(
469            || {
470                arg_at(self.cmd, &prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
471                    .and_then(Arg::get_help)
472                    .map_or_else(|| prompted_arg.attr.arg_id.to_owned(), ToString::to_string)
473            },
474            |question| question.get().into_owned(),
475        );
476        self.cmd.render_question(&question)
477    }

479    /// `self.cmd` with `prompted_args` relaxed, and those of them the user left out excused.
480    fn relaxed<'prompted_arg>(
481        &self,
482        prompted_args: impl IntoIterator<Item = &'prompted_arg PromptedArg> + Clone,
483    ) -> Command {
484        excuse(relax(self.cmd.clone(), prompted_args.clone()), self.passed, prompted_args)
485    }
486
487    /// Check whether leaving the arg being asked unset fails the parse: it is `required`, or
488    /// required through `required_unless_present`, a required group or a passed arg's `requires`.
489    /// The accepted answers go in, and only the args still to be asked, `later`, stay relaxed, so
490    /// a group another of them can still satisfy doesn't make this one required.
491    fn required(&self, answers: &[Answer<'_>], later: &[&PromptedArg]) -> bool {
492        let accepted =
493            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice()));
494        let answered = answers.iter().map(|answer| answer.prompted_arg);
495        let cmd =
496            excuse(inject(self.relaxed(later.iter().copied()), accepted), self.passed, answered);
497        cmd.try_get_matches_from(self.argv)
498            .is_err_and(|err| err.kind() == ErrorKind::MissingRequiredArgument)
499    }
500
501    /// Parse with `candidate` as `prompted_arg`'s answer on top of the accepted ones.
502    fn trial(
503        &self,
504        answers: &[Answer<'_>],
505        prompted_arg: &PromptedArg,
506        candidate: &[String],
507    ) -> Result<(), clap::Error> {
508        let accepted =
509            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice()));
510        let with_candidate: Vec<_> =
511            accepted.chain(iter::once((prompted_arg, candidate))).collect();
512        for (answered, values) in &with_candidate {
513            self.dash_args.fill(answered, values);
514        }
515        let cmd = inject(self.relaxed(self.prompted_args), with_candidate);
516        cmd.try_get_matches_from(self.argv).map(drop)
517    }
518
519    /// Call `ask` until clap accepts its answer and it fits `count`, showing clap's complaint
520    /// before each retry.
521    fn until_accepted(
522        &self,
523        answers: &[Answer<'_>],
524        prompted_arg: &PromptedArg,
525        count: Option<ValueCount<'_>>,
526        mut ask: impl FnMut() -> Result<Option<Vec<String>>, PromptError>,
527    ) -> Result<Option<Vec<String>>, PromptError> {
528        loop {
529            let Some(values) = ask()? else { return Ok(None) };
530            let fits = count.map_or(Ok(()), |count| count.check(&values));
531            match fits.and_then(|()| self.trial(answers, prompted_arg, &values)) {
532                Ok(()) => return Ok(Some(values)),
533                Err(err) => {
534                    let error = summary(&err, &values);
535                    self.backend.report(&error, CommandStyles(self.cmd.get_styles()))?;
536                }
537            }
538        }
539    }

542impl<'arg> ValueCount<'arg> {
543    /// Check `values` as clap checks them typed after the arg once; no values leave it unset.
544    pub(super) fn check(self, values: &[String]) -> Result<(), clap::Error> {
545        let (min, max) = (self.range.min_values(), self.range.max_values());
546        if let Some(surplus) = values.get(max) {
547            return Err(self.error(ErrorKind::TooManyValues, [(
548                ContextKind::InvalidValue,
549                ContextValue::String(surplus.clone()),
550            )]));
551        }
552        if values.is_empty() || values.len() >= min {
553            return Ok(());
554        }
555        let number =
556            |count: usize| ContextValue::Number(isize::try_from(count).unwrap_or(isize::MAX));
557        let actual = (ContextKind::ActualNumValues, number(values.len()));
558        // clap's own wording: an exact count when there is one, else the least it takes.
559        Err(if min == max {
560            self.error(ErrorKind::WrongNumberOfValues, [
561                actual,
562                (ContextKind::ExpectedNumValues, number(min)),
563            ])
564        } else {
565            self.error(ErrorKind::TooFewValues, [actual, (ContextKind::MinValues, number(min))])
566        })
567    }
568
569    /// clap's `kind` of error about the arg, with `context` filled in.
570    fn error(
571        self,
572        kind: ErrorKind,
573        context: impl IntoIterator<Item = (ContextKind, ContextValue)>,
574    ) -> clap::Error {
575        let mut err = clap::Error::new(kind);
576        err.insert(ContextKind::InvalidArg, ContextValue::String(self.arg.to_string()));
577        for (kind, value) in context {
578            err.insert(kind, value);
579        }
580        err
581    }
582
583    /// Check whether `count` values leave room for another.
584    fn has_room(self, count: usize) -> bool {
585        count < self.range.max_values()
586    }

588    /// `arg`'s `num_args`; `None` when it takes one value each time and may be passed again, as a
589    /// plain `Vec` does, so any number of values is one it could be typed with.
590    pub(super) fn of(arg: &'arg Arg) -> Option<Self> {
591        let range = arg.get_num_args()?;
592        let repeats_single_values =
593            matches!(arg.get_action(), ArgAction::Append) && range.max_values() <= 1;
594        (!repeats_single_values).then_some(Self { arg, range })
595    }

598/// Parse `argv` against `cmd`, filling its stdin arg from `stdin`, asking `backend` for the
599/// prompted args that got no value, and standing in for each `-` a dash arg got what was asked or
600/// read for it. The help clap prints is `help`.
601pub(super) fn parse<T>(
602    cmd: Command,
603    prompted_arg_tree: &'static PromptedArgTree,
604    messages: &Messages,
605    argv: &[OsString],
606    backend: &dyn Backend,
607    stdin: &impl Stdin,
608    help: &DeferredHelp,
609) -> Result<T, ParseError>
610where
611    T: FromArgMatches,
612{
613    let cmd = cmd.bin_named(argv);
614    let mut prompted_args = prompted_arg_tree.prompted_args();
615    let stdin_args = StdinArgs::of(prompted_arg_tree);
616    let dash_args = DashArgs::of(&prompted_args, &stdin_args);
617    let nothing_to_ask = prompted_args.is_empty() || !backend.is_available();
618    if nothing_to_ask && stdin_args.is_empty() && dash_args.is_empty() {
619        return parse_unprompted(cmd, &prompted_args, argv, help);
620    }
621    // Re-guarded with the vault this parse's answers go into; trial and final parses share it.
622    let vault = Vault::default();
623    let mut cmd = secrets::guard(cmd, &prompted_args, &vault, &messages.environment.get());
624    let relaxed = stdin_args.unrequired(relax(dash_args.checking(cmd.clone()), &prompted_args));
625    // Prompts read the terminal, so a `-` that may ask or read stdin reads it when it is no
626    // terminal, as when piped.
627    let can_ask = stdin.is_terminal() && backend.is_available();
628    // Help, usage and errors come from clap, with `-` let through.
629    let Ok(passed) = parse_relaxed(relaxed, &prompted_args, argv) else {
630        return super::parse(dash_args.checking(cmd), argv, help);
631    };
632    let dashed = dash_args.passed(&passed, can_ask);
633    // `-` is the only way to ask a `prompt(dash)` arg.
634    prompted_args.retain(|prompted_arg| {
635        prompted_arg.attr.dash.is_none()
636            || dashed.asks(&prompted_arg.subcommand_path, prompted_arg.attr.arg_id)
637    });
638    if let Some(asked) = dashed.first_asked().filter(|_| !backend.is_available()) {
639        return Err(asked.unaskable(&cmd));
640    }
641    let checking = dash_args.checking(cmd.clone());
642    if let Some(stdin_value) = stdin_args.read(&checking, &prompted_args, &dashed, argv, stdin)? {
643        cmd = stdin_value.fill(cmd, &mut prompted_args, &dashed, &vault);
644    }
645    if prompted_args.is_empty() || !backend.is_available() {
646        return parse_unprompted(dash_args.typed(cmd), &prompted_args, argv, help);
647    }
648    let checking = dash_args.checking(cmd.clone());
649    // Taken before any were dropped from `prompted_args`: a secret read from stdin is masked too.
650    let secret_args: Vec<PromptedArg> = prompted_arg_tree
651        .prompted_args()
652        .into_iter()
653        .filter(|prompted_arg| prompted_arg.attr.secret.is_some())
654        .collect();
655    // Help, usage and errors come from the command the user sees, not the relaxed one.
656    let relaxed = excuse(relax(checking.clone(), &prompted_args), &passed, &prompted_args);
657    let Ok(matches) = relaxed.try_get_matches_from(argv) else {
658        return super::parse(checking, argv, help);
659    };
660
661    let session = Session {
662        argv,
663        backend,
664        cmd: &checking,
665        dash_args: &dash_args,
666        dashed: &dashed,
667        messages,
668        passed: &passed,
669        prompted_args: &prompted_args,
670        secret_args: &secret_args,
671        vault: &vault,
672    };
673    let mut answers = Vec::new();
674    let mut unanswered = pending(&checking, &matches, &prompted_args, &dashed);
675    // Stable, so values and confirmations each keep declaration order.
676    unanswered
677        .sort_by_key(|(prompted_arg, _)| prompted_arg.attr.purpose == PromptPurpose::Confirmation);
678    for (index, (prompted_arg, current)) in unanswered.iter().enumerate() {
679        let (path, id) = (prompted_arg.subcommand_path.as_slice(), prompted_arg.attr.arg_id);
680        if blocked(&checking, &matches, path, id, &answers) {
681            continue;
682        }
683        let later: Vec<&PromptedArg> =
684            unanswered.iter().skip(index.saturating_add(1)).map(|(later, _)| *later).collect();
685        let answer = match prompted_arg.attr.purpose {
686            PromptPurpose::Confirmation => match session.confirm(prompted_arg, &answers) {
687                Ok(false) => return Err(ParseError::Declined),
688                confirmed => confirmed.map(|_| Some(Vec::from(["true".to_owned()]))),
689            },
690            PromptPurpose::Value => session.answer(prompted_arg, current, &answers, &later),
691        };
692        match answer {
693            Ok(Some(injected_values)) => answers.push(Answer { injected_values, prompted_arg }),
694            Ok(None) => {}
695            // Backing out counts as Ctrl-C.
696            Err(PromptError::Cancelled) => return Err(ParseError::Cancelled(Signal::Interrupt)),
697            Err(PromptError::Signalled(signal)) => return Err(ParseError::Cancelled(signal)),
698            Err(PromptError::Io(err)) => return Err(ParseError::Io(err)),
699            Err(PromptError::NotATerminal) => {
700                if let Some(asked) = dashed.first_asked() {
701                    return Err(asked.unaskable(&cmd));
702                }
703                return parse_unprompted(dash_args.typed(cmd), &prompted_args, argv, help);
704            }
705        }
706    }
707    let cmd_with_answers = excuse(
708        inject(
709            cmd,
710            answers.iter().map(|answer| (answer.prompted_arg, answer.injected_values.as_slice())),
711        ),
712        &passed,
713        answers.iter().map(|answer| answer.prompted_arg),
714    );
715    super::parse(dash_args.typed(cmd_with_answers), argv, help)
716}

718/// Parse as clap would, without asking. A confirmation left unanswered fails the parse with
719/// clap's error for a missing required arg, so only passing its flag gets past it. The help clap
720/// prints is `help`.
721fn parse_unprompted<T>(
722    mut cmd: Command,
723    prompted_args: &[PromptedArg],
724    argv: &[OsString],
725    help: &DeferredHelp,
726) -> Result<T, ParseError>
727where
728    T: FromArgMatches,
729{
730    let confirmations: Vec<_> = prompted_args
731        .iter()
732        .filter(|prompted_arg| prompted_arg.attr.purpose == PromptPurpose::Confirmation)
733        .collect();
734    if confirmations.is_empty() {
735        return super::parse(cmd, argv, help);
736    }
737    // Kept unbuilt: requiring an arg on the built command makes clap show help instead.
738    let declared = cmd.clone();
739    let mut matches = help.matches(&mut cmd, argv).map_err(ParseError::Clap)?;
740    let unconfirmed = confirmations.iter().any(|confirmation| {
741        let id = confirmation.attr.arg_id;
742        let passed = matches_at(&matches, &confirmation.subcommand_path)
743            .is_none_or(|level| explicit(level, &Id::from(id)));
744        !passed && !blocked(&cmd, &matches, &confirmation.subcommand_path, id, &[])
745    });
746    if unconfirmed {
747        // Parsed again only to fail: clap words the error, and the help stays as declared.
748        let required = confirmations.iter().fold(declared, |cmd, confirmation| {
749            with_arg_at(cmd, &confirmation.subcommand_path, confirmation.attr.arg_id, |arg| {
750                arg.required(true)
751            })
752        });
753        return super::parse(required, argv, help);
754    }
755    T::from_arg_matches_mut(&mut matches).map_err(|err| ParseError::Clap(err.format(&mut cmd)))
756}

762/// Check whether arg `id` of the command at `path` conflicts with an arg the user passed or an
763/// earlier answer.
764///
765/// Answers go in as default values, which clap never checks for conflicts, so asking here
766/// would let the parse land in a state clap rejects when typed.
767pub(super) fn blocked(
768    cmd: &Command,
769    matches: &ArgMatches,
770    path: &[&str],
771    id: &str,
772    answers: &[Answer<'_>],
773) -> bool {
774    let (Some(level), Some(level_matches)) = (command_at(cmd, path), matches_at(matches, path))
775    else {
776        return false;
777    };
778    if !level.get_arguments().any(|arg| arg.get_id() == id) {
779        return false;
780    }
781    let answered = |other: &Id| {
782        answers.iter().any(|answer| {
783            answer.prompted_arg.subcommand_path == path && answer.prompted_arg.attr.arg_id == other
784        })
785    };
786    let together: Vec<&str> = level
787        .get_arguments()
788        .map(Arg::get_id)
789        .filter(|other| *other != id)
790        .filter(|other| answered(other) || explicit(level_matches, other))
791        .map(Id::as_str)
792        .chain(iter::once(id))
793        .collect();
794    // Alone, it conflicts with nothing.
795    if together.len() < 2 {
796        return false;
797    }
798    let inherited: Vec<&Arg> = (0..path.len())
799        .filter_map(|depth| command_at(cmd, path.get(..depth)?))
800        .flat_map(Command::get_arguments)
801        .filter(|arg| arg.is_global_set())
802        .collect();
803    level.refuses_together(&together, &inherited)
804}

806/// `arg`'s declared default values, split at its value delimiter.
807/// Without a delimiter, each default is one value. Defaults are set from strings, so any that
808/// is not UTF-8 is skipped.
809pub(super) fn default_values(arg: &Arg) -> impl Iterator<Item = &str> {
810    let delimiter = arg.get_value_delimiter();
811    arg.get_default_values()
812        .iter()
813        .filter_map(|value| value.to_str())
814        .flat_map(move |value| value.split(move |char| Some(char) == delimiter))
815}

835/// Let each of `prompted_args` the user left out in `passed` go missing though a passed arg
836/// `requires` it: clap lets a required arg go missing beside a passed arg it conflicts with, so it
837/// conflicts with every other arg of its command. Having no value from the user, it never trips
838/// that conflict. A global one is left as is, as its conflicts would reach subcommands.
839fn excuse<'prompted_arg>(
840    cmd: Command,
841    passed: &ArgMatches,
842    prompted_args: impl IntoIterator<Item = &'prompted_arg PromptedArg>,
843) -> Command {
844    prompted_args.into_iter().fold(cmd, |cmd, prompted_arg| {
845        let (path, id) = (prompted_arg.subcommand_path.as_slice(), prompted_arg.attr.arg_id);
846        if matches_at(passed, path).is_some_and(|level| explicit(level, &Id::from(id))) {
847            return cmd;
848        }
849        with_command_at(cmd, path, |level| {
850            let others: Vec<Id> = level
851                .get_arguments()
852                .map(Arg::get_id)
853                .filter(|other| *other != id)
854                .cloned()
855                .collect();
856            with_arg_at(level, &[], id, |arg| {
857                if arg.is_global_set() { arg } else { arg.conflicts_with_all(others) }
858            })
859        })
860    })
861}

896pub(super) fn inject_at(cmd: Command, path: &[&str], id: &str, values: &[String]) -> Command {

904/// `question` with each `{id}` replaced by arg `id`'s values from `values_of`, joined with `, `,
905/// and `{{` and `}}` by single braces. A `{id}` naming no arg stays as written.
906fn interpolate(question: &str, values_of: impl Fn(&str) -> Option<Vec<String>>) -> String {
907    let mut interpolated = String::with_capacity(question.len());
908    let mut rest = question;
909    while let Some(brace) = rest.find(['{', '}']) {
910        let (text, tail) = rest.split_at(brace);
911        interpolated.push_str(text);
912        let escaped = tail.strip_prefix("{{").map(|after| ('{', after));
913        if let Some((brace, after)) =
914            escaped.or_else(|| tail.strip_prefix("}}").map(|after| ('}', after)))
915        {
916            interpolated.push(brace);
917            rest = after;
918            continue;
919        }
920        let placeholder = tail.strip_prefix('{').and_then(|inner| inner.split_once('}'));
921        if let Some((values, after)) =
922            placeholder.and_then(|(id, after)| values_of(id).map(|values| (values, after)))
923        {
924            interpolated.push_str(&values.join(", "));
925            rest = after;
926            continue;
927        }
928        // A lone brace, or a `{id}` naming no arg: keep the brace as written.
929        let (brace, after) = tail.split_at(1);
930        interpolated.push_str(brace);
931        rest = after;
932    }
933    interpolated.push_str(rest);
934    interpolated
935}

946/// The prompted args on the path the user took that got no value or only their default, or `-`
947/// to be asked for, with the default's values, split at the arg's value delimiter as clap splits
948/// them. One whose `default_value_if` fired is left to the arg that fired it: the conditional
949/// default beats an injected answer, so asking would drop the answer.
950fn pending<'prompted_arg>(
951    cmd: &Command,
952    matches: &ArgMatches,
953    prompted_args: &'prompted_arg [PromptedArg],
954    dashed: &Dashed<'_>,
955) -> Vec<(&'prompted_arg PromptedArg, Vec<String>)> {
956    // Built, so a flag's action has filled in its implicit default (`false` for `SetTrue`).
957    let mut built = cmd.clone();
958    built.build();
959    prompted_args
960        .iter()
961        .filter_map(|prompted_arg| {
962            let (path, id) = (&prompted_arg.subcommand_path, prompted_arg.attr.arg_id);
963            // Passed `-`, it has a value, but one the user asked to be asked for.
964            if dashed.asks(path, id) {
965                return Some((prompted_arg, Vec::new()));
966            }
967            unfilled(&built, matches, path, id).map(|current| (prompted_arg, current))
968        })
969        .collect()
970}

972/// The values of arg `id` of the command at `path`, when it is on the path the user took and got
973/// no value or only its default; see [`pending`]. `built` must be built.
974pub(super) fn unfilled(
975    built: &Command,
976    matches: &ArgMatches,
977    path: &[&str],
978    id: &str,
979) -> Option<Vec<String>> {
980    let arg = arg_at(built, path, id)?;
981    let level = matches_at(matches, path)?;
982    match level.value_source(id) {
983        None if arg.is_required_by_default() || default_values(arg).next().is_none() => {
984            Some(Vec::new())
985        }
986        Some(ValueSource::DefaultValue) => {
987            let current = raw_values(level, id);
988            current.iter().map(String::as_str).eq(default_values(arg)).then_some(current)
989        }
990        None | Some(_) => None,
991    }
992}

1021/// Parse `argv` with `relaxed`, whose `prompted_args` are relaxed. When a passed arg requires one
1022/// left out, it is parsed again with those excused, found by a parse that sees past clap's errors.
1023fn parse_relaxed(
1024    relaxed: Command,
1025    prompted_args: &[PromptedArg],
1026    argv: &[OsString],
1027) -> Result<ArgMatches, clap::Error> {
1028    match relaxed.clone().try_get_matches_from(argv) {
1029        Err(err) if err.kind() == ErrorKind::MissingRequiredArgument => {
1030            let passed = relaxed.clone().ignore_errors(true).try_get_matches_from(argv)?;
1031            excuse(relaxed, &passed, prompted_args).try_get_matches_from(argv)
1032        }
1033        parsed => parsed,
1034    }
1035}

1037/// The first line of clap's message, without its `error: ` prefix. clap quotes the rejected
1038/// `values`, so a multi-line one is shortened to its first line first, keeping the reason after it.
1039pub(super) fn summary(err: &clap::Error, values: &[String]) -> String {
1040    let mut rendered = err.render().to_string();
1041    for value in values {
1042        if let Some((first_line, _)) = value.split_once('\n') {
1043            rendered = rendered.replace(value.as_str(), &format!("{first_line}\u{2026}"));
1044        }
1045    }
1046    let line = rendered.lines().next().unwrap_or_default();
1047    line.strip_prefix("error: ").unwrap_or(line).to_owned()
1048}

1050/// Drop the rules that make arg `id` of the command at `path` required: its `required`, its
1051/// `required_unless_present`, a secret's required default, and the `required` of the groups it
1052/// belongs to. clap can't clear `required_unless_present_all` or `required_if_eq`, so those stay.
1053pub(super) fn unrequire(cmd: Command, path: &[&str], id: &str) -> Command {
1054    with_command_at(cmd, path, |level| {
1055        if !level.get_arguments().any(|arg| arg.get_id() == id) {
1056            return level;
1057        }
1058        let groups: Vec<Id> = level
1059            .get_groups()
1060            .filter(|group| group.is_required_set() && group.get_args().any(|arg| arg == id))
1061            .map(|group| group.get_id().clone())
1062            .collect();
1063        let level = level.mut_arg(id, |arg| {
1064            arg.required(false)
1065                .required_unless_present(Resettable::Reset)
1066                .without_required_default()
1067        });
1068        groups
1069            .iter()
1070            .fold(level, |level, group| level.mut_group(group, |group| group.required(false)))
1071    })
1072}

1083            level
1084        }
1085    })
