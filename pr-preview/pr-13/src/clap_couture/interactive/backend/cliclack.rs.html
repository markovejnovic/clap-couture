16impl Backend for Cliclack {
17    fn confirm(&self, prompt: &ConfirmPrompt<'_>) -> Result<bool, PromptError> {
18        Theme::from(prompt.styles).scoped(|| {
19            ::cliclack::confirm(prompt.question).initial_value(prompt.default).interact()
20        })
21    }

23    /// The waiting hint sits in cliclack's gutter, and the closed editor leaves the question
24    /// with the answer's first line, as a submitted input does.
25    #[cfg(feature = "interactive-editor")]
26    fn editor(&self, prompt: &EditorPrompt<'_>) -> Result<Option<String>, PromptError> {
27        use ::cliclack::{Theme as _, ThemeState};
28
29        let theme = Theme::from(prompt.styles);
30        // cliclack's gutter bar, `│`, as its footer draws it under a submitted prompt.
31        let footer = theme.format_footer(&ThemeState::Submit);
32        let bar = footer.trim_end();
33        let waiting = if prompt.waiting.is_empty() {
34            String::new()
35        } else {
36            format!("{bar}  {}", prompt.waiting)
37        };
38        let hinted = EditorPrompt { waiting: &waiting, ..*prompt };
39        let answer = super::EditorInteraction::start(self, &hinted)
40            .and_then(super::EditorInteraction::answer);
41        let transcript = match Edited::of(&answer, prompt.optional) {
42            Edited::Answered(line) => {
43                let input = if line.is_empty() {
44                    String::new()
45                } else {
46                    let line = theme.input_style(&ThemeState::Submit).apply_to(line);
47                    format!("{bar}  {line}\n")
48                };
49                let header = theme.format_header(&ThemeState::Submit, prompt.question);
50                format!("{header}{input}{footer}")
51            }
52            Edited::Cancelled => {
53                format!("{}{footer}", theme.format_header(&ThemeState::Cancel, prompt.question))
54            }
55            Edited::Failed => return answer,
56        };
57        // Best effort: after a hang-up there is no terminal left to draw on.
58        drop(console::Term::stderr().write_str(&transcript));
59        answer
60    }

62    fn multi_select(&self, prompt: &MultiSelectPrompt<'_>) -> Result<Vec<String>, PromptError> {
63        Theme::from(prompt.styles).scoped(|| {
64            prompt
65                .options
66                .iter()
67                .fold(::cliclack::multiselect(prompt.question), |select, option| {
68                    let hint = option.get_help().map(ToString::to_string).unwrap_or_default();
69                    select.item(option.get_name().to_owned(), option.get_name(), hint)
70                })
71                .initial_values(prompt.defaults.to_vec())
72                // TODO(markovejnovic): cliclack says "Input required" on an empty selection and
73                // can't be told to say `prompt.selection_required` instead.
74                .required(!prompt.optional)
75                .interact()
76        })
77    }
78
79    fn report(&self, error: &str, styles: CommandStyles<'_>) -> Result<(), PromptError> {
80        Theme::from(styles).scoped(|| ::cliclack::log::error(error))
81    }

83    fn select(&self, prompt: &SelectPrompt<'_>) -> Result<Option<String>, PromptError> {
84        Theme::from(prompt.styles).scoped(|| {
85            let select = prompt.options.iter().fold(
86                ::cliclack::select(prompt.question),
87                |select, option| {
88                    let hint = option.get_help().map(ToString::to_string).unwrap_or_default();
89                    select.item(Some(option.get_name().to_owned()), option.get_name(), hint)
90                },
91            );
92            let mut select =
93                if prompt.optional { select.item(None, prompt.none_option, "") } else { select };
94            match prompt.default {
95                Some(default) => select.initial_value(Some(default.to_owned())).interact(),
96                None => select.interact(),
97            }
98        })
99    }
100
101    fn text(&self, prompt: &TextPrompt<'_>) -> Result<Option<String>, PromptError> {
102        let answer: String = Theme::from(prompt.styles).scoped(|| {
103            let input = ::cliclack::input(prompt.question).required(!prompt.optional);
104            // TODO(markovejnovic): cliclack's `Input` can't start out holding editable text (no
105            // equivalent of inquire's `with_initial_value`), so `initial` is passed as a default
106            // instead. The cost: a list prompt pre-filled from the arg's defaults can't be
107            // finished early, as Enter on it keeps the default instead of ending the list. Once
108            // cliclack supports starting text, pass `initial` that way and show the hint even
109            // when there's a default.
110            //
111            // When there is a default, the hint is left off: cliclack shows the hint where it
112            // would show "(default)", hiding it, and Enter wouldn't finish the list anyway.
113            let mut input = match (prompt.default.or(prompt.initial), prompt.hint) {
114                (Some(default), _) => input.default_input(default),
115                (None, Some(hint)) => input.placeholder(hint),
116                (None, None) => input,
117            };
118            input.interact()
119        })?;
120        Ok(Some(answer).filter(|answer| !answer.is_empty()))
121    }
122}

124/// The command's styles on cliclack's input line.
125///
126/// By default, [`::cliclack`] uses global theming, which is somewhat inconvenient for the use-case
127/// of following clap's theme. The [`Theme::scoped`] function allows you to run a function with a
128/// temporarily globally set theme.
129struct Theme {
130    input: console::Style,
131    placeholder: console::Style,
132}
133
134impl Theme {
135    /// Run `interact` with this as cliclack's theme.
136    fn scoped<T>(self, interact: impl FnOnce() -> io::Result<T>) -> Result<T, PromptError> {
137        ::cliclack::set_theme(self);
138        let answer = interact();
139        ::cliclack::reset_theme();
140        answer.map_err(PromptError::from)
141    }
142}
143
144impl From<CommandStyles<'_>> for Theme {
145    fn from(CommandStyles(styles): CommandStyles<'_>) -> Self {
146        Self {
147            input: console::Style::from(ClapStyle(styles.get_literal())),
148            placeholder: console::Style::from(ClapStyle(styles.get_placeholder())),
149        }
150    }
151}
152
153impl ::cliclack::Theme for Theme {
154    fn input_style(&self, _state: &::cliclack::ThemeState) -> console::Style {
155        self.input.clone()
156    }
157
158    fn placeholder_style(&self, _state: &::cliclack::ThemeState) -> console::Style {
159        self.placeholder.clone()
160    }
