21impl TextRenderer for MdTextRenderer<'_> {

23    where
24        W: fmt::Write,

30    fn render_str<W>(&self, out: &mut W, text: &str) -> fmt::Result
31    where
32        W: fmt::Write,
33    {
34        self.render_blocks(out, text, false)
35    }

37    fn restyle(&self, text: &StyledStr) -> StyledStr {
38        let mut rendered = StyledStr::new();
39        // `StyledStr`'s `fmt::Write` is infallible.
40        self.render_blocks(&mut rendered, &text.to_string(), true).ok();
41        rendered
42    }
43}

45impl MdTextRenderer<'_> {
46    /// Writes `text`, keeping paragraphs and list items apart when `keep_blocks`, else on one line.
47    fn render_blocks<W>(&self, out: &mut W, text: &str, keep_blocks: bool) -> fmt::Result
48    where
49        W: fmt::Write,
50    {
51        let mut line = LineWriter::new(out, text, self.styles.get_literal(), keep_blocks);
52        Parser::new(text).into_offset_iter().try_for_each(|(event, span)| line.write(event, span))
53    }
54}

56/// Writes markdown events as one line of styled text, keeping every character a reader needs.
57struct LineWriter<'out, 'src, W> {
58    /// What the pending break writes instead of a space, after a paragraph or list item.
59    block_break: Option<&'static str>,
60    /// Whether a block or line break awaits a space before the next text.
61    break_pending: bool,
62    /// Whether text is a code block's.
63    code_block: bool,
64    /// Each open link's or image's destination, `None` when its text already shows it.
65    destinations: Vec<Option<CowStr<'src>>>,
66    emphasis: usize,
67    /// What ends a list item: a newline when blocks are kept, else a space.
68    item_break: &'static str,
69    literal: Style,
70    out: &'out mut W,
71    /// What ends a paragraph or heading: a blank line when blocks are kept, else a space.
72    paragraph_break: &'static str,
73    source: &'src str,
74    strong: usize,
75    written: bool,
76}
77
78impl<'out, 'src, W> LineWriter<'out, 'src, W>
79where
80    W: fmt::Write,
81{
82    fn close(&mut self, tag: TagEnd) -> fmt::Result {
83        match tag {
84            TagEnd::Emphasis => {
85                self.emphasis = self.emphasis.saturating_sub(1);
86                Ok(())
87            }
88            TagEnd::Strong => {
89                self.strong = self.strong.saturating_sub(1);
90                Ok(())
91            }
92            TagEnd::Link | TagEnd::Image => {
93                let destination = self.destinations.pop().flatten();
94                destination.map_or(Ok(()), |url| self.write_lines(&format!(" ({url})"), None))
95            }
96            TagEnd::CodeBlock => {
97                self.code_block = false;
98                self.break_pending = true;
99                Ok(())
100            }
101            TagEnd::Paragraph | TagEnd::Heading(_) | TagEnd::Item => {
102                self.break_pending = true;
103                self.block_break = Some(if matches!(tag, TagEnd::Item) {
104                    self.item_break
105                } else {
106                    self.paragraph_break
107                });
108                Ok(())
109            }
110            TagEnd::BlockQuote(_)
111            | TagEnd::HtmlBlock
112            | TagEnd::List(_)
113            | TagEnd::FootnoteDefinition
114            | TagEnd::DefinitionList
115            | TagEnd::DefinitionListTitle
116            | TagEnd::DefinitionListDefinition
117            | TagEnd::Table
118            | TagEnd::TableHead
119            | TagEnd::TableRow
120            | TagEnd::TableCell
121            | TagEnd::MetadataBlock(_) => {
122                self.break_pending = true;
123                Ok(())
124            }
125            TagEnd::Strikethrough | TagEnd::Superscript | TagEnd::Subscript => Ok(()),
126        }
127    }
128
129    const fn new(out: &'out mut W, source: &'src str, literal: &Style, keep_blocks: bool) -> Self {
130        Self {
131            block_break: None,
132            break_pending: false,
133            code_block: false,
134            destinations: Vec::new(),
135            emphasis: 0,
136            item_break: if keep_blocks { "\n" } else { " " },
137            literal: *literal,
138            out,
139            paragraph_break: if keep_blocks { "\n\n" } else { " " },
140            source,
141            strong: 0,
142            written: false,
143        }
144    }
145
146    fn open(&mut self, tag: Tag<'src>, span: Range<usize>) -> fmt::Result {
147        match tag {
148            Tag::Emphasis => {
149                self.emphasis = self.emphasis.saturating_add(1);
150                Ok(())
151            }
152            Tag::Strong => {
153                self.strong = self.strong.saturating_add(1);
154                Ok(())
155            }
156            Tag::Link { link_type, dest_url, title: _, id: _ } => {
157                let shown = matches!(link_type, LinkType::Autolink | LinkType::Email);
158                self.destinations.push((!shown && !dest_url.is_empty()).then_some(dest_url));
159                Ok(())
160            }
161            Tag::Image { link_type: _, dest_url, title: _, id: _ } => {
162                self.destinations.push((!dest_url.is_empty()).then_some(dest_url));
163                Ok(())
164            }
165            // The marker (`-`, `3.`) is the item's first word in the source.
166            Tag::Item => {
167                self.break_pending = true;
168                let marker = self.source.get(span).and_then(|item| item.split_whitespace().next());
169                marker.map_or(Ok(()), |marker| self.write_lines(marker, None))?;
170                self.break_pending = true;
171                Ok(())
172            }
173            Tag::CodeBlock(_) => {
174                self.code_block = true;
175                self.break_pending = true;
176                Ok(())
177            }
178            Tag::Paragraph
179            | Tag::Heading { level: _, id: _, classes: _, attrs: _ }
180            | Tag::BlockQuote(_)
181            | Tag::HtmlBlock
182            | Tag::List(_)
183            | Tag::FootnoteDefinition(_)
184            | Tag::DefinitionList
185            | Tag::DefinitionListTitle
186            | Tag::DefinitionListDefinition
187            | Tag::Table(_)
188            | Tag::TableHead
189            | Tag::TableRow
190            | Tag::TableCell
191            | Tag::MetadataBlock(_) => {
192                self.break_pending = true;
193                Ok(())
194            }
195            Tag::Strikethrough | Tag::Superscript | Tag::Subscript => Ok(()),
196        }
197    }
198
199    /// Writes `event`, which `span` locates in the source.
200    fn write(&mut self, event: Event<'src>, span: Range<usize>) -> fmt::Result {
201        match event {
202            Event::Start(tag) => self.open(tag, span),
203            Event::End(tag) => self.close(tag),
204            Event::Text(text) if self.code_block => self.write_lines(&text, Some(self.literal)),
205            Event::Text(text) | Event::Html(text) | Event::InlineHtml(text) => {
206                self.write_lines(&text, None)
207            }
208            Event::Code(code) => self.write_lines(&code, Some(self.literal)),
209            Event::SoftBreak | Event::HardBreak => {
210                self.break_pending = true;
211                Ok(())
212            }
213            Event::Rule
214            | Event::InlineMath(_)
215            | Event::DisplayMath(_)
216            | Event::FootnoteReference(_)
217            | Event::TaskListMarker(_) => self.write_source(span),
218        }
219    }
220
221    /// Writes `text` over `base` (plain when `None`), with each newline as a break.
222    fn write_lines(&mut self, text: &str, base: Option<Style>) -> fmt::Result {
223        text.split('\n').enumerate().try_for_each(|(index, line)| {
224            self.break_pending |= index > 0;
225            let line = line.trim_end_matches('\r');
226            if line.is_empty() {
227                return Ok(());
228            }
229            if self.break_pending && self.written {
230                self.out.write_str(self.block_break.unwrap_or(" "))?;
231            }
232            self.block_break = None;
233            self.break_pending = false;
234            self.written = true;
235            self.write_styled(line, base)
236        })
237    }
238
239    /// Writes the source `span` covers as is, set apart by breaks.
240    fn write_source(&mut self, span: Range<usize>) -> fmt::Result {
241        let source = self.source;
242        source.get(span).map_or(Ok(()), |text| {
243            self.break_pending = true;
244            self.write_lines(text.trim(), None)?;
245            self.break_pending = true;
246            Ok(())
247        })
248    }
249
250    fn write_styled(&mut self, text: &str, base: Option<Style>) -> fmt::Result {
251        let mut style = base.unwrap_or_default();
252        if self.strong > 0 {
253            style = style.bold();
254        }
255        if self.emphasis > 0 {
256            style = style.italic();
257        }
258        if style == Style::new() {
259            self.out.write_str(text)
260        } else {
261            write!(self.out, "{style}{text}{style:#}")
262        }
263    }
264}

272    fn rendered(text: &str) -> String {
273        let styles = Styles::plain();
274        let mut out = String::new();
275        let written = MdTextRenderer::new(&styles).render_str(&mut out, text);
276        assert!(written.is_ok(), "writing to a `String` can't fail");
277        out
